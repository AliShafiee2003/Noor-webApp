   419→          q.set("pg", current);
   420→          window.history.replaceState(null, "", `${base}?${q.toString()}`);
   421→        }
   422→      });
   423→    };
   424→    main.addEventListener("scroll", onScroll, { passive: true });
   425→    return () => {
   426→      main.removeEventListener("scroll", onScroll);

  1240→/* ---------- سند روایت‌های جمع‌شده (HAD-10/HAD-14) — وضعیت در سطح نما ---------- */
  1241→function useCollapsedSanads() {
  1242→  const [collapsed, setCollapsed] = useState<ReadonlySet<string>>(new Set());
  1243→  const toggle = useCallback((blockId: string) => {
  1244→    setCollapsed((prev) => {
  1245→      const next = new Set(prev);
  1246→      if (next.has(blockId)) next.delete(blockId);
  1247→      else next.add(blockId);
  1248→      return next;
  1249→    });
  1250→  }, []);
  1251→  return { collapsed, toggle };
  1252→}

  1262→function parseHighlight(raw: string | null): [number, number] | null {
  1263→  if (!raw) return null;
  1264→  const parts = raw.split(":");
  1265→  const s = Number.parseInt(parts[0] ?? "", 10);
  1266→  const e = Number.parseInt(parts[1] ?? "", 10);
  1267→  if (!Number.isFinite(s) || !Number.isFinite(e) || e <= s) return null;
  1268→  return [s, e];
  1269→}
