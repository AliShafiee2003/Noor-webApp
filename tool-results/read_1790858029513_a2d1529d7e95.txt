   237→// dbHandle اتصال فقط‌خواندنی نمایه؛ با تعویض فایل (نسل جدید) خودکار بازگشایی می‌شود.
   238→// تشخیص تعویض: inode + اندازه؛ و اگر release_uid اتصال با نسخهٔ درخواستی نخواند، بازگشایی اجباری.
   239→func (e *LocalEngine) dbHandle() (*sql.DB, error) {
   240→        e.dbMu.Lock()
   241→        defer e.dbMu.Unlock()
   242→        fi, err := osStat(e.path)
   243→        if err == nil {
   244→                ino := fileInode(fi)
   245→                sz := fi.Size()
   246→                if e.db != nil && (ino != e.dbInode || sz != e.dbSize) {
   247→                        e.db.Close()
   248→                        e.db = nil
   249→                }
   250→                if e.db == nil {
   251→                        db, err := sql.Open("sqlite", "file:"+e.path+"?mode=ro")
   252→                        if err != nil {
   253→                                return nil, err
   254→                        }
   255→                        db.SetMaxOpenConns(3)
   256→                        e.db = db
   257→                        e.dbInode = ino
   258→                        e.dbSize = sz
   259→                }
   260→                return e.db, nil
   261→        }
   262→        if e.db != nil {
   263→                return e.db, nil
   264→        }
   265→        return nil, errors.New("نمایهٔ جست‌وجو در دسترس نیست")
   266→}

   268→// ensureRelease اگر اتصالِ فعلی نسلِ درخواستی را ندارد، اتصال را نوسازی می‌کند
   269→// (مقاوم در برابر inode reuse بین نسل‌ها).
   270→func (e *LocalEngine) ensureRelease(db *sql.DB, releaseUID string) (*sql.DB, error) {
   271→        var rel string
   272→        err := db.QueryRow(`SELECT release_uid FROM passages LIMIT 1`).Scan(&rel)
   273→        if err == nil && rel == releaseUID {
   274→                return db, nil
   275→        }
   276→        // نوسازی اجباری
   277→        e.dbMu.Lock()
   278→        if e.db != nil {
   279→                e.db.Close()
   280→                e.db = nil
   281→        }
   282→        e.dbMu.Unlock()
   283→        return e.dbHandle()
   284→}

   288→// validateRequest قواعد مشترک اعتبارسنجی پرس‌وجو (هر دو موتور).
   289→func validateRequest(req *Request) error {
   290→        // جست‌وجوی صرفاً استنادی (بدون عبارت) مجاز است — QUR-03
   291→        if len([]rune(req.Query)) == 0 && req.Scope.Quran == nil {
   292→                return errors.New("عبارت جست‌وجو خالی است")
   293→        }
   294→        modeOK := false
   295→        for _, m := range []string{StrictPhrase, NormalizedPhrase, AllWords, AnyWords, Near, Linguistic, LongQuote} {
   296→                if req.Mode == m {
   297→                        modeOK = true
   298→                }
   299→        }
   300→        if !modeOK {
   301→                return fmt.Errorf("حالت جست‌وجو معتبر نیست")
   302→        }
   303→        limit := 2048
   304→        if req.Mode == LongQuote {
   305→                limit = 8192
   306→        }
   307→        if len([]rune(req.Query)) > limit {
   308→                return errors.New("عبارت بیش از حد مجاز است؛ برای نقل‌قول بلند از حالت مخصوص آن استفاده کنید")
   309→        }
   310→        if req.PageSize < 5 || req.PageSize > 50 {
   311→                req.PageSize = 20
   312→        }
   313→        return nil
   314→}

   316→// Search اجرای جست‌وجو: کاندید → verifier → نتایج تأییدشده.
   317→func (e *LocalEngine) Search(ctx context.Context, req *Request, releaseUID string) (*Response, error) {
   318→        if err := e.validate(req); err != nil {
   319→                return nil, err
   320→        }
   321→        if req.Scope.Target == "titles" {
   322→                hits, err := e.SearchTitles(ctx, req.Query, req.PageSize)
   323→                if err != nil {
   324→                        return nil, err
   325→                }
   326→                resp := &Response{ReleaseUID: releaseUID, CompletionState: "complete", CountRelation: "exact", Results: []Result{}}
   327→                for _, h := range hits {
   328→                        resp.Results = append(resp.Results, Result{
   329→                                PassageUID: "title-" + h.WorkUID, MatchType: "exact",
   330→                                Work: Ref{UID: h.WorkUID, Title: h.Title, Label: h.ShortTitle},
   331→                                Page: Ref{Label: "نام کتاب"},
   332→                        })
   333→                }
   334→                resp.VerifiedBooks = len(hits)
   335→                resp.VerifiedOccurrences = len(hits)
   336→                return resp, nil
   337→        }
   338→        db, err := e.dbHandle()
   339→        if err != nil {
   340→                return nil, err
   341→        }
   342→        db, err = e.ensureRelease(db, releaseUID)
   343→        if err != nil {
   344→                return nil, err
   345→        }
   346→        verified, capped, err := e.run(ctx, db, req, releaseUID)
   347→        if err != nil {
   348→                return nil, err
   349→        }
   350→        sid := newID("s")
   351→        if err := e.saveSession(sid, releaseUID, req); err != nil {
   352→                return nil, err
   353→        }
   354→        return e.buildResponse(sid, releaseUID, req, verified, 0, capped), nil
   355→}

   357→// Continue ادامهٔ بررسی با cursor امضاشده (§۴۷) — re-execute قطعی + skip موقعیتی.
   358→func (e *LocalEngine) Continue(ctx context.Context, sessionUID, cursor, releaseUID string) (*Response, error) {
   359→        if !verifyCursor(cursor, e.Secret) {
   360→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   361→        }
   362→        payload, ok := parseCursorPayload(cursor)
   363→        if !ok || payload.S != sessionUID {
   364→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   365→        }
   366→        release, req, err := e.loadSession(sessionUID)
   367→        if err != nil {
   368→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   369→        }
   370→        db, err := e.dbHandle()
   371→        if err != nil {
   372→                return nil, err
   373→        }
   374→        db, err = e.ensureRelease(db, release)
   375→        if err != nil {
   376→                return nil, err
   377→        }
   378→        verified, capped, err := e.run(ctx, db, req, release)
   379→        if err != nil {
   380→                return nil, err
   381→        }
   382→        _ = releaseUID // پرس‌وجو با release نشست ادامه می‌یابد تا نتایج صفحه‌ها هم‌خوان بمانند
   383→        return e.buildResponse(sessionUID, release, req, verified, payload.O, capped), nil
   384→}

   386→// cursorPayload محتوای امضاشدهٔ cursor: شناسهٔ نشست + موقعیت.
   387→type cursorPayloadT struct {
   388→        S string `json:"s"`
   389→        O int    `json:"o"`
   390→}
   391→
   392→func parseCursorPayload(cursor string) (cursorPayloadT, bool) {
   393→        var out cursorPayloadT
   394→        part := strings.SplitN(cursor, ".", 2)[0]
   395→        b, err := base64.RawURLEncoding.DecodeString(part)
   396→        if err != nil || json.Unmarshal(b, &out) != nil {
   397→                return out, false
   398→        }
   399→        return out, true
   400→}

   402→// buildResponse پاسخ از روی نتایج کامل اجرای جاری می‌سازد؛ صفحه از offset شروع می‌شود.
   403→func (e *LocalEngine) buildResponse(sid, release string, req *Request, verified []Result, offset int, capped bool) *Response {
   404→        if offset < 0 {
   405→                offset = 0
   406→        }
   407→        if offset > len(verified) {
   408→                offset = len(verified)
   409→        }
   410→        pageSize := req.PageSize
   411→        if pageSize < 5 || pageSize > 50 {
   412→                pageSize = 20
   413→        }
   414→        page := verified[offset:]
   415→        if len(page) > pageSize {
   416→                page = page[:pageSize]
   417→        }
   418→        books := map[string]bool{}
   419→        passages := map[string]bool{}
   420→        hadiths := map[string]bool{}
   421→        for _, r := range verified {
   422→                if w, _, ok := e.catalog.DocumentMeta(r.Document.UID); ok {
   423→                        books[w] = true
   424→                } else {
   425→                        books[r.Work.UID] = true
   426→                }
   427→                if r.PassageUID != "" {
   428→                        passages[r.PassageUID] = true
   429→                }
   430→                if r.LocalHadith != "" {
   431→                        // روایت درون هر سند متمایز است؛ شناسهٔ محلی جهانی نیست
   432→                        hadiths[r.Document.UID+"/"+r.LocalHadith] = true
   433→                }
   434→        }
   435→        newOffset := offset + len(page)
   436→        continuation := newOffset < len(verified)
   437→        rel := "exact"
   438→        if capped {
   439→                rel = "lower_bound" // سقف فعال شد؛ شمارش‌ها حداقل واقعی‌اند
   440→        }
   441→        resp := &Response{
   442→                SessionUID:            sid,
   443→                ReleaseUID:            release,
   444→                CompletionState:       "complete",
   445→                VerifiedOccurrences:   len(verified),
   446→                VerifiedBooks:         len(books),
   447→                VerifiedPassages:      len(passages),
   448→                VerifiedHadiths:       len(hadiths),
   449→                CountRelation:         rel,
   450→                ContinuationAvailable: continuation,
   451→                Results:               page,
   452→        }
   453→        if capped {
   454→                resp.Notice = fmt.Sprintf("نتایج بیشتر از سقف جست‌وجو است؛ شمارش‌ها حداقل واقعی‌اند (سقف %s موضع).", faNum(e.maxResults))
   455→        }
   456→        if continuation {
   457→                resp.NextCursor = signCursor(map[string]any{"s": sid, "o": newOffset}, e.Secret)
   458→        }
   459→        return resp
   460→}

   462→// BooksGrouped «این عبارت در چه کتاب‌هایی آمده؟» — فقط تأییدشده‌ها (§۴۷).
   463→// با re-execute از روی درخواست ذخیره‌شدهٔ نشست (بدون نگهداری نتایج در حافظه).
   464→func (e *LocalEngine) BooksGrouped(ctx context.Context, sessionUID string) ([]BookGroup, error) {
   465→        release, req, err := e.loadSession(sessionUID)
   466→        if err != nil {
   467→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   468→        }
   469→        db, err := e.dbHandle()
   470→        if err != nil {
   471→                return nil, err
   472→        }
   473→        db, err = e.ensureRelease(db, release)
   474→        if err != nil {
   475→                return nil, err
   476→        }
   477→        verified, _, err := e.run(ctx, db, req, release)
   478→        if err != nil {
   479→                return nil, err
   480→        }
   481→        workOrder := []string{}
   482→        groups := map[string]*BookGroup{}
   483→        docOrder := map[string][]string{}
   484→        docs := map[string]*DocEvidence{}
   485→        for _, r := range verified {
   486→                bg, ok2 := groups[r.Work.UID]
   487→                if !ok2 {
   488→                        bg = &BookGroup{Work: r.Work, Documents: []DocEvidence{}}
   489→                        groups[r.Work.UID] = bg
   490→                        workOrder = append(workOrder, r.Work.UID)
   491→                }
   492→                key := r.Work.UID + "|" + r.Document.UID
   493→                dg, ok3 := docs[key]
   494→                if !ok3 {
   495→                        dg = &DocEvidence{Document: r.Document, Evidence: []Result{}}
   496→                        docs[key] = dg
   497→                        docOrder[r.Work.UID] = append(docOrder[r.Work.UID], r.Document.UID)
   498→                }
   499→                dg.VerifiedOccurrences++
   500→                if len(dg.Evidence) < 3 {
   501→                        dg.Evidence = append(dg.Evidence, r)
   502→                }
   503→
   504→        }
   505→        var out []BookGroup
   506→        for _, wUID := range workOrder {
   507→                bg := groups[wUID]
   508→                bg.Documents = []DocEvidence{}
   509→                for _, dUID := range docOrder[wUID] {
   510→                        bg.Documents = append(bg.Documents, *docs[wUID+"|"+dUID])
   511→                }
   512→                out = append(out, *bg)
   513→        }
   514→        if out == nil {
   515→                out = []BookGroup{}
   516→        }
   517→        return out, nil
   518→}

   520→// ---- نشست جست‌وجو: ذخیرهٔ فقط «درخواست» — SQLite یا حافظه با تخلیهٔ per-session ----
   521→
   522→func (e *LocalEngine) saveSession(sid, releaseUID string, req *Request) error {
   523→        reqJSON, err := json.Marshal(req)
   524→        if err != nil {
   525→                return err
   526→        }
   527→        if e.sessStore != nil {
   528→                return e.sessStore.SaveSearchSession(sid, releaseUID, reqJSON, e.sessionTTL)
   529→        }
   530→        e.mu.Lock()
   531→        defer e.mu.Unlock()
   532→        e.evictMemSessionsLocked()
   533→        e.memSess[sid] = &memSession{release: releaseUID, reqJSON: reqJSON, expire: time.Now().Add(e.sessionTTL)}
   534→        return nil
   535→}
   536→
   537→func (e *LocalEngine) loadSession(sid string) (string, *Request, error) {
   538→        if e.sessStore != nil {
   539→                release, reqJSON, err := e.sessStore.LoadSearchSession(sid)
   540→                if err != nil {
   541→                        return "", nil, err

   570→        now := time.Now()
