   237→   237→// dbHandle اتصال فقط‌خواندنی نمایه؛ با تعویض فایل (نسل جدید) خودکار بازگشایی می‌شود.
   238→   238→// تشخیص تعویض: inode + اندازه؛ و اگر release_uid اتصال با نسخهٔ درخواستی نخواند، بازگشایی اجباری.
   239→   239→func (e *LocalEngine) dbHandle() (*sql.DB, error) {
   240→   240→        e.dbMu.Lock()
   241→   241→        defer e.dbMu.Unlock()
   242→   242→        fi, err := osStat(e.path)
   243→   243→        if err == nil {
   244→   244→                ino := fileInode(fi)
   245→   245→                sz := fi.Size()
   246→   246→                if e.db != nil && (ino != e.dbInode || sz != e.dbSize) {
   247→   247→                        e.db.Close()
   248→   248→                        e.db = nil
   249→   249→                }
   250→   250→                if e.db == nil {
   251→   251→                        db, err := sql.Open("sqlite", "file:"+e.path+"?mode=ro")
   252→   252→                        if err != nil {
   253→   253→                                return nil, err
   254→   254→                        }
   255→   255→                        db.SetMaxOpenConns(3)
   256→   256→                        e.db = db
   257→   257→                        e.dbInode = ino
   258→   258→                        e.dbSize = sz
   259→   259→                }
   260→   260→                return e.db, nil
   261→   261→        }
   262→   262→        if e.db != nil {
   263→   263→                return e.db, nil
   264→   264→        }
   265→   265→        return nil, errors.New("نمایهٔ جست‌وجو در دسترس نیست")
   266→   266→}

   268→   268→// ensureRelease اگر اتصالِ فعلی نسلِ درخواستی را ندارد، اتصال را نوسازی می‌کند
   269→   269→// (مقاوم در برابر inode reuse بین نسل‌ها).
   270→   270→func (e *LocalEngine) ensureRelease(db *sql.DB, releaseUID string) (*sql.DB, error) {
   271→   271→        var rel string
   272→   272→        err := db.QueryRow(`SELECT release_uid FROM passages LIMIT 1`).Scan(&rel)
   273→   273→        if err == nil && rel == releaseUID {
   274→   274→                return db, nil
   275→   275→        }
   276→   276→        // نوسازی اجباری
   277→   277→        e.dbMu.Lock()
   278→   278→        if e.db != nil {
   279→   279→                e.db.Close()
   280→   280→                e.db = nil
   281→   281→        }
   282→   282→        e.dbMu.Unlock()
   283→   283→        return e.dbHandle()
   284→   284→}

   288→   288→// validateRequest قواعد مشترک اعتبارسنجی پرس‌وجو (هر دو موتور).
   289→   289→func validateRequest(req *Request) error {
   290→   290→        // جست‌وجوی صرفاً استنادی (بدون عبارت) مجاز است — QUR-03
   291→   291→        if len([]rune(req.Query)) == 0 && req.Scope.Quran == nil {
   292→   292→                return errors.New("عبارت جست‌وجو خالی است")
   293→   293→        }
   294→   294→        modeOK := false
   295→   295→        for _, m := range []string{StrictPhrase, NormalizedPhrase, AllWords, AnyWords, Near, Linguistic, LongQuote} {
   296→   296→                if req.Mode == m {
   297→   297→                        modeOK = true
   298→   298→                }
   299→   299→        }
   300→   300→        if !modeOK {
   301→   301→                return fmt.Errorf("حالت جست‌وجو معتبر نیست")
   302→   302→        }
   303→   303→        limit := 2048
   304→   304→        if req.Mode == LongQuote {
   305→   305→                limit = 8192
   306→   306→        }
   307→   307→        if len([]rune(req.Query)) > limit {
   308→   308→                return errors.New("عبارت بیش از حد مجاز است؛ برای نقل‌قول بلند از حالت مخصوص آن استفاده کنید")
   309→   309→        }
   310→   310→        if req.PageSize < 5 || req.PageSize > 50 {
   311→   311→                req.PageSize = 20
   312→   312→        }
   313→   313→        return nil
   314→   314→}

   316→   316→// Search اجرای جست‌وجو: کاندید → verifier → نتایج تأییدشده.
   317→   317→func (e *LocalEngine) Search(ctx context.Context, req *Request, releaseUID string) (*Response, error) {
   318→   318→        if err := e.validate(req); err != nil {
   319→   319→                return nil, err
   320→   320→        }
   321→   321→        if req.Scope.Target == "titles" {
   322→   322→                hits, err := e.SearchTitles(ctx, req.Query, req.PageSize)
   323→   323→                if err != nil {
   324→   324→                        return nil, err
   325→   325→                }
   326→   326→                resp := &Response{ReleaseUID: releaseUID, CompletionState: "complete", CountRelation: "exact", Results: []Result{}}
   327→   327→                for _, h := range hits {
   328→   328→                        resp.Results = append(resp.Results, Result{
   329→   329→                                PassageUID: "title-" + h.WorkUID, MatchType: "exact",
   330→   330→                                Work: Ref{UID: h.WorkUID, Title: h.Title, Label: h.ShortTitle},
   331→   331→                                Page: Ref{Label: "نام کتاب"},
   332→   332→                        })
   333→   333→                }
   334→   334→                resp.VerifiedBooks = len(hits)
   335→   335→                resp.VerifiedOccurrences = len(hits)
   336→   336→                return resp, nil
   337→   337→        }
   338→   338→        db, err := e.dbHandle()
   339→   339→        if err != nil {
   340→   340→                return nil, err
   341→   341→        }
   342→   342→        db, err = e.ensureRelease(db, releaseUID)
   343→   343→        if err != nil {
   344→   344→                return nil, err
   345→   345→        }
   346→   346→        verified, capped, err := e.run(ctx, db, req, releaseUID)
   347→   347→        if err != nil {
   348→   348→                return nil, err
   349→   349→        }
   350→   350→        sid := newID("s")
   351→   351→        if err := e.saveSession(sid, releaseUID, req); err != nil {
   352→   352→                return nil, err
   353→   353→        }
   354→   354→        return e.buildResponse(sid, releaseUID, req, verified, 0, capped), nil
   355→   355→}

   357→   357→// Continue ادامهٔ بررسی با cursor امضاشده (§۴۷) — re-execute قطعی + skip موقعیتی.
   358→   358→func (e *LocalEngine) Continue(ctx context.Context, sessionUID, cursor, releaseUID string) (*Response, error) {
   359→   359→        if !verifyCursor(cursor, e.Secret) {
   360→   360→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   361→   361→        }
   362→   362→        payload, ok := parseCursorPayload(cursor)
   363→   363→        if !ok || payload.S != sessionUID {
   364→   364→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   365→   365→        }
   366→   366→        release, req, err := e.loadSession(sessionUID)
   367→   367→        if err != nil {
   368→   368→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   369→   369→        }
   370→   370→        db, err := e.dbHandle()
   371→   371→        if err != nil {
   372→   372→                return nil, err
   373→   373→        }
   374→   374→        db, err = e.ensureRelease(db, release)
   375→   375→        if err != nil {
   376→   376→                return nil, err
   377→   377→        }
   378→   378→        verified, capped, err := e.run(ctx, db, req, release)
   379→   379→        if err != nil {
   380→   380→                return nil, err
   381→   381→        }
   382→   382→        _ = releaseUID // پرس‌وجو با release نشست ادامه می‌یابد تا نتایج صفحه‌ها هم‌خوان بمانند
   383→   383→        return e.buildResponse(sessionUID, release, req, verified, payload.O, capped), nil
   384→   384→}

   386→   386→// cursorPayload محتوای امضاشدهٔ cursor: شناسهٔ نشست + موقعیت.
   387→   387→type cursorPayloadT struct {
   388→   388→        S string `json:"s"`
   389→   389→        O int    `json:"o"`
   390→   390→}
   391→   391→
   392→   392→func parseCursorPayload(cursor string) (cursorPayloadT, bool) {
   393→   393→        var out cursorPayloadT
   394→   394→        part := strings.SplitN(cursor, ".", 2)[0]
   395→   395→        b, err := base64.RawURLEncoding.DecodeString(part)
   396→   396→        if err != nil || json.Unmarshal(b, &out) != nil {
   397→   397→                return out, false
   398→   398→        }
   399→   399→        return out, true
   400→   400→}

   402→   402→// buildResponse پاسخ از روی نتایج کامل اجرای جاری می‌سازد؛ صفحه از offset شروع می‌شود.
   403→   403→func (e *LocalEngine) buildResponse(sid, release string, req *Request, verified []Result, offset int, capped bool) *Response {
   404→   404→        if offset < 0 {
   405→   405→                offset = 0
   406→   406→        }
   407→   407→        if offset > len(verified) {
   408→   408→                offset = len(verified)
   409→   409→        }
   410→   410→        pageSize := req.PageSize
   411→   411→        if pageSize < 5 || pageSize > 50 {
   412→   412→                pageSize = 20
   413→   413→        }
   414→   414→        page := verified[offset:]
   415→   415→        if len(page) > pageSize {
   416→   416→                page = page[:pageSize]
   417→   417→        }
   418→   418→        books := map[string]bool{}
   419→   419→        passages := map[string]bool{}
   420→   420→        hadiths := map[string]bool{}
   421→   421→        for _, r := range verified {
   422→   422→                if w, _, ok := e.catalog.DocumentMeta(r.Document.UID); ok {
   423→   423→                        books[w] = true
   424→   424→                } else {
   425→   425→                        books[r.Work.UID] = true
   426→   426→                }
   427→   427→                if r.PassageUID != "" {
   428→   428→                        passages[r.PassageUID] = true
   429→   429→                }
   430→   430→                if r.LocalHadith != "" {
   431→   431→                        // روایت درون هر سند متمایز است؛ شناسهٔ محلی جهانی نیست
   432→   432→                        hadiths[r.Document.UID+"/"+r.LocalHadith] = true
   433→   433→                }
   434→   434→        }
   435→   435→        newOffset := offset + len(page)
   436→   436→        continuation := newOffset < len(verified)
   437→   437→        rel := "exact"
   438→   438→        if capped {
   439→   439→                rel = "lower_bound" // سقف فعال شد؛ شمارش‌ها حداقل واقعی‌اند
   440→   440→        }
   441→   441→        resp := &Response{
   442→   442→                SessionUID:            sid,
   443→   443→                ReleaseUID:            release,
   444→   444→                CompletionState:       "complete",
   445→   445→                VerifiedOccurrences:   len(verified),
   446→   446→                VerifiedBooks:         len(books),
   447→   447→                VerifiedPassages:      len(passages),
   448→   448→                VerifiedHadiths:       len(hadiths),
   449→   449→                CountRelation:         rel,
   450→   450→                ContinuationAvailable: continuation,
   451→   451→                Results:               page,
   452→   452→        }
   453→   453→        if capped {
   454→   454→                resp.Notice = fmt.Sprintf("نتایج بیشتر از سقف جست‌وجو است؛ شمارش‌ها حداقل واقعی‌اند (سقف %s موضع).", faNum(e.maxResults))
   455→   455→        }
   456→   456→        if continuation {
   457→   457→                resp.NextCursor = signCursor(map[string]any{"s": sid, "o": newOffset}, e.Secret)
   458→   458→        }
   459→   459→        return resp
   460→   460→}

   464→   464→func (e *LocalEngine) BooksGrouped(ctx context.Context, sessionUID string) ([]BookGroup, error) {

   466→   466→        if err != nil {

   468→   468→        }
   469→   469→        db, err := e.dbHandle()
   470→   470→        if err != nil {
   471→   471→                return nil, err
   472→   472→        }
   473→   473→        db, err = e.ensureRelease(db, release)
   474→   474→        if err != nil {
   475→   475→                return nil, err
   476→   476→        }
   477→   477→        verified, _, err := e.run(ctx, db, req, release)
   478→   478→        if err != nil {
   479→   479→                return nil, err
   480→   480→        }
   481→   481→        workOrder := []string{}
   482→   482→        groups := map[string]*BookGroup{}
   483→   483→        docOrder := map[string][]string{}

   485→   485→        for _, r := range verified {
   486→   486→                bg, ok2 := groups[r.Work.UID]

   488→   488→                        bg = &BookGroup{Work: r.Work, Documents: []DocEvidence{}}
   489→   489→                        groups[r.Work.UID] = bg
   490→   490→                        workOrder = append(workOrder, r.Work.UID)
   491→   491→                }
   492→   492→                key := r.Work.UID + "|" + r.Document.UID

   495→   495→                        dg = &DocEvidence{Document: r.Document, Evidence: []Result{}}
   496→   496→                        docs[key] = dg
   497→   497→                        docOrder[r.Work.UID] = append(docOrder[r.Work.UID], r.Document.UID)
   498→   498→                }
   499→   499→                dg.VerifiedOccurrences++
   500→   500→                if len(dg.Evidence) < 3 {
   501→   501→                        dg.Evidence = append(dg.Evidence, r)
   502→   502→                }
   503→   503→
   504→   504→        }
   505→   505→        var out []BookGroup
   506→   506→        for _, wUID := range workOrder {
   507→   507→                bg := groups[wUID]
   508→   508→                bg.Documents = []DocEvidence{}
   509→   509→                for _, dUID := range docOrder[wUID] {
   510→   510→                        bg.Documents = append(bg.Documents, *docs[wUID+"|"+dUID])
   511→   511→                }
   512→   512→                out = append(out, *bg)
   513→   513→        }
   514→   514→        if out == nil {
   515→   515→                out = []BookGroup{}
   516→   516→        }
   517→   517→        return out, nil
   518→   518→}

   523→   523→        reqJSON, err := json.Marshal(req)
   524→   524→        if err != nil {

   531→   531→        defer e.mu.Unlock()
   532→   532→        e.evictMemSessionsLocked()

   537→   537→func (e *LocalEngine) loadSession(sid string) (string, *Request, error) {
   538→   538→        if e.sessStore != nil {
   539→   539→                release, reqJSON, err := e.sessStore.LoadSearchSession(sid)
   540→   540→                if err != nil {
   541→   541→                        return "", nil, err
   542→   542→                }
   543→   543→                _ = e.sessStore.TouchSearchSession(sid, e.sessionTTL)
   544→   544→                var req Request
   545→   545→                if err := json.Unmarshal(reqJSON, &req); err != nil {
   546→   546→                        return "", nil, ErrSearchSessionGone
   547→   547→                }
   548→   548→                return release, &req, nil
   549→   549→        }
   550→   550→        e.mu.Lock()
   551→   551→        defer e.mu.Unlock()
   552→   552→        ms, ok := e.memSess[sid]
   553→   553→        if !ok || time.Now().After(ms.expire) {
   554→   554→                if ok {
   555→   555→                        delete(e.memSess, sid) // تخلیهٔ per-session، نه تخلیهٔ کلی
   556→   556→                }
   557→   557→                return "", nil, ErrSearchSessionGone
   558→   558→        }
   559→   559→        ms.expire = time.Now().Add(e.sessionTTL)
   560→   560→        var req Request
   561→   561→        if err := json.Unmarshal(ms.reqJSON, &req); err != nil {
   562→   562→                return "", nil, ErrSearchSessionGone
   563→   563→        }
   564→   564→        return ms.release, &req, nil
   565→   565→}
   566→   566→
   567→   567→// evictMemSessionsLocked منقضی‌ها را حذف می‌کند؛ اگر هنوز بالای سقف بود،
   568→   568→// زودترین انقضا (قدیمی‌ترین) حذف می‌شود — هرگز تخلیهٔ کامل map نیست.
   569→   569→func (e *LocalEngine) evictMemSessionsLocked() {
   570→   570→        now := time.Now()
   571→   571→        for k, ms := range e.memSess {
   572→   572→                if now.After(ms.expire) {
   573→   573→                        delete(e.memSess, k)
   574→   574→                }
   575→   575→        }
   576→   576→        for len(e.memSess) >= maxMemSessions {
   577→   577→                oldestK := ""
   578→   578→                var oldest time.Time
   579→   579→                for k, ms := range e.memSess {
   580→   580→                        if oldestK == "" || ms.expire.Before(oldest) {
   581→   581→                                oldestK = k
   582→   582→                                oldest = ms.expire
   583→   583→                        }
   584→   584→                }
   585→   585→                if oldestK == "" {
   586→   586→                        break
   587→   587→                }
   588→   588→                delete(e.memSess, oldestK)
   589→   589→        }
   590→   590→}

   655→   655→// (اشتباهِ ترتیب = خطای اسکن = حذف خاموش همهٔ کاندیدها؛ رگرسیون واقعی این فایل).
   656→   656→const candCols = `passage_uid,release_uid,collection_uid,work_uid,document_uid,` +
   657→   657→        `section_uid,page_uid,page_label,language,local_hadith_uid,revayat_index,` +

   660→   660→func (e *LocalEngine) run(ctx context.Context, db *sql.DB, req *Request, releaseUID string) ([]Result, bool, error) {
   661→   661→        qN, _ := norm.Normalize(req.Query)
   662→   662→        toks := norm.Tokenize(qN)
   663→   663→        citationOnly := req.Scope.Quran != nil && len(toks) == 0
   664→   664→        if len(toks) == 0 && !citationOnly {
   665→   665→                return nil, true, nil
   666→   666→        }
   667→   667→        cap := e.maxResults
   668→   668→        if cap <= 0 {
   669→   669→                cap = defaultMaxResults
   670→   670→        }
   671→   671→        var rows *sql.Rows
   672→   672→        var err error
   673→   673→        if citationOnly {
   674→   674→                // جست‌وجوی استناد قرآنی (QUR-03): همهٔ passages دامنه، فیلتر در verifier
   675→   675→                rows, err = db.Query(`SELECT `+candCols+` FROM passages WHERE release_uid=? ORDER BY work_uid, document_uid, block_start`, releaseUID)
   676→   676→        } else if req.Mode == StrictPhrase {
   677→   677→                // عبارت دقیق: پیش‌فیلتر با سه‌گانه‌های نویسه‌ای text_n1 (بدون اسکن کامل)؛
   678→   678→                // تأیید نهایی تطبیق همیشه روی متن اصلی با مرز واژه در verifyInText می‌ماند —
   679→   679→                // رفتار نتایج تغییری نمی‌کند (pre-filter ابرمجموعهٔ sound است).
   680→   680→                tris := norm.Trigrams(qN)
   681→   681→                if len(tris) > 0 && index.HasTrigrams(db) {
   682→   682→                        sel := sampleTrigrams(tris, 48)
   683→   683→                        rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   684→   684→                                SELECT passage_uid FROM trigrams WHERE tri IN (`+placeholders(len(sel))+`) GROUP BY passage_uid HAVING COUNT(DISTINCT tri)=? LIMIT 20000)
   685→   685→                                ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, append(toStrs(sel), len(sel))...)...)
   686→   686→                } else {
   687→   687→                        // نمایهٔ قدیمی بدون جدول trigrams یا عبارتِ کوتاه‌تر از ۳ نویسه — اسکن کامل (منصفانه گزارش می‌شود)
   688→   688→                        log.Println("noor-search: strict_phrase بدون پیش‌فیلتر trigram (نمایه قدیمی یا عبارت کوتاه) — اسکن کامل")
   689→   689→                        rows, err = db.Query(`SELECT `+candCols+` FROM passages WHERE release_uid=? ORDER BY work_uid, document_uid, block_start`, releaseUID)
   690→   690→                }
   691→   691→        } else if req.Mode == AnyWords || req.Mode == Linguistic {
   692→   692→                ph := placeholders(len(toks))
   693→   693→                rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   694→   694→                        SELECT DISTINCT p.passage_uid FROM postings p JOIN tokens t ON t.token_id=p.token_id WHERE t.token IN (`+ph+`) LIMIT 20000)
   695→   695→                        ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, toStrs(toks)...)...)
   696→   696→        } else {
   697→   697→                // پیش‌فیلتر «همهٔ واژه‌ها باید در passage باشند»: شمار مقایسه‌ای =
   698→   698→                // شمار واژه‌های متمایزِ پرس‌وجو (نه توکن‌ها با تکرار) — وگرنه هر عبارت
   699→   699→                // دارای واژهٔ تکراری (مثل «الرسول ما الرسول») بی‌کاندید می‌ماند.
   700→   700→                uniq := dedupStrings(toks)
   701→   701→                ph := placeholders(len(uniq))
   702→   702→                rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   703→   703→                        SELECT p.passage_uid FROM postings p JOIN tokens t ON t.token_id=p.token_id WHERE t.token IN (`+ph+`)
   704→   704→                        GROUP BY p.passage_uid HAVING COUNT(DISTINCT t.token)=? LIMIT 20000)
   705→   705→                        ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, append(toStrs(uniq), len(uniq))...)...)
   706→   706→        }
   707→   707→        if err != nil {
   708→   708→                return nil, false, err
   709→   709→        }
   710→   710→        defer rows.Close()
   711→   711→        var out []Result
   712→   712→        seen := map[string]bool{}
   713→   713→        rowN := 0
   714→   714→        for rows.Next() {
   715→   715→                rowN++
   716→   716→                if ctx.Err() != nil {
   717→   717→                        return out, false, ctx.Err()
   718→   718→                }
   719→   719→                var r candRow
   720→   720→                if err := scanCand(rows, &r); err != nil {
   721→   721→                        // حذف خاموش ممنوع: خطای اسکن = کاندید نادیده = نتیجهٔ ناقص بی‌اعلام.
   722→   722→                        // طبق فلسفهٔ خطای §۳۹، این وضعیت باید صریحاً خطا باشد.
   723→   723→                        return nil, false, fmt.Errorf("scan candidate: %w", err)
   724→   724→                }
   725→   725→                // فیلتر دامنهٔ اثر/سند/زبان در verifier هم اعمال می‌شود (دفاع در عمق)
   726→   726→                if !scopeAllows(req, &r) {
   727→   727→                        continue
   728→   728→                }
   729→   729→                if !citationAllows(req, &r) {
   730→   730→                        continue // استناد باید واقعاً در همین passage باشد، نه فقط در سند
   731→   731→                }
   732→   732→                hits, matchType := verifyInText(req, &r)
   733→   733→                if req.Scope.Quran != nil && len([]rune(req.Query)) == 0 && len(hits) == 0 {
   734→   734→                        // جست‌وجوی صرفاً استنادی: خود passage با شاهد ابتدایی نمایش داده می‌شود
   735→   735→                        n1r := []rune(r.TextOriginal)
   736→   736→                        hi := clampInt(160, 0, len(n1r))
   737→   737→                        hits = []hit{{Range: [2]int{0, hi}}}
   738→   738→                        matchType = "exact"
   739→   739→                }
   740→   740→
   741→   741→                for _, h := range hits {
   742→   742→                        // کلید رخداد = document + محدودهٔ واقعی مطلق در جریان متن سند (§۴۵)
   743→   743→                        key := r.DocumentUID + ":" + fmt.Sprint(r.SourceStart+h.Range[0]) + ":" + fmt.Sprint(r.SourceStart+h.Range[1])
   744→   744→                        if seen[key] {
   745→   745→                                continue // overlap تکراری نتیجه نمی‌سازد؛ رخدادهای واقعی جدا می‌مانند
   746→   746→                        }
   747→   747→                        seen[key] = true
   748→   748→                        if len(req.Scope.Roles) > 0 && !matchInsideRoles(r.RoleRanges, h.Range) {
   749→   749→                                continue // فیلتر نقش: محل واقعی تطبیق باید داخل نقش باشد (§۴۵)
   750→   750→                        }
   751→   751→                        out = append(out, e.toResult(&r, h, matchType))
   752→   752→                        if len(out) >= cap {
   753→   753→                                // سقف کل نتایج فعال شد (ب-۵): جمع‌آوری متوقف؛ صادقانه lower_bound اعلام می‌شود.
   754→   754→                                return out, true, nil
   755→   755→                        }
   756→   756→                }
   757→   757→        }
   758→   758→        return out, false, nil
   759→   759→}
   760→   760→
   761→   761→// sampleTrigrams نمونهٔ پخش‌شده از سه‌گانه‌ها برای پرس‌وجوی پیش‌فیلتر —
   762→   762→// زیرمجموعه از سه‌گانه‌های عبارت است، پس صدا حفظ می‌شود (ابر‌مجموعهٔ کاندید).
   763→   763→func sampleTrigrams(tris []string, k int) []string {
   764→   764→        if len(tris) <= k {
   765→   765→                return tris
   766→   766→        }
   767→   767→        step := (len(tris) + k - 1) / k
   768→   768→        var out []string
   769→   769→        for i := 0; i < len(tris) && len(out) < k; i += step {
   770→   770→                out = append(out, tris[i])

   795→   795→func verifyInText(req *Request, r *candRow) ([]hit, string) {
   796→   796→        orig := []rune(r.TextOriginal)
   797→   797→        n1 := []rune(r.TextN1)
   798→   798→        var n1Map []int
   799→   799→        _ = json.Unmarshal([]byte(r.N1Map), &n1Map)
   800→   800→        switch req.Mode {
   801→   801→        case StrictPhrase:
   802→   802→                q := []rune(req.Query)
   803→   803→                var out []hit
   804→   804→                start := 0
   805→   805→                for {
   806→   806→                        idx := indexRunes(orig[start:], q)
   807→   807→                        if idx < 0 {
   808→   808→                                break
   809→   809→                        }
   810→   810→                        s := start + idx
   811→   811→                        e := s + len(q)
   812→   812→                        if norm.WordBoundary(string(orig), s, e) {
   813→   813→                                out = append(out, hit{Range: [2]int{s, e}})
   814→   814→                        }
   815→   815→                        start = s + 1
   816→   816→                        if len(out) >= 50 {
   817→   817→                                break
   818→   818→                        }
   819→   819→                }
   820→   820→                return out, "exact"
   821→   821→        default:
   822→   822→                qN, _ := norm.Normalize(req.Query)
   823→   823→                if req.Mode == Linguistic {
   824→   824→                        qN = norm.Fold(req.Query)
   825→   825→                }
   826→   826→                if req.Mode == AllWords || req.Mode == Near {
   827→   827→                        toks := norm.Tokenize(qN)
   828→   828→                        positions := make([][]int, len(toks))
   829→   829→                        for i, t := range toks {
   830→   830→                                positions[i] = findAllWord(n1, t)
   831→   831→                        }
   832→   832→                        var out []hit
   833→   833→                        if req.Mode == Near && req.Near != nil && req.Near.Slop > 0 {
   834→   834→                                for _, rng := range nearCombination(positions, req.Near.Slop, req.Near.Ordered, len(n1)) {
   835→   835→                                        out = append(out, mapN1Hit(rng, n1Map))
   836→   836→                                }
   837→   837→                                return out, "normalized"
   838→   838→                        }
   839→   839→                        if allPresent(positions) {
   840→   840→                                lo, hi := -1, -1
   841→   841→                                for i, ps := range positions {
   842→   842→                                        if i == 0 || ps[0] < lo {
   843→   843→                                                lo = ps[0]
   844→   844→                                        }
   845→   845→                                        last := ps[len(ps)-1] + len([]rune(toks[i]))
   846→   846→                                        if i == 0 || last > hi {
   847→   847→                                                hi = last
   848→   848→                                        }
   849→   849→                                }
   850→   850→                                out = append(out, mapN1Hit([2]int{lo, hi}, n1Map))
   851→   851→                        }
   852→   852→                        return out, "normalized"
   853→   853→                }
   854→   854→                // normalized_phrase / linguistic / long_quote: تطبیق متوالی با مرز واژه
   855→   855→                qRunes := []rune(qN)
   856→   856→                var out []hit
   857→   857→                start := 0
   858→   858→                for {
   859→   859→                        idx := indexRunes(n1[start:], qRunes)
   860→   860→                        if idx < 0 {
   861→   861→                                break
   862→   862→                        }
   863→   863→                        s := start + idx
   864→   864→                        e := s + len(qRunes)
   865→   865→                        if norm.WordBoundary(string(n1), s, e) {
   866→   866→                                out = append(out, mapN1Hit([2]int{s, e}, n1Map))
   867→   867→                        }
   868→   868→                        start = s + 1
   869→   869→                        if len(out) >= 50 {
   870→   870→                                break
   871→   871→                        }
   872→   872→                }
   873→   873→                if req.Mode == Linguistic {
   874→   874→                        return out, "linguistic"
   875→   875→                }
   876→   876→                return out, "normalized"
   877→   877→        }
   878→   878→}

   930→   930→func nearCombination(positions [][]int, slop int, ordered bool, textLen int) [][2]int {
   931→   931→        var out [][2]int
   932→   932→        if len(positions) == 0 || !allPresent(positions) {
   933→   933→                return nil
   934→   934→        }
   935→   935→        idx := make([]int, len(positions))
   936→   936→        for {
   937→   937→                var lo, hi int
   938→   938→                for i, p := range positions {
   939→   939→                        pos := p[idx[i]]
   940→   940→                        if i == 0 || pos < lo {
   941→   941→                                lo = pos
   942→   942→                        }
   943→   943→                        if i == 0 || pos > hi {
   944→   944→                                hi = pos
   945→   945→                        }
   946→   946→                }
   947→   947→                ok := (hi - lo) <= slop*6
   948→   948→                if ok && ordered && len(positions) > 1 {
   949→   949→                        for i := 1; i < len(positions); i++ {
   950→   950→                                if positions[i][idx[i]] < positions[i-1][idx[i-1]] {
   951→   951→                                        ok = false
   952→   952→                                        break
   953→   953→                                }
   954→   954→                        }
   955→   955→                }
   956→   956→                if ok {
   957→   957→                        out = append(out, [2]int{lo, clampInt(hi+len(positions[len(positions)-1]), 0, textLen)})
   958→   958→                }
   959→   959→                i := len(idx) - 1
   960→   960→                for i >= 0 {
   961→   961→                        idx[i]++
   962→   962→                        if idx[i] < len(positions[i]) {
   963→   963→                                break
   964→   964→                        }
   965→   965→                        idx[i] = 0
   966→   966→                        i--
   967→   967→                }
   968→   968→                if i < 0 || len(out) >= 10 {
   969→   969→                        break
   970→   970→                }
   971→   971→        }
   972→   972→        return out
   973→   973→}

  1187→  1187→// indexRunes یافتن زیررشته بر پایهٔ code point — نه بایت (صحت نشانی عربی/فارسی).
  1188→  1188→func indexRunes(s []rune, sub []rune) int {

  1190→  1190→        if n == 0 || n > len(s) {
  1191→  1191→                return -1
  1192→  1192→        }
  1193→  1193→        for i := 0; i+n <= len(s); i++ {
  1194→  1194→                match := true
  1195→  1195→                for j := 0; j < n; j++ {
  1196→  1196→                        if s[i+j] != sub[j] {
  1197→  1197→                                match = false
  1198→  1198→                                break
  1199→  1199→                        }
  1200→  1200→                }
  1201→  1201→                if match {
