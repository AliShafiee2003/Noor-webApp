    68→    68→function clampPage(raw: string | undefined): number {
    69→    69→  const n = Number.parseInt(raw ?? "", 10);
    70→    70→  return Number.isFinite(n) && n >= 1 && n <= 200 ? n : 1;

    73→    73→/** بازسازی hash از payload جست‌وجوی ذخیره‌شده (اجرای دوباره با همان نشانی) */
    74→    74→function hashFromPayload(payload: unknown): Record<string, string> | null {
    75→    75→  if (!payload || typeof payload !== "object") return null;
    76→    76→  const p = payload as Record<string, unknown>;
    77→    77→  if (typeof p.query !== "string" || typeof p.mode !== "string") return null;
    78→    78→  const out: Record<string, string> = { q: p.query, mode: p.mode };
    79→    79→  const scope = (p.scope && typeof p.scope === "object" ? p.scope : {}) as Record<string, unknown>;
    80→    80→  const target = typeof scope.target === "string" ? scope.target : "library";
    81→    81→  if (target === "titles") {
    82→    82→    out.scope = "titles";
    83→    83→  } else if (target === "work" && typeof scope.work_uid === "string" && scope.work_uid) {
    84→    84→    out.scope = "work";
    85→    85→    out.work = scope.work_uid;
    86→    86→  } else if (
    87→    87→    target === "document" &&
    88→    88→    Array.isArray(scope.document_uids) &&
    89→    89→    typeof scope.document_uids[0] === "string"
    90→    90→  ) {
    91→    91→    out.scope = "document";
    92→    92→    out.doc = scope.document_uids[0];
    93→    93→    if (typeof scope.work_uid === "string") out.work = scope.work_uid;
    94→    94→  } else {
    95→    95→    out.scope = "library";
    96→    96→  }
    97→    97→  const roles = Array.isArray(scope.roles)
    98→    98→    ? scope.roles.filter((r): r is string => typeof r === "string")
    99→    99→    : [];
   100→   100→  const roleValues = ROLE_FILTERS.filter((f) => f.codes.some((c) => roles.includes(c))).map(
   101→   101→    (f) => f.value,
   102→   102→  );
   103→   103→  if (roleValues.length > 0) out.roles = roleValues.join(",");
   104→   104→  if (typeof scope.language === "string" && scope.language) out.lang = scope.language;
   105→   105→  const near = (p.near && typeof p.near === "object" ? p.near : null) as Record<string, unknown> | null;
   106→   106→  if (near) {
   107→   107→    out.slop = String(typeof near.slop === "number" ? near.slop : 3);
   108→   108→    if (near.ordered === false) out.ordered = "0";
   109→   109→  }
   110→   110→  return out;
   111→   111→}

   315→   315→
   316→   316→export function SearchView() {
   317→   317→  const { route, navigate } = useRoute();
   318→   318→
   319→   319→  /* وضعیت اجراشدهٔ جست‌وجو (از hash) */
   320→   320→  const params = parseSearchParams(route.query);
   321→   321→  const searchKey = useMemo(() => searchStateKey(params), [params]);
   322→   322→  const hashPage = clampPage(route.query.page);
   323→   323→  const [saveOpen, setSaveOpen] = useState(false);
   324→   324→  const view: "results" | "books" = route.query.view === "books" ? "books" : "results";
   325→   325→  // عبارت یا استناد قرآنی (QUR-03) هر دو شروع‌کنندهٔ جست‌وجو هستند
   326→   326→  const hasQuery = params.q.trim().length > 0 || params.sure.trim().length > 0;
   327→   327→
   328→   328→  /* پیش‌نویس فرم — هم‌گام‌سازی با hash در زمان رندر (الگوی مجاز React) */
   329→   329→  const [draft, setDraft] = useState<SearchHashState>(() => parseSearchParams(route.query));
   330→   330→  const [draftHash, setDraftHash] = useState(route.fullHash);
   331→   331→  if (draftHash !== route.fullHash) {
   332→   332→    setDraftHash(route.fullHash);
   333→   333→    setDraft(parseSearchParams(route.query));
   334→   334→  }
   335→   335→
   336→   336→  /** ساخت پارامترهای hash از وضعیت فرم — مشترک میان setQuery و submit */
   337→   337→  const searchQueryParams = useCallback(
   338→   338→    (state: SearchHashState): Record<string, string | undefined> => ({
   339→   339→      q: state.q.trim() || undefined,
   340→   340→      mode: state.mode,
   341→   341→      scope: state.scope,
   342→   342→      work: state.work || undefined,
   343→   343→      doc: state.doc || undefined,
   344→   344→      roles: state.roles.length > 0 ? state.roles.join(",") : undefined,
   345→   345→      lang: state.lang || undefined,
   346→   346→      slop: state.mode === "near" ? String(state.slop) : undefined,
   347→   347→      ordered: state.mode === "near" && !state.ordered ? "0" : undefined,
   348→   348→    }),
   349→   349→    [],
   350→   350→  );
   351→   351→
   352→   352→  /** نوشتن وضعیت در hash — هر بار کل پارامترها داده می‌شود */
   353→   353→  const setQuery = useCallback(
   354→   354→    (state: SearchHashState, extra: Record<string, string | undefined> = {}, replace = false) => {
   355→   355→      navigate(
   356→   356→        "/search",
   357→   357→        {
   358→   358→          ...searchQueryParams(state),
   359→   359→          page: undefined,
   360→   360→          view: undefined,
   361→   361→          ...extra,
   362→   362→        },
   363→   363→        { replace },
   364→   364→      );
   365→   365→    },
   366→   366→    [navigate, searchQueryParams],
   367→   367→  );
   368→   368→
   369→   369→  const queryClient = useQueryClient();
   370→   370→  const submit = useCallback(() => {
   371→   371→    /* اگر پارامترهای جست‌وجو تغییر نکرده باشد، navigate بی‌اثر است و کشِ
   372→   372→       نشست دوباره استفاده می‌شد — برای «اجرای دوبارهٔ همان جست‌وجو» (پس از
   373→   373→       انقضای نشست سرور یا بازسازی نمایه) کش باطل و بازاجرایی می‌شود
   374→   374→       (SEARCH-20/WORK-02). در غیر این صورت hash با پارامترهای تازه نوشته
   375→   375→       می‌شود و کلید تازه، نشست تازه می‌سازد. */
   376→   376→    if (searchStateKey(draft) === searchKey) {
   377→   377→      queryClient.invalidateQueries({ queryKey: ["search-session", searchKey] });
   378→   378→      return;
   379→   379→    }
   380→   380→    setQuery(draft);
   381→   381→  }, [draft, queryClient, searchKey, setQuery]);
   382→   382→
   383→   383→  const patchCommitted = useCallback(
   384→   384→    (patch: Partial<SearchHashState>) => {
   385→   385→      setQuery({ ...params, ...patch });
   386→   386→    },
   387→   387→    [params, setQuery],
   388→   388→  );
   389→   389→
   390→   390→  /* لایهٔ ۱ — نشست جست‌وجو: صفحهٔ نخست + شمارش صادقانه */
   391→   391→  const request = buildSearchRequest(params);
   392→   392→
   393→   393→  const sessionQuery = useQuery({
   394→   394→    queryKey: ["search-session", searchKey],
   395→   395→    queryFn: async ({ signal }) => {
   396→   396→      const resp = await api.post<SearchResponse>("/search", request);
   397→   397→      if (signal.aborted) throw new DOMException("aborted", "AbortError");
   398→   398→      return resp;
   399→   399→    },
   400→   400→    enabled: hasQuery,
   401→   401→    staleTime: 4 * 60_000,
   402→   402→    gcTime: 10 * 60_000,
   403→   403→    retry: false,
   404→   404→    refetchOnWindowFocus: false,
   405→   405→    refetchOnReconnect: false,
   406→   406→  });
   407→   407→
