   230→func NewLocal(path string, secret []byte, meta MetaSource, opts ...Option) *LocalEngine {
   231→        e := &LocalEngine{path: path, Secret: secret, catalog: meta,
   232→                memSess:    map[string]*memSession{},

   234→                sessionTTL: defaultSessionTTL,

   242→// dbHandle اتصال فقط‌خواندنی نمایه؛ با تعویض فایل (نسل جدید) خودکار بازگشایی می‌شود.
   243→// تشخیص تعویض: inode + اندازه؛ و اگر release_uid اتصال با نسخهٔ درخواستی نخواند، بازگشایی اجباری.
   244→func (e *LocalEngine) dbHandle() (*sql.DB, error) {
   245→        e.dbMu.Lock()
   246→        defer e.dbMu.Unlock()
   247→        fi, err := osStat(e.path)
   248→        if err == nil {
   249→                ino := fileInode(fi)
   250→                sz := fi.Size()
   251→                if e.db != nil && (ino != e.dbInode || sz != e.dbSize) {
   252→                        e.db.Close()
   253→                        e.db = nil
   254→                }
   255→                if e.db == nil {
   256→                        db, err := sql.Open("sqlite", "file:"+e.path+"?mode=ro")
   257→                        if err != nil {
   258→                                return nil, err
   259→                        }
   260→                        db.SetMaxOpenConns(3)
   261→                        e.db = db
   262→                        e.dbInode = ino
   263→                        e.dbSize = sz
   264→                }
   265→                return e.db, nil
   266→        }
   267→        if e.db != nil {
   268→                return e.db, nil
   269→        }
   270→        return nil, errors.New("نمایهٔ جست‌وجو در دسترس نیست")
   271→}

   273→// ensureRelease اگر اتصالِ فعلی نسلِ درخواستی را ندارد، اتصال را نوسازی می‌کند
   274→// (مقاوم در برابر inode reuse بین نسل‌ها).
   275→func (e *LocalEngine) ensureRelease(db *sql.DB, releaseUID string) (*sql.DB, error) {
   276→        var rel string
   277→        err := db.QueryRow(`SELECT release_uid FROM passages LIMIT 1`).Scan(&rel)
   278→        if err == nil && rel == releaseUID {
   279→                return db, nil
   280→        }
   281→        // نوسازی اجباری
   282→        e.dbMu.Lock()
   283→        if e.db != nil {
   284→                e.db.Close()
   285→                e.db = nil
   286→        }
   287→        e.dbMu.Unlock()
   288→        return e.dbHandle()
   289→}

   301→// validateRequest قواعد مشترک اعتبارسنجی پرس‌وجو (هر دو موتور).
   302→func validateRequest(req *Request) error {
   303→        // جست‌وجوی صرفاً استنادی (بدون عبارت) مجاز است — QUR-03
   304→        if len([]rune(req.Query)) == 0 && req.Scope.Quran == nil {
   305→                return errors.New("عبارت جست‌وجو خالی است")
   306→        }
   307→        modeOK := false
   308→        for _, m := range []string{StrictPhrase, NormalizedPhrase, AllWords, AnyWords, Near, Linguistic, LongQuote} {
   309→                if req.Mode == m {
   310→                        modeOK = true
   311→                }
   312→        }
   313→        if !modeOK {
   314→                return fmt.Errorf("حالت جست‌وجو معتبر نیست")
   315→        }
   316→        limit := 2048
   317→        if req.Mode == LongQuote {
   318→                limit = 8192
   319→        }
   320→        if len([]rune(req.Query)) > limit {
   321→                return errors.New("عبارت بیش از حد مجاز است؛ برای نقل‌قول بلند از حالت مخصوص آن استفاده کنید")
   322→        }
   323→        if req.PageSize < 5 || req.PageSize > 50 {
   324→                req.PageSize = 20
   325→        }
   326→        return nil
   327→}

   329→// Search اجرای جست‌وجو: کاندید → verifier → نتایج تأییدشده.
   330→func (e *LocalEngine) Search(ctx context.Context, req *Request, releaseUID string) (*Response, error) {
   331→        if err := e.validate(req); err != nil {
   332→                return nil, err
   333→        }
   334→        if req.Scope.Target == "titles" {
   335→                hits, err := e.SearchTitles(ctx, req.Query, req.PageSize)
   336→                if err != nil {
   337→                        return nil, err
   338→                }
   339→                resp := &Response{ReleaseUID: releaseUID, CompletionState: "complete", CountRelation: "exact", Results: []Result{}}
   340→                for _, h := range hits {
   341→                        resp.Results = append(resp.Results, Result{
   342→                                PassageUID: "title-" + h.WorkUID, MatchType: "exact",
   343→                                Work: Ref{UID: h.WorkUID, Title: h.Title, Label: h.ShortTitle},
   344→                                Page: Ref{Label: "نام کتاب"},
   345→                        })
   346→                }
   347→                resp.VerifiedBooks = len(hits)
   348→                resp.VerifiedOccurrences = len(hits)
   349→                return resp, nil
   350→        }
   351→        db, err := e.dbHandle()
   352→        if err != nil {
   353→                return nil, err
   354→        }
   355→        db, err = e.ensureRelease(db, releaseUID)
   356→        if err != nil {
   357→                return nil, err
   358→        }
   359→        verified, capped, err := e.run(ctx, db, req, releaseUID)
   360→        if err != nil {
   361→                return nil, err
   362→        }
   363→        sid := newID("s")
   364→        if err := e.saveSession(sid, releaseUID, req); err != nil {
   365→                return nil, err
   366→        }
   367→        return e.buildResponse(sid, releaseUID, req, verified, 0, capped), nil
   368→}

   370→// Continue ادامهٔ بررسی با cursor امضاشده (§۴۷) — re-execute قطعی + skip موقعیتی.
   371→func (e *LocalEngine) Continue(ctx context.Context, sessionUID, cursor, releaseUID string) (*Response, error) {
   372→        if !verifyCursor(cursor, e.Secret) {
   373→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   374→        }
   375→        payload, ok := parseCursorPayload(cursor)
   376→        if !ok || payload.S != sessionUID {
   377→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   378→        }
   379→        release, req, err := e.loadSession(sessionUID)
   380→        if err != nil {
   381→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   382→        }
   383→        db, err := e.dbHandle()
   384→        if err != nil {
   385→                return nil, err
   386→        }
   387→        db, err = e.ensureRelease(db, release)
   388→        if err != nil {
   389→                return nil, err
   390→        }
   391→        verified, capped, err := e.run(ctx, db, req, release)
   392→        if err != nil {
   393→                return nil, err
   394→        }
   395→        _ = releaseUID // پرس‌وجو با release نشست ادامه می‌یابد تا نتایج صفحه‌ها هم‌خوان بمانند
   396→        return e.buildResponse(sessionUID, release, req, verified, payload.O, capped), nil
   397→}

   399→// cursorPayload محتوای امضاشدهٔ cursor: شناسهٔ نشست + موقعیت.
   400→type cursorPayloadT struct {
   401→        S string `json:"s"`
   402→        O int    `json:"o"`
   403→}
   404→
   405→func parseCursorPayload(cursor string) (cursorPayloadT, bool) {
   406→        var out cursorPayloadT
   407→        part := strings.SplitN(cursor, ".", 2)[0]
   408→        b, err := base64.RawURLEncoding.DecodeString(part)
   409→        if err != nil || json.Unmarshal(b, &out) != nil {
   410→                return out, false
   411→        }
   412→        return out, true
   413→}

   415→// buildResponse پاسخ از روی نتایج کامل اجرای جاری می‌سازد؛ صفحه از offset شروع می‌شود.
   416→func (e *LocalEngine) buildResponse(sid, release string, req *Request, verified []Result, offset int, capped bool) *Response {
   417→        if offset < 0 {
   418→                offset = 0
   419→        }
   420→        if offset > len(verified) {
   421→                offset = len(verified)
   422→        }
   423→        pageSize := req.PageSize
   424→        if pageSize < 5 || pageSize > 50 {
   425→                pageSize = 20
   426→        }
   427→        page := verified[offset:]
   428→        if len(page) > pageSize {
   429→                page = page[:pageSize]
   430→        }
   431→        books := map[string]bool{}
   432→        passages := map[string]bool{}
   433→        hadiths := map[string]bool{}
   434→        for _, r := range verified {
   435→                if w, _, ok := e.catalog.DocumentMeta(r.Document.UID); ok {
   436→                        books[w] = true
   437→                } else {
   438→                        books[r.Work.UID] = true
   439→                }
   440→                if r.PassageUID != "" {
   441→                        passages[r.PassageUID] = true
   442→                }
   443→                if r.LocalHadith != "" {
   444→                        // روایت درون هر سند متمایز است؛ شناسهٔ محلی جهانی نیست
   445→                        hadiths[r.Document.UID+"/"+r.LocalHadith] = true
   446→                }
   447→        }
   448→        newOffset := offset + len(page)
   449→        continuation := newOffset < len(verified)
   450→        rel := "exact"
   451→        if capped {
   452→                rel = "lower_bound" // سقف فعال شد؛ شمارش‌ها حداقل واقعی‌اند
   453→        }
   454→        resp := &Response{
   455→                SessionUID:            sid,
   456→                ReleaseUID:            release,
   457→                CompletionState:       "complete",
   458→                VerifiedOccurrences:   len(verified),
   459→                VerifiedBooks:         len(books),
   460→                VerifiedPassages:      len(passages),
   461→                VerifiedHadiths:       len(hadiths),
   462→                CountRelation:         rel,
   463→                ContinuationAvailable: continuation,
   464→                Results:               page,
   465→        }
   466→        if capped {
   467→                resp.Notice = fmt.Sprintf("نتایج بیشتر از سقف جست‌وجو است؛ شمارش‌ها حداقل واقعی‌اند (سقف %s موضع).", faNum(e.maxResults))
   468→        }
   469→        if continuation {
   470→                resp.NextCursor = signCursor(map[string]any{"s": sid, "o": newOffset}, e.Secret)
   471→        }
   472→        return resp
   473→}

   475→// BooksGrouped «این عبارت در چه کتاب‌هایی آمده؟» — فقط تأییدشده‌ها (§۴۷).
   476→// با re-execute از روی درخواست ذخیره‌شدهٔ نشست (بدون نگهداری نتایج در حافظه).
   477→func (e *LocalEngine) BooksGrouped(ctx context.Context, sessionUID string) ([]BookGroup, error) {
   478→        release, req, err := e.loadSession(sessionUID)
   479→        if err != nil {
   480→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   481→        }
   482→        db, err := e.dbHandle()
   483→        if err != nil {
   484→                return nil, err
   485→        }
   486→        db, err = e.ensureRelease(db, release)
   487→        if err != nil {
   488→                return nil, err
   489→        }
   490→        verified, _, err := e.run(ctx, db, req, release)
   491→        if err != nil {
   492→                return nil, err
   493→        }
   494→        workOrder := []string{}
   495→        groups := map[string]*BookGroup{}
   496→        docOrder := map[string][]string{}
   497→        docs := map[string]*DocEvidence{}
   498→        for _, r := range verified {
   499→                bg, ok2 := groups[r.Work.UID]
   500→                if !ok2 {
   501→                        bg = &BookGroup{Work: r.Work, Documents: []DocEvidence{}}
   502→                        groups[r.Work.UID] = bg
   503→                        workOrder = append(workOrder, r.Work.UID)
   504→                }
   505→                key := r.Work.UID + "|" + r.Document.UID
   506→                dg, ok3 := docs[key]
   507→                if !ok3 {
   508→                        dg = &DocEvidence{Document: r.Document, Evidence: []Result{}}
   509→                        docs[key] = dg
   510→                        docOrder[r.Work.UID] = append(docOrder[r.Work.UID], r.Document.UID)
   511→                }
   512→                dg.VerifiedOccurrences++
   513→                if len(dg.Evidence) < 3 {
   514→                        dg.Evidence = append(dg.Evidence, r)
   515→                }
   516→
   517→        }
   518→        var out []BookGroup
   519→        for _, wUID := range workOrder {
   520→                bg := groups[wUID]
   521→                bg.Documents = []DocEvidence{}
   522→                for _, dUID := range docOrder[wUID] {
   523→                        bg.Documents = append(bg.Documents, *docs[wUID+"|"+dUID])
   524→                }
   525→                out = append(out, *bg)
   526→        }
   527→        if out == nil {
   528→                out = []BookGroup{}
   529→        }
   530→        return out, nil
   531→}

   535→func (e *LocalEngine) saveSession(sid, releaseUID string, req *Request) error {
   536→        reqJSON, err := json.Marshal(req)
   537→        if err != nil {
   538→                return err
   539→        }
   540→        if e.sessStore != nil {
   541→                return e.sessStore.SaveSearchSession(sid, releaseUID, reqJSON, e.sessionTTL)
   542→        }
   543→        e.mu.Lock()
   544→        defer e.mu.Unlock()
   545→        e.evictMemSessionsLocked()
   546→        e.memSess[sid] = &memSession{release: releaseUID, reqJSON: reqJSON, expire: time.Now().Add(e.sessionTTL)}
   547→        return nil
   548→}

   550→func (e *LocalEngine) loadSession(sid string) (string, *Request, error) {
   551→        if e.sessStore != nil {
   552→                release, reqJSON, err := e.sessStore.LoadSearchSession(sid)
   553→                if err != nil {
   554→                        return "", nil, err
   555→                }
   556→                _ = e.sessStore.TouchSearchSession(sid, e.sessionTTL)
   557→                var req Request
   558→                if err := json.Unmarshal(reqJSON, &req); err != nil {
   559→                        return "", nil, ErrSearchSessionGone
   560→                }
   561→                return release, &req, nil
   562→        }
   563→        e.mu.Lock()
   564→        defer e.mu.Unlock()
   565→        ms, ok := e.memSess[sid]
   566→        if !ok || time.Now().After(ms.expire) {
   567→                if ok {
   568→                        delete(e.memSess, sid) // تخلیهٔ per-session، نه تخلیهٔ کلی
   569→                }
   570→                return "", nil, ErrSearchSessionGone
   571→        }
   572→        ms.expire = time.Now().Add(e.sessionTTL)
   573→        var req Request
   574→        if err := json.Unmarshal(ms.reqJSON, &req); err != nil {
   575→                return "", nil, ErrSearchSessionGone
   576→        }
   577→        return ms.release, &req, nil
   578→}

   580→// evictMemSessionsLocked منقضی‌ها را حذف می‌کند؛ اگر هنوز بالای سقف بود،
   581→// زودترین انقضا (قدیمی‌ترین) حذف می‌شود — هرگز تخلیهٔ کامل map نیست.
   582→func (e *LocalEngine) evictMemSessionsLocked() {
   583→        now := time.Now()
   584→        for k, ms := range e.memSess {
   585→                if now.After(ms.expire) {
   586→                        delete(e.memSess, k)
   587→                }
   588→        }
   589→        for len(e.memSess) >= maxMemSessions {
   590→                oldestK := ""
   591→                var oldest time.Time
   592→                for k, ms := range e.memSess {
   593→                        if oldestK == "" || ms.expire.Before(oldest) {
   594→                                oldestK = k
   595→                                oldest = ms.expire
   596→                        }
   597→                }
   598→                if oldestK == "" {
   599→                        break
   600→                }
   601→                delete(e.memSess, oldestK)
   602→        }
   603→}

   673→func (e *LocalEngine) run(ctx context.Context, db *sql.DB, req *Request, releaseUID string) ([]Result, bool, error) {
   674→        qN, _ := norm.Normalize(req.Query)
   675→        toks := norm.Tokenize(qN)
   676→        citationOnly := req.Scope.Quran != nil && len(toks) == 0
   677→        if len(toks) == 0 && !citationOnly {
   678→                return nil, true, nil
   679→        }
   680→        cap := e.maxResults
   681→        if cap <= 0 {
   682→                cap = defaultMaxResults
   683→        }
   684→        var rows *sql.Rows
   685→        var err error
   686→        if citationOnly {
   687→                // جست‌وجوی استناد قرآنی (QUR-03): همهٔ passages دامنه، فیلتر در verifier
   688→                rows, err = db.Query(`SELECT `+candCols+` FROM passages WHERE release_uid=? ORDER BY work_uid, document_uid, block_start`, releaseUID)
   689→        } else if req.Mode == StrictPhrase {
   690→                // عبارت دقیق: پیش‌فیلتر با سه‌گانه‌های نویسه‌ای text_n1 (بدون اسکن کامل)؛
   691→                // تأیید نهایی تطبیق همیشه روی متن اصلی با مرز واژه در verifyInText می‌ماند —
   692→                // رفتار نتایج تغییری نمی‌کند (pre-filter ابرمجموعهٔ sound است).
   693→                tris := norm.Trigrams(qN)
   694→                if len(tris) > 0 && index.HasTrigrams(db) {
   695→                        sel := sampleTrigrams(tris, 48)
   696→                        rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   697→                                SELECT passage_uid FROM trigrams WHERE tri IN (`+placeholders(len(sel))+`) GROUP BY passage_uid HAVING COUNT(DISTINCT tri)=? LIMIT 20000)
   698→                                ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, append(toStrs(sel), len(sel))...)...)
   699→                } else {
   700→                        // نمایهٔ قدیمی بدون جدول trigrams یا عبارتِ کوتاه‌تر از ۳ نویسه — اسکن کامل (منصفانه گزارش می‌شود)
   701→                        log.Println("noor-search: strict_phrase بدون پیش‌فیلتر trigram (نمایه قدیمی یا عبارت کوتاه) — اسکن کامل")
   702→                        e.statScans.Add(1)
   703→                        rows, err = db.Query(`SELECT `+candCols+` FROM passages WHERE release_uid=? ORDER BY work_uid, document_uid, block_start`, releaseUID)
   704→                }
   705→        } else if req.Mode == AnyWords || req.Mode == Linguistic {
   706→                ph := placeholders(len(toks))
   707→                rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   708→                        SELECT DISTINCT p.passage_uid FROM postings p JOIN tokens t ON t.token_id=p.token_id WHERE t.token IN (`+ph+`) LIMIT 20000)
   709→                        ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, toStrs(toks)...)...)
   710→        } else {
   711→                // پیش‌فیلتر «همهٔ واژه‌ها باید در passage باشند»: شمار مقایسه‌ای =
   712→                // شمار واژه‌های متمایزِ پرس‌وجو (نه توکن‌ها با تکرار) — وگرنه هر عبارت
   713→                // دارای واژهٔ تکراری (مثل «الرسول ما الرسول») بی‌کاندید می‌ماند.
   714→                uniq := dedupStrings(toks)
   715→                ph := placeholders(len(uniq))
   716→                rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   717→                        SELECT p.passage_uid FROM postings p JOIN tokens t ON t.token_id=p.token_id WHERE t.token IN (`+ph+`)
   718→                        GROUP BY p.passage_uid HAVING COUNT(DISTINCT t.token)=? LIMIT 20000)
   719→                        ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, append(toStrs(uniq), len(uniq))...)...)
   720→        }
   721→        if err != nil {
   722→                return nil, false, err
   723→        }
   724→        defer rows.Close()
   725→        var out []Result
   726→        seen := map[string]bool{}
   727→        rowN := 0
   728→        for rows.Next() {
   729→                rowN++
   730→                if ctx.Err() != nil {
   731→                        return out, false, ctx.Err()
   732→                }
   733→                var r candRow
   734→                if err := scanCand(rows, &r); err != nil {
   735→                        // حذف خاموش ممنوع: خطای اسکن = کاندید نادیده = نتیجهٔ ناقص بی‌اعلام.
   736→                        // طبق فلسفهٔ خطای §۳۹، این وضعیت باید صریحاً خطا باشد.
   737→                        return nil, false, fmt.Errorf("scan candidate: %w", err)
   738→                }
   739→                // فیلتر دامنهٔ اثر/سند/زبان در verifier هم اعمال می‌شود (دفاع در عمق)
   740→                if !scopeAllows(req, &r) {
   741→                        continue
   742→                }
   743→                if !citationAllows(req, &r) {
   744→                        continue // استناد باید واقعاً در همین passage باشد، نه فقط در سند
   745→                }
   746→                hits, matchType := verifyInText(req, &r)
   747→                if req.Scope.Quran != nil && len([]rune(req.Query)) == 0 && len(hits) == 0 {
   748→                        // جست‌وجوی صرفاً استنادی: خود passage با شاهد ابتدایی نمایش داده می‌شود
   749→                        n1r := []rune(r.TextOriginal)
   750→                        hi := clampInt(160, 0, len(n1r))
   751→                        hits = []hit{{Range: [2]int{0, hi}}}
   752→                        matchType = "exact"
   753→                }
   754→
   755→                for _, h := range hits {
   756→                        // کلید رخداد = document + محدودهٔ واقعی مطلق در جریان متن سند (§۴۵)
   757→                        key := r.DocumentUID + ":" + fmt.Sprint(r.SourceStart+h.Range[0]) + ":" + fmt.Sprint(r.SourceStart+h.Range[1])
   758→                        if seen[key] {
   759→                                continue // overlap تکراری نتیجه نمی‌سازد؛ رخدادهای واقعی جدا می‌مانند
   760→                        }
   761→                        seen[key] = true
   762→                        if len(req.Scope.Roles) > 0 && !matchInsideRoles(r.RoleRanges, h.Range) {
   763→                                continue // فیلتر نقش: محل واقعی تطبیق باید داخل نقش باشد (§۴۵)
   764→                        }
   765→                        out = append(out, e.toResult(&r, h, matchType))
   766→                        if len(out) >= cap {
   767→                                // سقف کل نتایج فعال شد (ب-۵): جمع‌آوری متوقف؛ صادقانه lower_bound اعلام می‌شود.
   768→                                e.statCands.Store(int64(rowN))
   769→                                return out, true, nil
   770→                        }
   771→                }
   772→        }
   773→        e.statCands.Store(int64(rowN))
   774→        return out, false, nil
   775→}

   777→// sampleTrigrams نمونهٔ پخش‌شده از سه‌گانه‌ها برای پرس‌وجوی پیش‌فیلتر —
   778→// زیرمجموعه از سه‌گانه‌های عبارت است، پس صدا حفظ می‌شود (ابر‌مجموعهٔ کاندید).
   779→func sampleTrigrams(tris []string, k int) []string {
   780→        if len(tris) <= k {
   781→                return tris
   782→        }
   783→        step := (len(tris) + k - 1) / k
   784→        var out []string
   785→        for i := 0; i < len(tris) && len(out) < k; i += step {
   786→                out = append(out, tris[i])
   787→        }
   788→        return out
   789→}

   791→// faNum نمایش عدد با ارقام فارسی برای پیام‌های کاربر.
   792→func faNum(n int) string {
   793→        fd := []rune("۰۱۲۳۴۵۶۷۸۹")
   794→        s := strconv.Itoa(n)
   795→        out := make([]rune, 0, len(s))
   796→        for _, c := range s {
   797→                if c >= '0' && c <= '9' {
   798→                        out = append(out, fd[c-'0'])
   799→                } else {
   800→                        out = append(out, c)
   801→                }
   802→        }
   803→        return string(out)
   804→}

   807→        Range [2]int // بازهٔ متن اصلی (code point)
   808→}

   810→// verifyInText بسته به mode تطبیق را روی متن واقعی همان نسخه تأیید می‌کند.
   811→func verifyInText(req *Request, r *candRow) ([]hit, string) {
   812→        orig := []rune(r.TextOriginal)
   813→        n1 := []rune(r.TextN1)
   814→        var n1Map []int
   815→        _ = json.Unmarshal([]byte(r.N1Map), &n1Map)
   816→        switch req.Mode {
   817→        case StrictPhrase:
   818→                q := []rune(req.Query)
   819→                var out []hit
   820→                start := 0
   821→                for {
   822→                        idx := indexRunes(orig[start:], q)
   823→                        if idx < 0 {
   824→                                break
   825→                        }
   826→                        s := start + idx
   827→                        e := s + len(q)
   828→                        if norm.WordBoundary(string(orig), s, e) {
   829→                                out = append(out, hit{Range: [2]int{s, e}})
   830→                        }
   831→                        start = s + 1
   832→                        if len(out) >= 50 {
   833→                                break
   834→                        }
   835→                }
   836→                return out, "exact"
   837→        default:
   838→                qN, _ := norm.Normalize(req.Query)
   839→                if req.Mode == Linguistic {
   840→                        qN = norm.Fold(req.Query)
   841→                }
   842→                if req.Mode == AllWords || req.Mode == Near {
   843→                        toks := norm.Tokenize(qN)
   844→                        positions := make([][]int, len(toks))
   845→                        for i, t := range toks {
   846→                                positions[i] = findAllWord(n1, t)
   847→                        }
   848→                        var out []hit
   849→                        if req.Mode == Near && req.Near != nil && req.Near.Slop > 0 {
   850→                                for _, rng := range nearCombination(positions, req.Near.Slop, req.Near.Ordered, len(n1)) {
   851→                                        out = append(out, mapN1Hit(rng, n1Map))
   852→                                }
   853→                                return out, "normalized"
   854→                        }
   855→                        if allPresent(positions) {
   856→                                lo, hi := -1, -1
   857→                                for i, ps := range positions {
   858→                                        if i == 0 || ps[0] < lo {
   859→                                                lo = ps[0]
   860→                                        }
   861→                                        last := ps[len(ps)-1] + len([]rune(toks[i]))
   862→                                        if i == 0 || last > hi {
   863→                                                hi = last
   864→                                        }
   865→                                }
   866→                                out = append(out, mapN1Hit([2]int{lo, hi}, n1Map))
   867→                        }
   868→                        return out, "normalized"
   869→                }
   870→                // normalized_phrase / linguistic / long_quote: تطبیق متوالی با مرز واژه
   871→                qRunes := []rune(qN)
   872→                var out []hit
   873→                start := 0
   874→                for {
   875→                        idx := indexRunes(n1[start:], qRunes)
   876→                        if idx < 0 {
   877→                                break
   878→                        }
   879→                        s := start + idx
   880→                        e := s + len(qRunes)
   881→                        if norm.WordBoundary(string(n1), s, e) {
   882→                                out = append(out, mapN1Hit([2]int{s, e}, n1Map))
   883→                        }
   884→                        start = s + 1
   885→                        if len(out) >= 50 {
   886→                                break
   887→                        }
   888→                }
   889→                if req.Mode == Linguistic {
   890→                        return out, "linguistic"
   891→                }
   892→                return out, "normalized"
   893→        }
   894→}

   896→// mapN1Hit نگاشت بازهٔ نرمال → بازهٔ متن اصلی (highlight روی اصل — SEARCH-11).
   897→func mapN1Hit(rng [2]int, n1Map []int) hit {
   898→        if len(n1Map) == 0 {
   899→                return hit{Range: rng}
   900→        }
   901→        s := clampInt(rng[0], 0, len(n1Map)-1)
   902→        e := clampInt(rng[1], s, len(n1Map))
   903→        if e > s {
   904→                return hit{Range: [2]int{n1Map[s], n1Map[e-1] + 1}}
   905→        }
   906→        return hit{Range: [2]int{n1Map[s], n1Map[s]}}
   907→}

   946→func nearCombination(positions [][]int, slop int, ordered bool, textLen int) [][2]int {
   947→        var out [][2]int
   948→        if len(positions) == 0 || !allPresent(positions) {
   949→                return nil
   950→        }
   951→        idx := make([]int, len(positions))
   952→        for {
   953→                var lo, hi int
   954→                for i, p := range positions {
   955→                        pos := p[idx[i]]
   956→                        if i == 0 || pos < lo {
   957→                                lo = pos
   958→                        }
   959→                        if i == 0 || pos > hi {
   960→                                hi = pos
   961→                        }
   962→                }
   963→                ok := (hi - lo) <= slop*6
   964→                if ok && ordered && len(positions) > 1 {
   965→                        for i := 1; i < len(positions); i++ {
   966→                                if positions[i][idx[i]] < positions[i-1][idx[i-1]] {
   967→                                        ok = false
   968→                                        break
   969→                                }
   970→                        }
   971→                }
   972→                if ok {
   973→                        out = append(out, [2]int{lo, clampInt(hi+len(positions[len(positions)-1]), 0, textLen)})
   974→                }
   975→                i := len(idx) - 1
   976→                for i >= 0 {
   977→                        idx[i]++
   978→                        if idx[i] < len(positions[i]) {
   979→                                break
   980→                        }
   981→                        idx[i] = 0
   982→                        i--
   983→                }
   984→                if i < 0 || len(out) >= 10 {
   985→                        break
   986→                }
   987→        }
   988→        return out
   989→}

  1204→func indexRunes(s []rune, sub []rune) int {
  1205→        n := len(sub)
  1206→        if n == 0 || n > len(s) {
  1207→                return -1
  1208→        }
  1209→        for i := 0; i+n <= len(s); i++ {
  1210→                match := true
  1211→                for j := 0; j < n; j++ {
  1212→                        if s[i+j] != sub[j] {
  1213→                                match = false
  1214→                                break
  1215→                        }
  1216→                }
  1217→                if match {
  1218→                        return i
  1219→                }
  1220→        }
  1221→        return -1
  1222→}

  1292→// between بررسی تعلق آیهٔ پرس‌شده به بازهٔ استناد (اعداد فارسی/عربی/لاتین نرمال می‌شوند).
  1293→func between(from, to, aya string) bool {

  1296→        a := numNorm(aya)
  1297→        if f == "" || t == "" || a == "" {

  1299→        }

  1303→func numNorm(s string) string {
  1304→        digits := map[rune]string{
  1305→                '٠': "0", '١': "1", '٢': "2", '٣': "3", '٤': "4", '٥': "5", '٦': "6", '٧': "7", '٨': "8", '٩': "9",
  1306→                '۰': "0", '۱': "1", '۲': "2", '۳': "3", '۴': "4", '۵': "5", '۶': "6", '۷': "7", '۸': "8", '۹': "9",
  1307→        }
  1308→        var b []byte
  1309→        for _, r := range s {
  1310→                if d, ok := digits[r]; ok {
  1311→                        b = append(b, d...)
  1312→                } else if r >= '0' && r <= '9' {
  1313→                        b = append(b, byte(r))
  1314→                }
  1315→        }
  1316→        return string(b)
  1317→}
