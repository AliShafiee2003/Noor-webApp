    66→function clampPage(raw: string | undefined): number {
    67→  const n = Number.parseInt(raw ?? "", 10);
    68→  return Number.isFinite(n) && n >= 1 && n <= 200 ? n : 1;
    69→}

    71→/** بازسازی hash از payload جست‌وجوی ذخیره‌شده (اجرای دوباره با همان نشانی) */
    72→function hashFromPayload(payload: unknown): Record<string, string> | null {
    73→  if (!payload || typeof payload !== "object") return null;
    74→  const p = payload as Record<string, unknown>;
    75→  if (typeof p.query !== "string" || typeof p.mode !== "string") return null;
    76→  const out: Record<string, string> = { q: p.query, mode: p.mode };
    77→  const scope = (p.scope && typeof p.scope === "object" ? p.scope : {}) as Record<string, unknown>;
    78→  const target = typeof scope.target === "string" ? scope.target : "library";
    79→  if (target === "titles") {
    80→    out.scope = "titles";
    81→  } else if (target === "work" && typeof scope.work_uid === "string" && scope.work_uid) {
    82→    out.scope = "work";
    83→    out.work = scope.work_uid;
    84→  } else if (
    85→    target === "document" &&
    86→    Array.isArray(scope.document_uids) &&
    87→    typeof scope.document_uids[0] === "string"
    88→  ) {
    89→    out.scope = "document";
    90→    out.doc = scope.document_uids[0];
    91→    if (typeof scope.work_uid === "string") out.work = scope.work_uid;
    92→  } else {
    93→    out.scope = "library";
    94→  }
    95→  const roles = Array.isArray(scope.roles)
    96→    ? scope.roles.filter((r): r is string => typeof r === "string")
    97→    : [];
    98→  const roleValues = ROLE_FILTERS.filter((f) => f.codes.some((c) => roles.includes(c))).map(
    99→    (f) => f.value,
   100→  );
   101→  if (roleValues.length > 0) out.roles = roleValues.join(",");
   102→  if (typeof scope.language === "string" && scope.language) out.lang = scope.language;
   103→  const near = (p.near && typeof p.near === "object" ? p.near : null) as Record<string, unknown> | null;
   104→  if (near) {
   105→    out.slop = String(typeof near.slop === "number" ? near.slop : 3);
   106→    if (near.ordered === false) out.ordered = "0";
   107→  }
   108→  return out;
   109→}

   314→export function SearchView() {
   315→  const { route, navigate } = useRoute();
   316→
   317→  /* وضعیت اجراشدهٔ جست‌وجو (از hash) */
   318→  const params = parseSearchParams(route.query);
   319→  const searchKey = useMemo(() => searchStateKey(params), [params]);
   320→  const hashPage = clampPage(route.query.page);
   321→  const [saveOpen, setSaveOpen] = useState(false);
   322→  const view: "results" | "books" = route.query.view === "books" ? "books" : "results";
   323→  const hasQuery = params.q.trim().length > 0;
   324→
   325→  /* پیش‌نویس فرم — هم‌گام‌سازی با hash در زمان رندر (الگوی مجاز React) */
   326→  const [draft, setDraft] = useState<SearchHashState>(() => parseSearchParams(route.query));
   327→  const [draftHash, setDraftHash] = useState(route.fullHash);
   328→  if (draftHash !== route.fullHash) {
   329→    setDraftHash(route.fullHash);
   330→    setDraft(parseSearchParams(route.query));
   331→  }
   332→
   333→  /** نوشتن وضعیت در hash — هر بار کل پارامترها داده می‌شود */
   334→  const setQuery = useCallback(
   335→    (state: SearchHashState, extra: Record<string, string | undefined> = {}, replace = false) => {
   336→      navigate(
   337→        "/search",
   338→        {
   339→          q: state.q.trim() || undefined,
   340→          mode: state.mode,
   341→          scope: state.scope,
   342→          work: state.work || undefined,
   343→          doc: state.doc || undefined,
   344→          roles: state.roles.length > 0 ? state.roles.join(",") : undefined,
   345→          lang: state.lang || undefined,
   346→          slop: state.mode === "near" ? String(state.slop) : undefined,
   347→          ordered: state.mode === "near" && !state.ordered ? "0" : undefined,
   348→          page: undefined,
   349→          view: undefined,
   350→          ...extra,
   351→        },
   352→        { replace },
   353→      );
   354→    },
   355→    [navigate],
   356→  );
   357→
   358→  const submit = useCallback(() => {
   359→    setQuery(draft);
   360→  }, [draft, setQuery]);
   361→
   362→  const patchCommitted = useCallback(
   363→    (patch: Partial<SearchHashState>) => {
   364→      setQuery({ ...params, ...patch });
   365→    },
   366→    [params, setQuery],
   367→  );
   368→
   369→  /* لایهٔ ۱ — نشست جست‌وجو: صفحهٔ نخست + شمارش صادقانه */
   370→  const request = buildSearchRequest(params);
   371→
   372→  const sessionQuery = useQuery({
   373→    queryKey: ["search-session", searchKey],
   374→    queryFn: async ({ signal }) => {
   375→      const resp = await api.post<SearchResponse>("/search", request);
   376→      if (signal.aborted) throw new DOMException("aborted", "AbortError");
   377→      return resp;
   378→    },
   379→    enabled: hasQuery,
   380→    staleTime: 4 * 60_000,
   381→    gcTime: 10 * 60_000,
   382→    retry: false,
   383→    refetchOnWindowFocus: false,
   384→    refetchOnReconnect: false,
   385→  });
   386→
   387→  const session = hasQuery ? sessionQuery.data ?? null : null;
   388→  const sessionUid = session?.session_uid ?? null;
   389→  const sessionUidRef = useRef<string | null>(null);
   390→  useEffect(() => {
   391→    sessionUidRef.current = sessionUid;
   392→  }, [sessionUid]);
   393→
   394→  /* لایهٔ ۲ — صفحه‌های بیشتر با «ادامهٔ بررسی» (cursor امضاشده، §۴۷) */
   395→  const [extras, setExtras] = useState<ExtrasState>(EMPTY_EXTRAS);
   396→  const currentExtras = sessionUid && extras.uid === sessionUid ? extras : EMPTY_EXTRAS;
   397→  const extrasRef = useRef(currentExtras);
   398→  useEffect(() => {
   399→    extrasRef.current = currentExtras;
   400→  }, [currentExtras]);
   401→
   402→  const meta = currentExtras.meta ?? session;
   403→  const items = useMemo(
   404→    () => [...(session?.results ?? []), ...currentExtras.items],
   405→    [session, currentExtras.items],
   406→  );
   407→
   408→  const restoreStampRef = useRef<string>("");
   409→
   410→  const runContinue = useCallback(
   411→    async (targetSessionUid: string, count: number, updateHash: boolean) => {
   412→      let acc: ExtrasState =
   413→        extrasRef.current.uid === targetSessionUid
   414→          ? extrasRef.current
   415→          : { ...EMPTY_EXTRAS, uid: targetSessionUid };
   416→      if (acc.loading) return;
   417→      acc = { ...acc, loading: true, error: null };
   418→      extrasRef.current = acc;
   419→      setExtras(acc);
   420→
   421→      for (let i = 0; i < count; i += 1) {
   422→        if (sessionUidRef.current !== targetSessionUid) return; // جست‌وجوی تازه‌تری اجرا شده است
   423→        const cursorMeta = acc.meta ?? session ?? null;
   424→        if (!cursorMeta?.continuation_available || !cursorMeta.next_cursor) break;
   425→        try {
   426→          const resp = await api.post<SearchResponse>(
   427→            `/search/${targetSessionUid}/continue`,
   428→            { cursor: cursorMeta.next_cursor },
   429→          );
   430→          if (sessionUidRef.current !== targetSessionUid) return;
   431→          acc = {
   432→            ...acc,
   433→            count: acc.count + 1,
   434→            items: [...acc.items, ...(resp.results ?? [])],
   435→            meta: resp,
   436→            loading: i < count - 1,
   437→          };
   438→        } catch (err) {
   439→          if (err instanceof ApiError && err.code === "gone_cursor") {
   440→            acc = { ...acc, expired: true, loading: false };
   441→          } else {
   442→            acc = {
   443→              ...acc,
   444→              error: errorMessage(err, "ادامهٔ بررسی ناموفق بود؛ دوباره تلاش کنید."),
   445→              loading: false,
   446→            };
   447→          }
   448→          break;
   449→        }
   450→        extrasRef.current = acc;
   451→        setExtras(acc);
   452→      }
   453→      acc = { ...acc, loading: false };
   454→      extrasRef.current = acc;
   455→      setExtras(acc);
   456→
   457→      // هم‌گام‌سازی موضع در hash (WORK-07): شمارهٔ صفحه = شمار صفحه‌های اضافه + ۱
   458→      if (updateHash || acc.expired || acc.error) {
   459→        navigate(
   460→          "/search",
   461→          { ...route.query, page: acc.count > 0 ? String(acc.count + 1) : undefined },
   462→          { replace: true },
   463→        );
   464→      }
   465→    },
   466→    [navigate, route.query, session],  
   467→  );
   468→
   469→  /* بازسازی موضع پس از بازگشت به نتایج (WORK-07): زنجیرهٔ continue تا صفحهٔ hash */
   470→  useEffect(() => {
   471→    const targetExtras = hashPage - 1;
   472→    if (!sessionUid || targetExtras <= 0 || currentExtras.loading) return;
   473→    if (currentExtras.count >= targetExtras) return;
   474→    if (meta && !meta.continuation_available) return;
   475→    const stamp = `${searchKey}|${sessionUid}|${targetExtras}`;
   476→    if (restoreStampRef.current === stamp) return;
   477→    restoreStampRef.current = stamp;
   478→    const t = window.setTimeout(() => {
   479→      void runContinue(sessionUid, targetExtras - currentExtras.count, false);
   480→    }, 0);
   481→    return () => window.clearTimeout(t);
   482→  }, [sessionUid, hashPage, currentExtras.count, currentExtras.loading, meta, searchKey, runContinue]);
   483→
   484→  /* وضعیت نمایه — تفکیک «بدون نتیجه» از «نمایه آماده نیست» (SEARCH-21) */
   485→  const isEmpty = sessionQuery.isSuccess && items.length === 0 && !currentExtras.loading;
   486→  const statusQuery = useQuery({
   487→    queryKey: ["library-status"],
   488→    queryFn: ({ signal }) =>
   489→      api.get<{ index_ready?: boolean | null }>("/status", undefined, signal),
   490→    enabled: isEmpty,
   491→    staleTime: 60_000,
   492→  });
   493→
   494→  const loadingMore = currentExtras.loading;
   495→  const switchView = (next: string) => {
