    52→export function ReaderView({ documentUid }: { documentUid: string }) {
    53→  const queryClient = useQueryClient();
    54→  const prefs = useWorkspaceStore((s) => s.prefs);
    55→  const isCompact = useIsCompact();
    56→
    57→  /* ---------- پارامترهای hash: صفحهٔ فعلی و highlight نتیجه ---------- */
    58→  const [params, setParams] = useState(() => readHashParams());
    59→  useEffect(() => {
    60→    const onHash = () => setParams(readHashParams());
    61→    window.addEventListener("hashchange", onHash);
    62→    return () => window.removeEventListener("hashchange", onHash);
    63→  }, []);
    64→  const requestedPageUid = params.get("pg") || null;
    65→  const hl = parseHighlight(params.get("hl"));
    66→
    67→  /* ---------- داده‌ها ---------- */
    68→  const workQuery = useQuery({
    69→    queryKey: ["reader-work", documentUid],
    70→    queryFn: async () => {
    71→      const works = await api.get<{ items: WorkDetail[] }>("/works");
    72→      for (const w of works.items) {
    73→        const full = await api.get<WorkDetail>(`/works/${w.uid}`);
    74→        if (full.documents.some((d) => d.uid === documentUid)) return full;
    75→      }
    76→      throw new Error("این سند در کتابخانهٔ موجود نیست.");
    77→    },
    78→    retry: false,
    79→  });
    80→
    81→  const tocQuery = useQuery({
    82→    queryKey: ["reader-toc", documentUid],
    83→    queryFn: () =>
    84→      api.get<{ sections: TocSection[]; release_uid: string }>(
    85→        `/documents/${documentUid}/toc`,
    86→      ),
    87→    retry: false,
    88→  });
    89→
    90→  const pagesQuery = useQuery({
    91→    queryKey: ["reader-pages", documentUid],
    92→    queryFn: () =>
    93→      api.get<{ items: PageRef[] }>(`/documents/${documentUid}/pages`),
    94→    retry: false,
    95→  });
    96→
    97→  const pageRefs: PageRef[] = pagesQuery.data?.items ?? [];
    98→
    99→  /* صفحهٔ هدف: از hash یا ادامهٔ مطالعه (بازیابی موضع — READ-12) */
   100→  const [fallbackPageUid, setFallbackPageUid] = useState<string | null>(null);
   101→  const positionQuery = useQuery({
   102→    queryKey: ["workspace-position", documentUid],
   103→    queryFn: () => api.get<{ positions: Record<string, { page_uid?: string }> }>("/workspace"),
   104→    staleTime: 10_000,
   105→  });
   106→  useEffect(() => {
   107→    if (requestedPageUid || fallbackPageUid || !pagesQuery.isSuccess) return;
   108→    const saved = positionQuery.data?.positions?.[documentUid]?.page_uid;
   109→    if (saved && pageRefs.some((p) => p.page_uid === saved)) {
   110→      setFallbackPageUid(saved);
   111→    } else if (pageRefs.length > 0) {
   112→      setFallbackPageUid(pageRefs[0].page_uid);
   113→    }
   114→  }, [requestedPageUid, fallbackPageUid, pagesQuery.isSuccess, pageRefs, positionQuery.data, documentUid]);
   115→
   116→  const currentPageUid = requestedPageUid ?? fallbackPageUid;
   117→
   118→  const pageQuery = useQuery({
   119→    queryKey: ["reader-page", documentUid, currentPageUid],
   120→    enabled: !!currentPageUid,
   121→    queryFn: () =>
   122→      api.get<PagePayload>(`/documents/${documentUid}/pages/${currentPageUid}`),
   123→    retry: false,
   124→  });
   125→
   126→  const bookmarksQuery = useQuery({
   127→    queryKey: ["reader-bookmarks"],
   128→    queryFn: () => api.get<{ items: BookmarkT[] }>("/bookmarks"),
   129→  });
   130→
   131→  /* یادداشت‌های شخصی (RES) — شمار برای نشان؛ دادهٔ کامل در NotesPanel با همان کلید */
   132→  const notesCountQuery = useQuery({
   133→    queryKey: ["notes", documentUid],
   134→    queryFn: () => api.get<{ items: NoteT[] }>("/notes", { document_uid: documentUid }),
   135→    staleTime: 15_000,
   136→    retry: false,
   137→  });
   138→  const pageNoteCount =
   139→    notesCountQuery.data?.items.filter((n) => n.page_uid === currentPageUid).length ?? 0;
   140→  const docNoteCount = notesCountQuery.data?.items.length ?? 0;
   141→
   142→  /* ---------- نگاشت بلوک→بخش برای لنگر فهرست ---------- */
   143→  const sectionUidByBlockId = useMemo(() => {
   144→    const map: Record<string, string> = {};
   145→    const walk = (secs: TocSection[]) => {
   146→      for (const s of secs) {
   147→        if (s.block_id) map[s.block_id] = s.uid;
   148→        walk(s.children ?? []);
   149→      }
   150→    };
   151→    walk(tocQuery.data?.sections ?? []);
   152→    return map;
   153→  }, [tocQuery.data]);
   154→
   155→  const activeSectionUid = useMemo(() => {
   156→    if (!pageQuery.data) return null;
   157→    for (const b of pageQuery.data.blocks) {
   158→      const uid = sectionUidByBlockId[b.id];
   159→      if (uid) return uid;
   160→    }
   161→    return null;
   162→  }, [pageQuery.data, sectionUidByBlockId]);
   163→
   164→  /* ---------- حالت‌های محلی ---------- */
   165→  const [tocOpen, setTocOpen] = useState(false);
   166→  const [flashBlockId, setFlashBlockId] = useState<string | null>(null);
   167→  const [footnote, setFootnote] = useState<FootnotePanelData | null>(null);
   168→  const [selection, setSelection] = useState<ReaderSelection | null>(null);
   169→  const [ficheOpen, setFicheOpen] = useState(false);
   170→  const [saveFlash, setSaveFlash] = useState<string | null>(null);
   171→  /** تب پنل زمینه: پاورقی/آیه/روایت یا یادداشت‌ها */
   172→  const [ctxTab, setCtxTab] = useState<"context" | "notes">("context");
   173→  /** شیت یادداشت‌ها در موبایل/تبلت */
   174→  const [notesSheetOpen, setNotesSheetOpen] = useState(false);
   175→  /* §۴۷: پرس‌وجوی میان‌کتابی روایت از پنل زمینه */
   176→  const [hadithLookupUid, setHadithLookupUid] = useState<string | null>(null);
   177→  /* RES: دامنهٔ روایت برای یادداشت روی روایت مشخص */
   178→  const [hadithNoteScope, setHadithNoteScope] = useState<{
   179→    localUid: string;
   180→    revayatIndex?: string | null;
   181→  } | null>(null);
   182→
   183→  /* تغییر صفحهٔ مقصد → پنل زمینه بسته می‌شود (پنل مالِ همان صفحه است) */
   184→  useEffect(() => {
   185→    setFootnote(null);
   186→    setSelection(null);
   187→    setHadithNoteScope(null);
   188→  }, [currentPageUid]);
   189→
   190→  /* بازشدن پاورقی/ارجاع → تب زمینه فعال می‌شود تا محتوا دیده شود */
   191→  useEffect(() => {
   192→    if (footnote) setCtxTab("context");
   193→  }, [footnote]);
   194→
   195→  /* highlight نتیجهٔ جست‌وجو → بلوک مقصد */
   196→  /* آفست hl سطح سند است؛ با source_offsets بلوک‌ها به مختصات درون‌بلوکی نگاشت می‌شود (بدون hlb) */
   197→  const highlight = useMemo(() => {
   198→    if (!hl || !pageQuery.data) return null;
   199→    const explicitBlock = params.get("hlb");
   200→    if (explicitBlock) return { start: hl[0], end: hl[1], blockId: explicitBlock };
   201→    for (const block of pageQuery.data.blocks) {
   202→      const raw = block.source_offsets;
   203→      const range = Array.isArray(raw)
   204→        ? { start: raw[0], end: raw[1] }
   205→        : raw && typeof raw.start === "number" && typeof raw.end === "number"
   206→          ? { start: raw.start, end: raw.end }
   207→          : null;
   208→      if (range && hl[0] >= range.start && hl[0] < range.end) {
   209→        const relStart = hl[0] - range.start;
   210→        const relEnd = Math.max(relStart + 1, Math.min(hl[1], range.end) - range.start);
   211→        return { start: relStart, end: relEnd, blockId: block.id };
   212→      }
   213→    }
   214→    return null;
   215→  }, [hl, pageQuery.data, params]);
   216→
   217→  /* ---------- پرش فهرست / صفحه ---------- */
   218→  const goToPage = useCallback(
   219→    (pageUid: string, extra: Record<string, string> = {}) => {
   220→      const q = new URLSearchParams({ pg: pageUid, ...extra });
   221→      window.location.hash = `#/read/${documentUid}?${q.toString()}`;
   222→      setFootnote(null);
   223→    },
   224→    [documentUid],
   225→  );
   226→
   227→  const onJumpSection = useCallback(
   228→    (s: TocSection) => {
   229→      // مقصد فهرست: صفحهٔ دارای آن بلوک (از سمت کاربر: بخش فعلیِ همان صفحه)
   230→      // block_id به صفحه نگاشت نمی‌شود در قرارداد؛ برای سادگی، اگر در صفحهٔ فعلی هست flash می‌کنیم
   231→      setTocOpen(false);
   232→      if (pageQuery.data?.blocks.some((b) => b.id === s.block_id)) {
   233→        setFlashBlockId(s.block_id ?? null);
   234→        window.setTimeout(() => setFlashBlockId(null), 1800);
   235→      } else if (s.block_id) {
   236→        // اسکرول در حالت پیوسته روی همان صفحهٔ فعلی
   237→        const el = document.getElementById(`blk-${s.block_id}`);
   238→        if (el) {
   239→          el.scrollIntoView({ behavior: "smooth", block: "start" });
   240→          setFlashBlockId(s.block_id);
   241→          window.setTimeout(() => setFlashBlockId(null), 1800);
   242→        }
   243→      }
   244→    },
   245→    [pageQuery.data],
   246→  );
   247→
   248→  /* ---------- ذخیرهٔ موضع (READ-12) — PUT /workspace/position با debounce ---------- */
   249→  const putTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   250→  useEffect(() => {
   251→    if (!currentPageUid || !pageQuery.isSuccess) return;
   252→    if (putTimer.current) clearTimeout(putTimer.current);
   253→    putTimer.current = setTimeout(() => {
   254→      void api.put("/workspace/position", {
   255→        document_uid: documentUid,
   256→        position: { page_uid: currentPageUid, progress: 0 },
   257→      });
   258→    }, 900);
   259→    return () => {
   260→      if (putTimer.current) clearTimeout(putTimer.current);
   261→    };
   262→  }, [currentPageUid, pageQuery.isSuccess, documentUid]);
   263→
   264→  /* ---------- نشانک ---------- */
   265→  const isBookmarked = bookmarksQuery.data?.items.some(
   266→    (b) => b.document_uid === documentUid && b.page_uid === currentPageUid,
   267→  ) ?? false;
   268→  const [bookmarkPending, setBookmarkPending] = useState(false);
   269→  const toggleBookmark = useCallback(async () => {
   270→    if (!currentPageUid || bookmarkPending) return;
   271→    setBookmarkPending(true);
   272→    try {
   273→      if (isBookmarked) {
   274→        const existing = bookmarksQuery.data?.items.find(
   275→          (b) => b.document_uid === documentUid && b.page_uid === currentPageUid,
   276→        );
   277→        if (existing) await api.del(`/bookmarks/${existing.uid}`);
   278→      } else {
   279→        await api.post("/bookmarks", {
   280→          document_uid: documentUid,
   281→          page_uid: currentPageUid,
   282→          label: pageQuery.data?.label ?? "",
   283→        });
   284→      }
   285→      await queryClient.invalidateQueries({ queryKey: ["reader-bookmarks"] });
   286→    } finally {
   287→      setBookmarkPending(false);
   288→    }
   289→  }, [currentPageUid, bookmarkPending, isBookmarked, bookmarksQuery.data, documentUid, pageQuery.data?.label, queryClient]);
   290→
   291→  /* ---------- کپی نشانی (READ-17) ---------- */
   292→  const copyAddress = useCallback(async () => {
   293→    const parts = [
   294→      workQuery.data?.title ?? "",
   295→      "جلد " + (workQuery.data?.documents.find((d) => d.uid === documentUid)?.label ?? ""),
   296→      "صفحهٔ " + (pageQuery.data?.label ?? ""),
   297→    ].filter(Boolean);
   298→    const text = parts.join("، ") + "\n" + window.location.hash;
   299→    try {
   300→      await navigator.clipboard.writeText(text);
   301→      setSaveFlash("نشانی کپی شد");
   302→    } catch {
   303→      setSaveFlash("کپی ممکن نشد؛ دستی انتخاب کنید");
   304→    }
   305→    window.setTimeout(() => setSaveFlash(null), 1600);
   306→  }, [workQuery.data, documentUid, pageQuery.data?.label]);
   307→
   308→  /* ---------- انتخاب متن → فیش ---------- */
   309→  const onSelection = useCallback((sel: ReaderSelection | null) => {
   310→    setSelection(sel);
   311→    if (sel) setFicheOpen(true);
   312→  }, []);
   313→
   314→  const onFootnoteClick = useCallback(
   315→    (refUid: string) => {
   316→      const fn = pageQuery.data?.footnotes.find((f) => f.uid === refUid);
   317→      if (fn) {
   318→        setFootnote({ kind: "footnote", uid: fn.uid, label: fn.label, text: fn.text });
   319→      } else {
   320→        setFootnote({ kind: "link", originalAddress: "پاورقی " + refUid });
   321→      }
   322→    },
   323→    [pageQuery.data],
   324→  );
   325→
   326→  const onLinkClick = useCallback((link: BlockLink) => {
   327→    // همهٔ ارجاع‌ها در این نسخه حل‌نشده‌اند — نشانی اصلی حفظ می‌شود (REL-10)
   328→    setFootnote({ kind: "link", originalAddress: link.original_address || "ارجاع" });
   329→  }, []);
   330→
   331→  /* ---------- پنل آیه (QUR-01/02) و اطلاعات روایت (HAD-08) ---------- */
   332→  const onVerseClick = useCallback((block: PageBlock) => {
   333→    const cit = block.citations?.[0];
   334→    if (!cit) return;
   335→    setFootnote({
   336→      kind: "verse",
   337→      verse: {
   338→        citation: cit,
   339→        quotedText: block.text,
   340→        inHadith: block.hadith_uid ?? cit.in_hadith ?? null,
   341→        hasParent: cit.has_parent,
   342→      },
   343→    });
   344→  }, []);
   345→
   346→  const onHadithClick = useCallback((block: PageBlock) => {
   347→    if (!block.hadith_uid) return;
   348→    setFootnote({
   349→      kind: "hadith",
   350→      hadith: {
   351→        localUid: block.hadith_uid,
   352→        revayatIndex: block.revayat_index ?? null,
   353→      },
   354→    });
   355→  }, []);
   356→
   357→  /* ---------- RES: ثبت فیش این روایت — متن کامل روایتِ همین صفحه ----------
   358→     همهٔ بلوک‌های همان روایت در صفحه جمع می‌شود (روایت ممکن است چند بلوک باشد؛
   359→     بازه = آغاز نخستین تا پایان آخرین بلوک). متن با \n میان بلوک‌ها می‌پیوندد. */
   360→  const hadithFiche = useCallback(
   361→    (localUid: string) => {
   362→      if (!pageQuery.data) return false;
   363→      const parts = pageQuery.data.blocks.filter((b) => b.hadith_uid === localUid);
   364→      if (parts.length === 0) return false;
   365→      const first = blockOffsets(parts[0]);
   366→      const last = blockOffsets(parts[parts.length - 1]);
   367→      if (!first || !last) return false;
   368→      setSelection({
   369→        blockId: parts[0].id,
   370→        start: first.start,
   371→        end: last.end,
   372→        text: parts.map((b) => b.text).join("\n"),
   373→        hadithUid: localUid,
   374→      });
   375→      setFicheOpen(true);
   376→      return true;
   377→    },
   378→    [pageQuery.data],
   379→  );
   380→
   381→  const onHadithFiche = useCallback(
   382→    (localUid: string) => {
   383→      const ok = hadithFiche(localUid);
   384→      if (ok && isCompact) {
   385→        setFootnote(null);
   386→      }
   387→    },
   388→    [hadithFiche, isCompact],
   389→  );
   390→
   391→  const onSearchInDocument = useCallback(
   392→    (q: string) => {
   393→      window.location.hash = `#/search?q=${encodeURIComponent(q)}&mode=normalized_phrase&scope=document&doc=${documentUid}`;
   394→    },
   395→    [documentUid],
   396→  );
   397→
   398→  /* ---------- ناوبری صفحه‌ای ---------- */
   399→  const pageIdx = pageRefs.findIndex((p) => p.page_uid === currentPageUid);
   400→  const prevRef = pageIdx > 0 ? pageRefs[pageIdx - 1] : null;
   401→  const nextRef = pageIdx >= 0 && pageIdx < pageRefs.length - 1 ? pageRefs[pageIdx + 1] : null;
   402→
   403→  /* ---------- بارگذاری keyboard ---------- */
   404→  useEffect(() => {
   405→    const onKey = (e: KeyboardEvent) => {
   406→      const tag = (e.target as HTMLElement)?.tagName;
   407→      if (tag === "INPUT" || tag === "TEXTAREA" || (e.target as HTMLElement)?.isContentEditable) return;
   408→      if (e.key === "ArrowLeft" && nextRef) goToPage(nextRef.page_uid);
   409→      if (e.key === "ArrowRight" && prevRef) goToPage(prevRef.page_uid);
   410→    };
   411→    window.addEventListener("keydown", onKey);
   412→    return () => window.removeEventListener("keydown", onKey);
   413→  }, [nextRef, prevRef, goToPage]);
   414→
   415→  /* ---------- حالت‌های بارگذاری/خطا ---------- */
   416→  const errMsg =
   417→    workQuery.error instanceof Error
   418→      ? workQuery.error.message
   419→      : pagesQuery.error instanceof Error
   420→        ? pagesQuery.error.message
   421→        : "";
   422→
   423→  if (workQuery.isPending || pagesQuery.isPending) {
   424→    return (
   425→      <div className="mx-auto w-full max-w-3xl space-y-4 p-4">
   426→        <Skeleton className="h-10 w-2/3" />
   427→        <Skeleton className="h-4 w-1/3" />
   428→        <Skeleton className="h-64 w-full" />
   429→        <p className="flex items-center gap-2 text-sm text-muted-foreground">
   430→          <Loader2 className="size-4 animate-spin" /> در حال بازکردن متن…
   431→        </p>
   432→      </div>
   433→    );
   434→  }
   435→
   436→  if (workQuery.isError || pagesQuery.isError) {
   437→    return (
   438→      <div className="mx-auto max-w-md p-8 text-center">
   439→        <p className="text-sm text-destructive">
   440→          بازکردن این سند ممکن نشد{errMsg ? ": " + errMsg : ""}
   441→        </p>
   442→        <Button variant="outline" className="mt-3" onClick={() => { workQuery.refetch(); pagesQuery.refetch(); }}>
   443→          تلاش دوباره
   444→        </Button>
   445→      </div>
   446→    );
   447→  }
   448→
   449→  const work = workQuery.data!;
   450→  const docLabel = work.documents.find((d) => d.uid === documentUid)?.label ?? "";
   451→
   452→  const tocPanel = (
   453→    <TocPanel
   454→      sections={tocQuery.data?.sections ?? []}
   455→      isLoading={tocQuery.isPending}
   456→      error={tocQuery.isError ? "فهرست در دسترس نیست." : null}
   457→      onRetry={() => tocQuery.refetch()}
   458→      activeSectionUid={activeSectionUid}
   459→      onJumpSection={onJumpSection}
   460→      bookmarks={bookmarksQuery.data?.items.filter((b) => b.document_uid === documentUid) ?? []}
   461→      bookmarksLoading={bookmarksQuery.isPending}
   462→      onJumpBookmark={(b) => goToPage(b.page_uid)}
   463→      onDeleteBookmark={async (b) => {
   464→        await api.del(`/bookmarks/${b.uid}`);
   465→        await queryClient.invalidateQueries({ queryKey: ["reader-bookmarks"] });

   802→function readHashParams(): URLSearchParams {
   803→  const hash = window.location.hash;
   804→  const qi = hash.indexOf("?");
   805→  return new URLSearchParams(qi >= 0 ? hash.slice(qi + 1) : "");
   806→}

   808→function parseHighlight(raw: string | null): [number, number] | null {
   809→  if (!raw) return null;
   810→  const parts = raw.split(":");
   811→  const s = Number.parseInt(parts[0] ?? "", 10);
   812→  const e = Number.parseInt(parts[1] ?? "", 10);
   813→  if (!Number.isFinite(s) || !Number.isFinite(e) || e <= s) return null;
   814→  return [s, e];
   815→}
