    42→func (s *Service) ListProjects(owner string) ([]Project, error) {
    43→        rows, err := s.db.Query(`SELECT p.uid, p.name, p.archived, p.version, p.updated_at,
    44→                (SELECT COUNT(*) FROM clips c WHERE c.project_uid=p.uid AND c.deleted_at IS NULL)
    45→                FROM projects p WHERE p.owner_user_id=? AND p.deleted_at IS NULL ORDER BY p.updated_at DESC`, owner)
    46→        if err != nil {
    47→                return nil, err
    48→        }
    49→        defer rows.Close()
    50→        var out []Project
    51→        for rows.Next() {
    52→                var p Project
    53→                var arch int
    54→                if err := rows.Scan(&p.UID, &p.Name, &arch, &p.Version, &p.UpdatedAt, &p.ClipCount); err != nil {
    55→                        continue
    56→                }
    57→                p.Archived = arch == 1
    58→                out = append(out, p)
    59→        }
    60→        return out, nil
    61→}

    63→func (s *Service) CreateProject(owner, name string) (*Project, error) {
    64→        if name == "" {
    65→                return nil, errors.New("نام پرونده الزامی است")
    66→        }
    67→        uid := uidOf("prj")
    68→        _, err := s.db.Exec(`INSERT INTO projects(uid, owner_user_id, name) VALUES(?,?,?)`, uid, owner, name)
    69→        if err != nil {
    70→                return nil, err
    71→        }
    72→        return &Project{UID: uid, Name: name, Version: 1, UpdatedAt: time.Now().UTC().Format(time.RFC3339)}, nil

    75→func (s *Service) UpdateProject(owner, uid string, name *string, archived *bool, expectedVersion int) (*Project, error) {
    76→        p, err := s.getProject(owner, uid)
    77→        if err != nil {
    78→                return nil, err
    79→        }
    80→        if p.Version != expectedVersion {
    81→                return nil, &ErrConflict{CurrentVersion: p.Version}
    82→        }
    83→        if name != nil {
    84→                p.Name = *name
    85→        }
    86→        if archived != nil {
    87→                p.Archived = *archived
    88→        }
    89→        _, err = s.db.Exec(`UPDATE projects SET name=?, archived=?, version=version+1, updated_at=datetime('now') WHERE uid=? AND owner_user_id=? AND version=?`,
    90→                p.Name, boolInt(p.Archived), uid, owner, expectedVersion)
    91→        if err != nil {
    92→                return nil, err
    93→        }
    94→        return s.getProject(owner, uid)
    95→}

    97→func (s *Service) DeleteProject(owner, uid string) error {
    98→        if _, err := s.getProject(owner, uid); err != nil {
    99→                return err
   100→        }
   101→        // فیش‌های پروژه به سطل می‌روند (حذف نرم — RES-09)
   102→        _, err := s.db.Exec(`UPDATE clips SET deleted_at=datetime('now') WHERE project_uid=? AND deleted_at IS NULL`, uid)
   103→        if err != nil {
   104→                return err
   105→        }
   106→        _, err = s.db.Exec(`UPDATE projects SET deleted_at=datetime('now') WHERE uid=? AND owner_user_id=?`, uid, owner)
   107→        return err
   108→}

   167→/*
   168→normTags قرارداد «tags همیشه آرایه» را تضمین می‌کند: ردیف‌های قدیمی با
   169→tags='null' (خروجی json.Marshal روی اسلایس nil) و ورودیِ کلاینت بدون tags
   170→نباید در پاسخ به null سری شوند — قرارداد API tags: string[] است و فرانت
   171→روی null می‌شکند («c.tags is not iterable»).
   172→*/
   173→func normTags(t *[]string) {
   174→        if *t == nil {
   175→                *t = []string{}
   176→        }
   177→}

   179→// CreateClip داخل transaction: اول مالکیت project، بعد فیش (§۳۹).
   180→func (s *Service) CreateClip(owner string, in *ClipInput) (*Clip, error) {
   181→        if in.QuoteOriginal == "" {
   182→                return nil, errors.New("متن نقل‌قول خالی است")
   183→        }
   184→        if in.Kind != "original" && in.Kind != "as_read" && in.Kind != "note" {
   185→                in.Kind = "original"
   186→        }
   187→        tx, err := s.db.Begin()
   188→        if err != nil {
   189→                return nil, err
   190→        }
   191→        defer tx.Rollback()
   192→        var cnt int
   193→        if err := tx.QueryRow(`SELECT COUNT(*) FROM projects WHERE uid=? AND owner_user_id=? AND deleted_at IS NULL`, in.ProjectUID, owner).Scan(&cnt); err != nil || cnt == 0 {
   194→                return nil, ErrNotFound
   195→        }
   196→        uid := uidOf("clip")
   197→        normTags(&in.Tags) /* ردیف 'null' در DB ذخیره نشود */
   198→        tags, _ := json.Marshal(in.Tags)
   199→        if _, err := tx.Exec(`INSERT INTO clips(uid, owner_user_id, project_uid, kind, quote_original, quote_as_read, prefix_context, suffix_context, note, tags,
   200→                work_uid, document_uid, page_uid, page_label, section_uid, source_offsets_start, source_offsets_end, offset_unit,
   201→                source_version_uid, release_uid, overlay_revision_uid, local_hadith_uid)
   202→                VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
   203→                uid, owner, in.ProjectUID, in.Kind, in.QuoteOriginal, in.QuoteAsRead, in.PrefixContext, in.SuffixContext, in.Note, string(tags),
   204→                in.Source.WorkUID, in.Source.DocumentUID, in.Source.PageUID, in.Source.PageLabel, in.Source.SectionUID,
   205→                in.Source.Start, in.Source.End, orStr(in.Source.OffsetUnit, "codepoint"),
   206→                in.Source.SourceVersion, in.Source.ReleaseUID, in.Source.OverlayRev, in.Source.LocalHadith); err != nil {
   207→                return nil, err
   208→        }
   209→        if err := tx.Commit(); err != nil {
   210→                return nil, err
   211→        }
   212→        return s.GetClip(owner, uid)
   213→}

   215→func (s *Service) GetClip(owner, uid string) (*Clip, error) {
   216→        row := s.db.QueryRow(`SELECT uid, project_uid, kind, quote_original, quote_as_read, prefix_context, suffix_context, note, tags, version,
   217→                work_uid, document_uid, page_uid, page_label, section_uid, source_offsets_start, source_offsets_end, offset_unit,
   218→                source_version_uid, release_uid, overlay_revision_uid, local_hadith_uid, created_at, updated_at
   219→                FROM clips WHERE uid=? AND owner_user_id=? AND deleted_at IS NULL`, uid, owner)
   220→        var c Clip
   221→        var tags string
   222→        var src ClipSource
   223→        if err := row.Scan(&c.UID, &c.ProjectUID, &c.Kind, &c.QuoteOriginal, &c.QuoteAsRead, &c.PrefixContext, &c.SuffixContext, &c.Note, &tags, &c.Version,
   224→                &src.WorkUID, &src.DocumentUID, &src.PageUID, &src.PageLabel, &src.SectionUID, &src.Start, &src.End, &src.OffsetUnit,
   225→                &src.SourceVersion, &src.ReleaseUID, &src.OverlayRev, &src.LocalHadith, &c.CreatedAt, &c.UpdatedAt); err != nil {
   226→                return nil, ErrNotFound
   227→        }
   228→        c.Source = src
   229→        _ = json.Unmarshal([]byte(tags), &c.Tags)
   230→        normTags(&c.Tags) /* ردیف قدیمی با 'null' همیشه به [] برگردد */
   231→        return &c, nil
   232→}

   234→func (s *Service) ListClips(owner, projectUID, state, q, tag string) ([]Clip, error) {
   235→        where := `c.owner_user_id=?`
   236→        args := []any{owner}
   237→        if projectUID != "" {
   238→                where += ` AND c.project_uid=?`
   239→                args = append(args, projectUID)
   240→        }
   241→        if state == "trashed" {
   242→                where += ` AND c.deleted_at IS NOT NULL`
   243→        } else {
   244→                where += ` AND c.deleted_at IS NULL`
   245→        }
   246→        if q != "" {
   247→                where += ` AND (c.quote_original LIKE '%'||?||'%' OR c.note LIKE '%'||?||'%')`
   248→                args = append(args, q, q)
   249→        }
   250→        if tag != "" {
   251→                where += ` AND c.tags LIKE '%'||?||'%'`
   252→                args = append(args, tag)
   253→        }
   254→        rows, err := s.db.Query(`SELECT uid, project_uid, kind, quote_original, quote_as_read, prefix_context, suffix_context, note, tags, version,
   255→                work_uid, document_uid, page_uid, page_label, section_uid, source_offsets_start, source_offsets_end, offset_unit,
   256→                source_version_uid, release_uid, overlay_revision_uid, local_hadith_uid, created_at, updated_at
   257→                FROM clips c WHERE `+where+` ORDER BY c.updated_at DESC LIMIT 200`, args...)
   258→        if err != nil {
   259→                return nil, err
   260→        }
   261→        defer rows.Close()
   262→        var out []Clip
   263→        for rows.Next() {
   264→                var c Clip
   265→                var tags string
   266→                var src ClipSource
   267→                if err := rows.Scan(&c.UID, &c.ProjectUID, &c.Kind, &c.QuoteOriginal, &c.QuoteAsRead, &c.PrefixContext, &c.SuffixContext, &c.Note, &tags, &c.Version,
   268→                        &src.WorkUID, &src.DocumentUID, &src.PageUID, &src.PageLabel, &src.SectionUID, &src.Start, &src.End, &src.OffsetUnit,
   269→                        &src.SourceVersion, &src.ReleaseUID, &src.OverlayRev, &src.LocalHadith, &c.CreatedAt, &c.UpdatedAt); err != nil {
   270→                        continue
   271→                }
   272→                c.Source = src
   273→                _ = json.Unmarshal([]byte(tags), &c.Tags)
   274→                normTags(&c.Tags)
   275→                out = append(out, c)
   276→        }
   277→        /* §۳۹: پایان اسکن هم باید صریح کنترل شود — خطای اسکن خاموش نمی‌ماند */
   278→        if err := rows.Err(); err != nil {
   279→                return nil, err
   280→        }
   281→        return out, nil
   282→}

   369→func (s *Service) ListMyClips(owner, kind, q, tag string) ([]MyClip, int, error) {
   370→        where := `c.owner_user_id=? AND c.deleted_at IS NULL AND p.deleted_at IS NULL`
   371→        args := []any{owner}
   372→        switch kind {
   373→        case "original", "as_read", "note":
   374→                where += ` AND c.kind=?`
   375→                args = append(args, kind)
   376→        case "":
   377→                /* بدون پالایش نوع */
   378→        default:
   379→                return nil, 0, errors.New("نوع فیش نامعتبر است")
   380→        }
   381→        if q != "" {
   382→                where += ` AND (c.quote_original LIKE '%'||?||'%' OR c.note LIKE '%'||?||'%')`
   383→                args = append(args, q, q)
   384→        }
   385→        if tag != "" {
   386→                where += ` AND c.tags LIKE '%'||?||'%'`
   387→                args = append(args, tag)
   388→        }
   389→        var total int
   390→        if err := s.db.QueryRow(`SELECT COUNT(*) FROM clips c JOIN projects p ON p.uid=c.project_uid WHERE `+where, args...).Scan(&total); err != nil {
   391→                return nil, 0, err
   392→        }
   393→        rows, err := s.db.Query(`SELECT c.uid, c.project_uid, p.name, c.kind, c.quote_original, c.note, c.tags,
   394→                c.work_uid, c.document_uid, c.page_uid, c.page_label, c.updated_at
   395→                FROM clips c JOIN projects p ON p.uid = c.project_uid
   396→                WHERE `+where+` ORDER BY c.updated_at DESC LIMIT 100`, args...)
   397→        if err != nil {
   398→                return nil, 0, err
   399→        }
   400→        defer rows.Close()
   401→        var out []MyClip
   402→        for rows.Next() {
   403→                var m MyClip
   404→                var tags string
   405→                if err := rows.Scan(&m.UID, &m.ProjectUID, &m.ProjectName, &m.Kind, &m.QuoteOriginal, &m.Note, &tags,
   406→                        &m.WorkUID, &m.DocumentUID, &m.PageUID, &m.PageLabel, &m.UpdatedAt); err != nil {
   407→                        return nil, 0, err
   408→                }
   409→                _ = json.Unmarshal([]byte(tags), &m.Tags)
   410→                normTags(&m.Tags)
   411→                out = append(out, m)
   412→        }
   413→        if err := rows.Err(); err != nil {
   414→                return nil, 0, err
   415→        }
   416→        return out, total, nil
   417→}

   432→
   433→        تجزیهٔ JSON در Go انجام می‌شود نه در SQL: برنامه‌ریز کوئری SQLite
   434→        جدول‌مجازی json_each را پیش از پالایش WHERE ارزیابی می‌کند و ردیف
   435→        خراب (مثلاً tags='null' قدیمی) کل کوئری را با «malformed JSON» می‌شکند.
   436→        ردیف خراب = بدون برچسب، نه خطا (صداقت داده: مقادیر نامعتبر نادیده
   437→        گرفته می‌شوند، پنهان نمی‌شوند). شمار صادقانه از کل فیش‌های فعال با
   438→        COUNT در Go؛ سقف ۵۰ برچسب پرکاربردتر — نمای پالایش است، نه آرشیو.
   439→*/
   440→func (s *Service) ListMyClipTags(owner string) ([]ClipTag, error) {
   441→        rows, err := s.db.Query(`SELECT c.tags FROM clips c
   442→                JOIN projects p ON p.uid = c.project_uid
   443→                WHERE c.owner_user_id=? AND c.deleted_at IS NULL AND p.deleted_at IS NULL`, owner)
   444→        if err != nil {
   445→                return nil, err
   446→        }
   447→        defer rows.Close()
   448→        counts := map[string]int{}
   449→        for rows.Next() {
   450→                var tags string
   451→                if err := rows.Scan(&tags); err != nil {
   452→                        return nil, err
   453→                }
   454→                var arr []string
   455→                if json.Unmarshal([]byte(tags), &arr) != nil {
   456→                        continue // مقدار نامعتبر برچسبی ندارد
   457→                }
   458→                for _, t := range arr {
   459→                        if t = strings.TrimSpace(t); t != "" {
   460→                                counts[t]++
   461→                        }
   462→                }
   463→        }
   464→        if err := rows.Err(); err != nil {
   465→                return nil, err
   466→        }
   467→        out := make([]ClipTag, 0, len(counts))
   468→        for tag, n := range counts {
   469→                out = append(out, ClipTag{Tag: tag, Count: n})
   470→        }
   471→        sort.Slice(out, func(i, j int) bool {
   472→                if out[i].Count != out[j].Count {
   473→                        return out[i].Count > out[j].Count
   474→                }
   475→                return out[i].Tag < out[j].Tag
   476→        })
   477→        if len(out) > 50 {
   478→                out = out[:50]
   479→        }
   480→        return out, nil
   481→}

   512→// UpdateClip تعارض: شناسه + مالک + نسخه هر سه باید تطابق کنند (§۳۹).
   513→func (s *Service) UpdateClip(owner, uid string, expectedVersion int, note, quoteAsRead *string, tags *[]string, projectUID *string) (*Clip, error) {
   514→        c, err := s.GetClip(owner, uid)
   515→        if err != nil {
   516→                return nil, err
   517→        }
   518→        if c.Version != expectedVersion {
   519→                return nil, &ErrConflict{CurrentVersion: c.Version}
   520→        }
   521→        tx, err := s.db.Begin()
   522→        if err != nil {
   523→                return nil, err
   524→        }
   525→        defer tx.Rollback()
   526→        // نسخهٔ قبلی در تاریخچه
   527→        hist, _ := json.Marshal(c)
   528→        if _, err := tx.Exec(`INSERT INTO clip_revisions(clip_uid, owner_user_id, body, version) VALUES(?,?,?,?)`, uid, owner, string(hist), c.Version); err != nil {
   529→                return nil, err
   530→        }
   531→        if note != nil {
   532→                c.Note = note
   533→        }
   534→        if quoteAsRead != nil {
   535→                c.QuoteAsRead = quoteAsRead
   536→        }
   537→        if tags != nil {
   538→                b, _ := json.Marshal(*tags)
   539→                if _, err := tx.Exec(`UPDATE clips SET tags=? WHERE uid=?`, string(b), uid); err != nil {
   540→                        return nil, err
   541→                }
   542→        }
   543→        if projectUID != nil {
   544→                var cnt int
   545→                if err := tx.QueryRow(`SELECT COUNT(*) FROM projects WHERE uid=? AND owner_user_id=?`, *projectUID, owner).Scan(&cnt); err != nil || cnt == 0 {
   546→                        return nil, ErrNotFound
   547→                }
   548→                if _, err := tx.Exec(`UPDATE clips SET project_uid=? WHERE uid=?`, *projectUID, uid); err != nil {
   549→                        return nil, err
   550→                }
   551→        }
   552→        if _, err := tx.Exec(`UPDATE clips SET note=?, quote_as_read=?, version=version+1, updated_at=datetime('now') WHERE uid=? AND owner_user_id=? AND version=?`,
   553→                c.Note, c.QuoteAsRead, uid, owner, expectedVersion); err != nil {
   554→                return nil, err
   555→        }
   556→        if err := tx.Commit(); err != nil {
   557→                return nil, err
   558→        }
   559→        return s.GetClip(owner, uid)
   560→}

   859→func randHex(n int) string {
   860→        const chars = "0123456789abcdef"
   861→        b := make([]byte, n)
   862→        _, _ = cryptoRead(b)
   863→        for i := range b {
   864→                b[i] = chars[int(b[i])%len(chars)]
   865→        }
   866→        return string(b)
   867→}
