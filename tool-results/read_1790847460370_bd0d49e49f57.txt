   185→   185→// dbHandle اتصال فقط‌خواندنی نمایه؛ با تعویض فایل (نسل جدید) خودکار بازگشایی می‌شود.
   186→   186→// تشخیص تعویض: inode + اندازه؛ و اگر release_uid اتصال با نسخهٔ درخواستی نخواند، بازگشایی اجباری.
   187→   187→func (e *LocalEngine) dbHandle() (*sql.DB, error) {
   188→   188→        e.dbMu.Lock()
   189→   189→        defer e.dbMu.Unlock()
   190→   190→        fi, err := osStat(e.path)
   191→   191→        if err == nil {
   192→   192→                ino := fileInode(fi)
   193→   193→                sz := fi.Size()
   194→   194→                if e.db != nil && (ino != e.dbInode || sz != e.dbSize) {
   195→   195→                        e.db.Close()
   196→   196→                        e.db = nil
   197→   197→                }
   198→   198→                if e.db == nil {
   199→   199→                        db, err := sql.Open("sqlite", "file:"+e.path+"?mode=ro")
   200→   200→                        if err != nil {
   201→   201→                                return nil, err
   202→   202→                        }
   203→   203→                        db.SetMaxOpenConns(3)
   204→   204→                        e.db = db
   205→   205→                        e.dbInode = ino
   206→   206→                        e.dbSize = sz
   207→   207→                }
   208→   208→                return e.db, nil
   209→   209→        }
   210→   210→        if e.db != nil {
   211→   211→                return e.db, nil
   212→   212→        }
   213→   213→        return nil, errors.New("نمایهٔ جست‌وجو در دسترس نیست")
   214→   214→}

   216→   216→// ensureRelease اگر اتصالِ فعلی نسلِ درخواستی را ندارد، اتصال را نوسازی می‌کند
   217→   217→// (مقاوم در برابر inode reuse بین نسل‌ها).
   218→   218→func (e *LocalEngine) ensureRelease(db *sql.DB, releaseUID string) (*sql.DB, error) {
   219→   219→        var rel string
   220→   220→        err := db.QueryRow(`SELECT release_uid FROM passages LIMIT 1`).Scan(&rel)
   221→   221→        if err == nil && rel == releaseUID {
   222→   222→                return db, nil
   223→   223→        }
   224→   224→        // نوسازی اجباری
   225→   225→        e.dbMu.Lock()
   226→   226→        if e.db != nil {
   227→   227→                e.db.Close()
   228→   228→                e.db = nil
   229→   229→        }
   230→   230→        e.dbMu.Unlock()
   231→   231→        return e.dbHandle()
   232→   232→}

   234→   234→func (e *LocalEngine) validate(req *Request) error {
   235→   235→        // جست‌وجوی صرفاً استنادی (بدون عبارت) مجاز است — QUR-03
   236→   236→        if len([]rune(req.Query)) == 0 && req.Scope.Quran == nil {
   237→   237→                return errors.New("عبارت جست‌وجو خالی است")
   238→   238→        }
   239→   239→        modeOK := false
   240→   240→        for _, m := range []string{StrictPhrase, NormalizedPhrase, AllWords, AnyWords, Near, Linguistic, LongQuote} {
   241→   241→                if req.Mode == m {
   242→   242→                        modeOK = true
   243→   243→                }
   244→   244→        }
   245→   245→        if !modeOK {
   246→   246→                return fmt.Errorf("حالت جست‌وجو معتبر نیست")
   247→   247→        }
   248→   248→        limit := 2048
   249→   249→        if req.Mode == LongQuote {
   250→   250→                limit = 8192
   251→   251→        }
   252→   252→        if len([]rune(req.Query)) > limit {
   253→   253→                return errors.New("عبارت بیش از حد مجاز است؛ برای نقل‌قول بلند از حالت مخصوص آن استفاده کنید")
   254→   254→        }
   255→   255→        if req.PageSize < 5 || req.PageSize > 50 {
   256→   256→                req.PageSize = 20
   257→   257→        }
   258→   258→        return nil
   259→   259→}

   261→   261→// Search اجرای جست‌وجو: کاندید → verifier → نتایج تأییدشده.
   262→   262→func (e *LocalEngine) Search(ctx context.Context, req *Request, releaseUID string) (*Response, error) {
   263→   263→        if err := e.validate(req); err != nil {
   264→   264→                return nil, err
   265→   265→        }
   266→   266→        if req.Scope.Target == "titles" {
   267→   267→                hits, err := e.SearchTitles(ctx, req.Query, req.PageSize)
   268→   268→                if err != nil {
   269→   269→                        return nil, err
   270→   270→                }
   271→   271→                resp := &Response{ReleaseUID: releaseUID, CompletionState: "complete", CountRelation: "exact", Results: []Result{}}
   272→   272→                for _, h := range hits {
   273→   273→                        resp.Results = append(resp.Results, Result{
   274→   274→                                PassageUID: "title-" + h.WorkUID, MatchType: "exact",
   275→   275→                                Work: Ref{UID: h.WorkUID, Title: h.Title, Label: h.ShortTitle},
   276→   276→                                Page: Ref{Label: "نام کتاب"},
   277→   277→                        })
   278→   278→                }
   279→   279→                resp.VerifiedBooks = len(hits)
   280→   280→                resp.VerifiedOccurrences = len(hits)
   281→   281→                return resp, nil
   282→   282→        }
   283→   283→        db, err := e.dbHandle()
   284→   284→        if err != nil {
   285→   285→                return nil, err
   286→   286→        }
   287→   287→        db, err = e.ensureRelease(db, releaseUID)
   288→   288→        if err != nil {
   289→   289→                return nil, err
   290→   290→        }
   291→   291→        verified, complete, err := e.run(ctx, db, req, releaseUID)
   292→   292→        if err != nil {
   293→   293→                return nil, err
   294→   294→        }
   295→   295→        page := verified
   296→   296→        if len(page) > req.PageSize {
   297→   297→                page = page[:req.PageSize]
   298→   298→        }
   299→   299→        sid := newID("s")
   300→   300→        sess := &searchSession{req: req, release: releaseUID, offset: len(page), verified: verified, complete: complete, expire: time.Now().Add(sessionTTL)}
   301→   301→        e.mu.Lock()
   302→   302→        if len(e.sessions) > 400 {
   303→   303→                e.sessions = map[string]*searchSession{}
   304→   304→        }
   305→   305→        e.sessions[sid] = sess
   306→   306→        e.mu.Unlock()
   307→   307→        return e.buildResponse(sid, sess, page), nil
   308→   308→}

   310→   310→// Continue ادامهٔ بررسی با cursor امضاشده (§۴۷).
   311→   311→func (e *LocalEngine) Continue(ctx context.Context, sessionUID, cursor, releaseUID string) (*Response, error) {
   312→   312→        e.mu.Lock()
   313→   313→        sess, ok := e.sessions[sessionUID]
   314→   314→        e.mu.Unlock()
   315→   315→        if !ok || time.Now().After(sess.expire) {
   316→   316→                if ok {
   317→   317→                        e.mu.Lock()
   318→   318→                        delete(e.sessions, sessionUID)
   319→   319→                        e.mu.Unlock()
   320→   320→                }
   321→   321→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   322→   322→        }
   323→   323→        if !verifyCursor(cursor, e.Secret) {
   324→   324→                return nil, &GoneError{Message: "نشانهٔ ادامهٔ جست‌وجو معتبر نیست؛ دوباره اجرا کنید."}
   325→   325→        }
   326→   326→        sess.expire = time.Now().Add(sessionTTL)
   327→   327→        remaining := sess.verified[sess.offset:]
   328→   328→        page := remaining
   329→   329→        if len(page) > sess.req.PageSize {
   330→   330→                page = page[:sess.req.PageSize]
   331→   331→        }
   332→   332→        sess.offset += len(page)
   333→   333→        return e.buildResponse(sessionUID, sess, page), nil
   334→   334→}

   336→   336→func (e *LocalEngine) buildResponse(sid string, sess *searchSession, page []Result) *Response {
   337→   337→        books := map[string]bool{}
   338→   338→        passages := map[string]bool{}
   339→   339→        hadiths := map[string]bool{}
   340→   340→        for _, r := range sess.verified {
   341→   341→                if w, _, ok := e.catalog.DocumentMeta(r.Document.UID); ok {
   342→   342→                        books[w] = true
   343→   343→                } else {
   344→   344→                        books[r.Work.UID] = true
   345→   345→                }
   346→   346→                if r.PassageUID != "" {
   347→   347→                        passages[r.PassageUID] = true
   348→   348→                }
   349→   349→                if r.LocalHadith != "" {
   350→   350→                        // روایت درون هر سند متمایز است؛ شناسهٔ محلی جهانی نیست
   351→   351→                        hadiths[r.Document.UID+"/"+r.LocalHadith] = true
   352→   352→                }
   353→   353→        }
   354→   354→        continuation := sess.offset < len(sess.verified)
   355→   355→        resp := &Response{
   356→   356→                SessionUID:            sid,
   357→   357→                ReleaseUID:            sess.release,
   358→   358→                CompletionState:       "complete",
   359→   359→                VerifiedOccurrences:   len(sess.verified),
   360→   360→                VerifiedBooks:         len(books),
   361→   361→                VerifiedPassages:      len(passages),
   362→   362→                VerifiedHadiths:       len(hadiths),
   363→   363→                CountRelation:         "exact",
   364→   364→                ContinuationAvailable: continuation,
   365→   365→                Results:               page,
   366→   366→        }
   367→   367→        if continuation {
   368→   368→                resp.NextCursor = signCursor(map[string]any{"s": sid, "o": sess.offset}, e.Secret)
   369→   369→        }
   370→   370→        return resp
   371→   371→}

   373→   373→// BooksGrouped «این عبارت در چه کتاب‌هایی آمده؟» — فقط تأییدشده‌ها (§۴۷).
   374→   374→func (e *LocalEngine) BooksGrouped(ctx context.Context, sessionUID string) ([]BookGroup, error) {
   375→   375→        e.mu.Lock()
   376→   376→        sess, ok := e.sessions[sessionUID]
   377→   377→        e.mu.Unlock()
   378→   378→        if !ok || time.Now().After(sess.expire) {
   379→   379→                return nil, &GoneError{Message: "مهلت این جست‌وجو پایان یافته است؛ دوباره اجرا کنید."}
   380→   380→        }
   381→   381→        sess.expire = time.Now().Add(sessionTTL)
   382→   382→        workOrder := []string{}
   383→   383→        groups := map[string]*BookGroup{}
   384→   384→        docOrder := map[string][]string{}
   385→   385→        docs := map[string]*DocEvidence{}
   386→   386→        for _, r := range sess.verified {
   387→   387→                bg, ok2 := groups[r.Work.UID]
   388→   388→                if !ok2 {
   389→   389→                        bg = &BookGroup{Work: r.Work, Documents: []DocEvidence{}}
   390→   390→                        groups[r.Work.UID] = bg
   391→   391→                        workOrder = append(workOrder, r.Work.UID)
   392→   392→                }
   393→   393→                key := r.Work.UID + "|" + r.Document.UID
   394→   394→                dg, ok3 := docs[key]
   395→   395→                if !ok3 {
   396→   396→                        dg = &DocEvidence{Document: r.Document, Evidence: []Result{}}
   397→   397→                        docs[key] = dg
   398→   398→                        docOrder[r.Work.UID] = append(docOrder[r.Work.UID], r.Document.UID)
   399→   399→                }
   400→   400→                dg.VerifiedOccurrences++
   401→   401→                if len(dg.Evidence) < 3 {
   402→   402→                        dg.Evidence = append(dg.Evidence, r)
   403→   403→                }
   404→   404→
   405→   405→        }
   406→   406→        var out []BookGroup
   407→   407→        for _, wUID := range workOrder {
   408→   408→                bg := groups[wUID]
   409→   409→                bg.Documents = []DocEvidence{}
   410→   410→                for _, dUID := range docOrder[wUID] {
   411→   411→                        bg.Documents = append(bg.Documents, *docs[wUID+"|"+dUID])
   412→   412→                }
   413→   413→                out = append(out, *bg)
   414→   414→        }
   415→   415→        if out == nil {
   416→   416→                out = []BookGroup{}
   417→   417→        }
   418→   418→        return out, nil
   419→   419→}

   421→   421→// SearchTitles جست‌وجوی نام کتاب.
   422→   422→func (e *LocalEngine) SearchTitles(ctx context.Context, q string, pageSize int) ([]TitleHit, error) {
   423→   423→        db, err := e.dbHandle()
   424→   424→        if err != nil {
   425→   425→                return nil, err
   426→   426→        }
   427→   427→        nf := norm.Fold(q)
   428→   428→        rows, err := db.Query(`SELECT work_uid, title, short_title, author FROM titles ORDER BY work_uid`)
   429→   429→        if err != nil {
   430→   430→                return nil, err
   431→   431→        }
   432→   432→        defer rows.Close()
   433→   433→        var out []TitleHit
   434→   434→        for rows.Next() {
   435→   435→                var uid, title, short, author string
   436→   436→                if err := rows.Scan(&uid, &title, &short, &author); err != nil {
   437→   437→                        continue
   438→   438→                }
   439→   439→                if nf == "" || strings.Contains(norm.Fold(title), nf) || strings.Contains(norm.Fold(short), nf) {
   440→   440→                        out = append(out, TitleHit{WorkUID: uid, Title: title, ShortTitle: short, Author: author})
   441→   441→                }
   442→   442→                if len(out) >= pageSize {
   443→   443→                        break
   444→   444→                }
   445→   445→        }
   446→   446→        return out, nil
   447→   447→}

   489→   489→func (e *LocalEngine) run(ctx context.Context, db *sql.DB, req *Request, releaseUID string) ([]Result, bool, error) {
   490→   490→        qN, _ := norm.Normalize(req.Query)
   491→   491→        toks := norm.Tokenize(qN)
   492→   492→        citationOnly := req.Scope.Quran != nil && len(toks) == 0
   493→   493→        if len(toks) == 0 && !citationOnly {
   494→   494→                return nil, true, nil
   495→   495→        }
   496→   496→        var rows *sql.Rows
   497→   497→        var err error
   498→   498→        if citationOnly {
   499→   499→                // جست‌وجوی استناد قرآنی (QUR-03): همهٔ passages دامنه، فیلتر در verifier
   500→   500→                rows, err = db.Query(`SELECT `+candCols+` FROM passages WHERE release_uid=? ORDER BY work_uid, document_uid, block_start`, releaseUID)
   501→   501→        } else if req.Mode == StrictPhrase {
   502→   502→                // عبارت دقیق: اسکن متنی scope با مرز واژه روی متن اصلی (بدون stemming/fuzzy)
   503→   503→                rows, err = db.Query(`SELECT `+candCols+` FROM passages WHERE release_uid=? ORDER BY work_uid, document_uid, block_start`, releaseUID)
   504→   504→        } else if req.Mode == AnyWords || req.Mode == Linguistic {
   505→   505→                ph := placeholders(len(toks))
   506→   506→                rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   507→   507→                        SELECT DISTINCT p.passage_uid FROM postings p JOIN tokens t ON t.token_id=p.token_id WHERE t.token IN (`+ph+`) LIMIT 20000)
   508→   508→                        ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, toStrs(toks)...)...)
   509→   509→        } else {
   510→   510→                // پیش‌فیلتر «همهٔ واژه‌ها باید در passage باشند»: شمار مقایسه‌ای =
   511→   511→                // شمار واژه‌های متمایزِ پرس‌وجو (نه توکن‌ها با تکرار) — وگرنه هر عبارت
   512→   512→                // دارای واژهٔ تکراری (مثل «الرسول ما الرسول») بی‌کاندید می‌ماند.
   513→   513→                uniq := dedupStrings(toks)
   514→   514→                ph := placeholders(len(uniq))
   515→   515→                rows, err = db.Query(`SELECT s.`+strings.ReplaceAll(candCols, ",", ",s.")+` FROM passages s WHERE s.release_uid=? AND s.passage_uid IN (
   516→   516→                        SELECT p.passage_uid FROM postings p JOIN tokens t ON t.token_id=p.token_id WHERE t.token IN (`+ph+`)
   517→   517→                        GROUP BY p.passage_uid HAVING COUNT(DISTINCT t.token)=? LIMIT 20000)
   518→   518→                        ORDER BY s.work_uid, s.document_uid, s.block_start`, append([]any{releaseUID}, append(toStrs(uniq), len(uniq))...)...)
   519→   519→        }
   520→   520→        if err != nil {
   521→   521→                return nil, false, err
   522→   522→        }
   523→   523→        defer rows.Close()
   524→   524→        var out []Result
   525→   525→        seen := map[string]bool{}
   526→   526→        rowN := 0
   527→   527→        for rows.Next() {
   528→   528→                rowN++
   529→   529→                if ctx.Err() != nil {
   530→   530→                        return out, false, ctx.Err()
   531→   531→                }
   532→   532→                var r candRow
   533→   533→                if err := scanCand(rows, &r); err != nil {
   534→   534→                        // حذف خاموش ممنوع: خطای اسکن = کاندید نادیده = نتیجهٔ ناقص بی‌اعلام.
   535→   535→                        // طبق فلسفهٔ خطای §۳۹، این وضعیت باید صریحاً خطا باشد.
   536→   536→                        return nil, false, fmt.Errorf("scan candidate: %w", err)
   537→   537→                }
   538→   538→                // فیلتر دامنهٔ اثر/سند/زبان در verifier هم اعمال می‌شود (دفاع در عمق)
   539→   539→                if req.Scope.Target == "work" && (req.Scope.WorkUID == nil || r.WorkUID != *req.Scope.WorkUID) {
   540→   540→                        continue
   541→   541→                }
   542→   542→                if req.Scope.Target == "document" || req.Scope.Target == "section" {
   543→   543→                        inScope := false
   544→   544→                        for _, d := range req.Scope.DocumentUIDs {
   545→   545→                                if d == r.DocumentUID {
   546→   546→                                        inScope = true
   547→   547→                                        break
   548→   548→                                }
   549→   549→                        }
   550→   550→                        if !inScope {
   551→   551→                                continue
   552→   552→                        }
   553→   553→                }
   554→   554→                if req.Scope.Language != nil && *req.Scope.Language != "" && r.Language != *req.Scope.Language {
   555→   555→                        continue
   556→   556→                }
   557→   557→                if req.Scope.Quran != nil {
   558→   558→                        cits, err := parseCitations(r.Citations.String)
   559→   559→                        if err != nil {
   560→   560→                                continue
   561→   561→                        }
   562→   562→                        matched := false
   563→   563→                        for _, c := range cits {
   564→   564→                                if c.Sure == req.Scope.Quran.Sure {
   565→   565→                                        if req.Scope.Quran.Aya == "" {
   566→   566→                                                matched = true
   567→   567→                                                break
   568→   568→                                        }
   569→   569→                                        if c.Aya == req.Scope.Quran.Aya {
   570→   570→                                                matched = true
   571→   571→                                                break
   572→   572→                                        }
   573→   573→                                        if c.AyaFrom != "" && c.AyaTo != "" {
   574→   574→                                                if between(c.AyaFrom, c.AyaTo, req.Scope.Quran.Aya) {
   575→   575→                                                        matched = true
   576→   576→                                                        break
   577→   577→                                                }
   578→   578→                                        }
   579→   579→                                }
   580→   580→                        }
   581→   581→                        if !matched {
   582→   582→                                continue // استناد باید واقعاً در همین passage باشد، نه فقط در سند
   583→   583→                        }
   584→   584→                }
   585→   585→                hits, matchType := verifyInText(req, &r)
   586→   586→                if req.Scope.Quran != nil && len([]rune(req.Query)) == 0 && len(hits) == 0 {
   587→   587→                        // جست‌وجوی صرفاً استنادی: خود passage با شاهد ابتدایی نمایش داده می‌شود
   588→   588→                        n1r := []rune(r.TextOriginal)
   589→   589→                        hi := clampInt(160, 0, len(n1r))
   590→   590→                        hits = []hit{{Range: [2]int{0, hi}}}
   591→   591→                        matchType = "exact"
   592→   592→                }
   593→   593→
   594→   594→                for _, h := range hits {
   595→   595→                        // کلید رخداد = document + محدودهٔ واقعی مطلق در جریان متن سند (§۴۵)
   596→   596→                        key := r.DocumentUID + ":" + fmt.Sprint(r.SourceStart+h.Range[0]) + ":" + fmt.Sprint(r.SourceStart+h.Range[1])
   597→   597→                        if seen[key] {
   598→   598→                                continue // overlap تکراری نتیجه نمی‌سازد؛ رخدادهای واقعی جدا می‌مانند
   599→   599→                        }
   600→   600→                        seen[key] = true
   601→   601→                        if len(req.Scope.Roles) > 0 && !matchInsideRoles(r.RoleRanges, h.Range) {
   602→   602→                                continue // فیلتر نقش: محل واقعی تطبیق باید داخل نقش باشد (§۴۵)
   603→   603→                        }
   604→   604→                        out = append(out, e.toResult(&r, h, matchType))
   605→   605→                }
   606→   606→        }
   607→   607→        return out, true, nil
   608→   608→}

   610→   610→type hit struct {

   614→   614→// verifyInText بسته به mode تطبیق را روی متن واقعی همان نسخه تأیید می‌کند.
   615→   615→func verifyInText(req *Request, r *candRow) ([]hit, string) {
   616→   616→        orig := []rune(r.TextOriginal)
   617→   617→        n1 := []rune(r.TextN1)
   618→   618→        var n1Map []int
   619→   619→        _ = json.Unmarshal([]byte(r.N1Map), &n1Map)
   620→   620→        switch req.Mode {
   621→   621→        case StrictPhrase:
   622→   622→                q := []rune(req.Query)
   623→   623→                var out []hit
   624→   624→                start := 0
   625→   625→                for {
   626→   626→                        idx := indexRunes(orig[start:], q)
   627→   627→                        if idx < 0 {
   628→   628→                                break
   629→   629→                        }
   630→   630→                        s := start + idx
   631→   631→                        e := s + len(q)
   632→   632→                        if norm.WordBoundary(string(orig), s, e) {
   633→   633→                                out = append(out, hit{Range: [2]int{s, e}})
   634→   634→                        }
   635→   635→                        start = s + 1
   636→   636→                        if len(out) >= 50 {
   637→   637→                                break
   638→   638→                        }
   639→   639→                }
   640→   640→                return out, "exact"
   641→   641→        default:
   642→   642→                qN, _ := norm.Normalize(req.Query)
   643→   643→                if req.Mode == Linguistic {
   644→   644→                        qN = norm.Fold(req.Query)
   645→   645→                }
   646→   646→                if req.Mode == AllWords || req.Mode == Near {
   647→   647→                        toks := norm.Tokenize(qN)
   648→   648→                        positions := make([][]int, len(toks))
   649→   649→                        for i, t := range toks {
   650→   650→                                positions[i] = findAllWord(n1, t)
   651→   651→                        }
   652→   652→                        var out []hit
   653→   653→                        if req.Mode == Near && req.Near != nil && req.Near.Slop > 0 {
   654→   654→                                for _, rng := range nearCombination(positions, req.Near.Slop, req.Near.Ordered, len(n1)) {
   655→   655→                                        out = append(out, mapN1Hit(rng, n1Map))
   656→   656→                                }
   657→   657→                                return out, "normalized"
   658→   658→                        }
   659→   659→                        if allPresent(positions) {
   660→   660→                                lo, hi := -1, -1
   661→   661→                                for i, ps := range positions {
   662→   662→                                        if i == 0 || ps[0] < lo {
   663→   663→                                                lo = ps[0]
   664→   664→                                        }
   665→   665→                                        last := ps[len(ps)-1] + len([]rune(toks[i]))
   666→   666→                                        if i == 0 || last > hi {
   667→   667→                                                hi = last
   668→   668→                                        }
   669→   669→                                }
   670→   670→                                out = append(out, mapN1Hit([2]int{lo, hi}, n1Map))
   671→   671→                        }
   672→   672→                        return out, "normalized"
   673→   673→                }
   674→   674→                // normalized_phrase / linguistic / long_quote: تطبیق متوالی با مرز واژه
   675→   675→                qRunes := []rune(qN)
   676→   676→                var out []hit
   677→   677→                start := 0
   678→   678→                for {
   679→   679→                        idx := indexRunes(n1[start:], qRunes)
   680→   680→                        if idx < 0 {
   681→   681→                                break
   682→   682→                        }
   683→   683→                        s := start + idx
   684→   684→                        e := s + len(qRunes)
   685→   685→                        if norm.WordBoundary(string(n1), s, e) {
   686→   686→                                out = append(out, mapN1Hit([2]int{s, e}, n1Map))
   687→   687→                        }
   688→   688→                        start = s + 1
   689→   689→                        if len(out) >= 50 {
   690→   690→                                break
   691→   691→                        }
   692→   692→                }
   693→   693→                if req.Mode == Linguistic {
   694→   694→                        return out, "linguistic"
   695→   695→                }
   696→   696→                return out, "normalized"
   697→   697→        }
   698→   698→}
   699→   699→
   700→   700→// mapN1Hit نگاشت بازهٔ نرمال → بازهٔ متن اصلی (highlight روی اصل — SEARCH-11).
   701→   701→func mapN1Hit(rng [2]int, n1Map []int) hit {
   702→   702→        if len(n1Map) == 0 {
   703→   703→                return hit{Range: rng}
   704→   704→        }
   705→   705→        s := clampInt(rng[0], 0, len(n1Map)-1)
   706→   706→        e := clampInt(rng[1], s, len(n1Map))
   707→   707→        if e > s {
   708→   708→                return hit{Range: [2]int{n1Map[s], n1Map[e-1] + 1}}
   709→   709→        }
   710→   710→        return hit{Range: [2]int{n1Map[s], n1Map[s]}}
   711→   711→}
   712→   712→
   713→   713→func clampInt(v, lo, hi int) int {
   714→   714→        if v < lo {
   715→   715→                return lo
   716→   716→        }
   717→   717→        if v > hi {
   718→   718→                return hi
   719→   719→        }
   720→   720→        return v
   721→   721→}

   723→   723→func findAllWord(s []rune, w string) []int {
   724→   724→        var out []int
   725→   725→        rs := []rune(w)
   726→   726→        start := 0
   727→   727→        for {
   728→   728→                idx := indexRunes(s[start:], rs)
   729→   729→                if idx < 0 {
   730→   730→                        break
   731→   731→                }
   732→   732→                p := start + idx
   733→   733→                if norm.WordBoundary(string(s), p, p+len(rs)) {
   734→   734→                        out = append(out, p)
   735→   735→                }
   736→   736→                start = p + 1
   737→   737→        }
   738→   738→        return out
   739→   739→}

   741→   741→func allPresent(ps [][]int) bool {
   742→   742→        for _, p := range ps {
   743→   743→                if len(p) == 0 {
   744→   744→                        return false
   745→   745→                }
   746→   746→        }
   747→   747→        return true
   748→   748→}

   750→   750→func nearCombination(positions [][]int, slop int, ordered bool, textLen int) [][2]int {
   751→   751→        var out [][2]int
   752→   752→        if len(positions) == 0 || !allPresent(positions) {
   753→   753→                return nil
   754→   754→        }
   755→   755→        idx := make([]int, len(positions))
   756→   756→        for {
   757→   757→                var lo, hi int
   758→   758→                for i, p := range positions {
   759→   759→                        pos := p[idx[i]]
   760→   760→                        if i == 0 || pos < lo {
   761→   761→                                lo = pos
   762→   762→                        }
   763→   763→                        if i == 0 || pos > hi {
   764→   764→                                hi = pos
   765→   765→                        }
   766→   766→                }
   767→   767→                ok := (hi - lo) <= slop*6
   768→   768→                if ok && ordered && len(positions) > 1 {
   769→   769→                        for i := 1; i < len(positions); i++ {
   770→   770→                                if positions[i][idx[i]] < positions[i-1][idx[i-1]] {
   771→   771→                                        ok = false
   772→   772→                                        break
   773→   773→                                }
   774→   774→                        }
   775→   775→                }
   776→   776→                if ok {
   777→   777→                        out = append(out, [2]int{lo, clampInt(hi+len(positions[len(positions)-1]), 0, textLen)})
   778→   778→                }
   779→   779→                i := len(idx) - 1
   780→   780→                for i >= 0 {
   781→   781→                        idx[i]++
   782→   782→                        if idx[i] < len(positions[i]) {
   783→   783→                                break
   784→   784→                        }
   785→   785→                        idx[i] = 0
   786→   786→                        i--
   787→   787→                }
   788→   788→                if i < 0 || len(out) >= 10 {
   789→   789→                        break
   790→   790→                }
   791→   791→        }
   792→   792→        return out
   793→   793→}

   802→   802→func matchInsideRoles(roleRangesJSON string, rng [2]int) bool {

   818→   818→func (e *LocalEngine) toResult(r *candRow, h hit, matchType string) Result {
   819→   819→        work := Ref{UID: r.WorkUID}
   820→   820→        if t, s, _, ok := e.catalog.WorkTitle(r.WorkUID); ok {
   821→   821→                work.Title = t
   822→   822→                work.Label = s
   823→   823→        }
   824→   824→        doc := Ref{UID: r.DocumentUID}
   825→   825→        if _, label, ok := e.catalog.DocumentMeta(r.DocumentUID); ok {
   826→   826→                doc.Label = label
   827→   827→        }
   828→   828→        doc.Title = work.Title
   829→   829→        sec := (*Ref)(nil)
   830→   830→        if r.SectionUID.Valid && r.SectionUID.String != "" {
   831→   831→                title, _ := e.catalog.SectionTitle(r.DocumentUID, r.SectionUID.String)
   832→   832→                sec = &Ref{UID: r.SectionUID.String, Title: title, Label: title}
   833→   833→        }
   834→   834→        var ranges []roleRangeT
   835→   835→        _ = json.Unmarshal([]byte(r.RoleRanges), &ranges)
   836→   836→        roles := []string{}
   837→   837→        for _, g := range ranges {
   838→   838→                if g.Start < h.Range[1] && g.End > h.Range[0] {
   839→   839→                        roles = append(roles, g.Role)
   840→   840→                }
   841→   841→        }
   842→   842→        roles = dedupStrings(roles)
   843→   843→        snippet := []rune(r.TextOriginal)
   844→   844→        lo := clampInt(h.Range[0]-60, 0, len(snippet))

   876→   876→func indexRunes(s []rune, sub []rune) int {

   878→   878→        if n == 0 || n > len(s) {
   879→   879→                return -1

   882→   882→                match := true

   884→   884→                        if s[i+j] != sub[j] {
   885→   885→                                match = false

   887→   887→                        }
   888→   888→                }
   889→   889→                if match {
   890→   890→                        return i
   891→   891→                }
   892→   892→        }
   893→   893→        return -1
   894→   894→}

   964→   964→// between بررسی تعلق آیهٔ پرس‌شده به بازهٔ استناد (اعداد فارسی/عربی/لاتین نرمال می‌شوند).
   965→   965→func between(from, to, aya string) bool {
   966→   966→        f := numNorm(from)
   967→   967→        t := numNorm(to)
   968→   968→        a := numNorm(aya)
   969→   969→        if f == "" || t == "" || a == "" {
   970→   970→                return false
   971→   971→        }
   972→   972→        return a >= f && a <= t
   973→   973→}
   974→   974→
   975→   975→func numNorm(s string) string {
   976→   976→        digits := map[rune]string{
   977→   977→                '٠': "0", '١': "1", '٢': "2", '٣': "3", '٤': "4", '٥': "5", '٦': "6", '٧': "7", '٨': "8", '٩': "9",
   978→   978→                '۰': "0", '۱': "1", '۲': "2", '۳': "3", '۴': "4", '۵': "5", '۶': "6", '۷': "7", '۸': "8", '۹': "9",
   979→   979→        }
   980→   980→        var b []byte
   981→   981→        for _, r := range s {
   982→   982→                if d, ok := digits[r]; ok {
   983→   983→                        b = append(b, d...)
   984→   984→                } else if r >= '0' && r <= '9' {
   985→   985→                        b = append(b, byte(r))
   986→   986→                }
   987→   987→        }
   988→   988→        return string(b)
   989→   989→}
