    66→function clampPage(raw: string | undefined): number {
    67→  const n = Number.parseInt(raw ?? "", 10);
    68→  return Number.isFinite(n) && n >= 1 && n <= 200 ? n : 1;
    69→}

    71→/** بازسازی hash از payload جست‌وجوی ذخیره‌شده (اجرای دوباره با همان نشانی) */
    72→function hashFromPayload(payload: unknown): Record<string, string> | null {
    73→  if (!payload || typeof payload !== "object") return null;
    74→  const p = payload as Record<string, unknown>;
    75→  if (typeof p.query !== "string" || typeof p.mode !== "string") return null;
    76→  const out: Record<string, string> = { q: p.query, mode: p.mode };
    77→  const scope = (p.scope && typeof p.scope === "object" ? p.scope : {}) as Record<string, unknown>;
    78→  const target = typeof scope.target === "string" ? scope.target : "library";
    79→  if (target === "titles") {
    80→    out.scope = "titles";
    81→  } else if (target === "work" && typeof scope.work_uid === "string" && scope.work_uid) {
    82→    out.scope = "work";
    83→    out.work = scope.work_uid;
    84→  } else if (
    85→    target === "document" &&
    86→    Array.isArray(scope.document_uids) &&
    87→    typeof scope.document_uids[0] === "string"
    88→  ) {
    89→    out.scope = "document";
    90→    out.doc = scope.document_uids[0];
    91→    if (typeof scope.work_uid === "string") out.work = scope.work_uid;
    92→  } else {
    93→    out.scope = "library";
    94→  }
    95→  const roles = Array.isArray(scope.roles)
    96→    ? scope.roles.filter((r): r is string => typeof r === "string")
    97→    : [];
    98→  const roleValues = ROLE_FILTERS.filter((f) => f.codes.some((c) => roles.includes(c))).map(
    99→    (f) => f.value,
   100→  );
   101→  if (roleValues.length > 0) out.roles = roleValues.join(",");
   102→  if (typeof scope.language === "string" && scope.language) out.lang = scope.language;
   103→  const near = (p.near && typeof p.near === "object" ? p.near : null) as Record<string, unknown> | null;
   104→  if (near) {
   105→    out.slop = String(typeof near.slop === "number" ? near.slop : 3);
   106→    if (near.ordered === false) out.ordered = "0";
   107→  }
   108→  return out;
   109→}

   314→export function SearchView() {
   315→  const { route, navigate } = useRoute();
   316→
   317→  /* وضعیت اجراشدهٔ جست‌وجو (از hash) */
   318→  const params = parseSearchParams(route.query);
   319→  const searchKey = useMemo(() => searchStateKey(params), [params]);
   320→  const hashPage = clampPage(route.query.page);
   321→  const [saveOpen, setSaveOpen] = useState(false);
   322→  const view: "results" | "books" = route.query.view === "books" ? "books" : "results";
   323→  // عبارت یا استناد قرآنی (QUR-03) هر دو شروع‌کنندهٔ جست‌وجو هستند
   324→  const hasQuery = params.q.trim().length > 0 || params.sure.trim().length > 0;
   325→
   326→  /* پیش‌نویس فرم — هم‌گام‌سازی با hash در زمان رندر (الگوی مجاز React) */
   327→  const [draft, setDraft] = useState<SearchHashState>(() => parseSearchParams(route.query));
   328→  const [draftHash, setDraftHash] = useState(route.fullHash);
   329→  if (draftHash !== route.fullHash) {
   330→    setDraftHash(route.fullHash);
   331→    setDraft(parseSearchParams(route.query));
   332→  }
   333→
   334→  /** نوشتن وضعیت در hash — هر بار کل پارامترها داده می‌شود */
   335→  const setQuery = useCallback(
   336→    (state: SearchHashState, extra: Record<string, string | undefined> = {}, replace = false) => {
   337→      navigate(
   338→        "/search",
   339→        {
   340→          q: state.q.trim() || undefined,
   341→          mode: state.mode,
   342→          scope: state.scope,
   343→          work: state.work || undefined,
   344→          doc: state.doc || undefined,
   345→          roles: state.roles.length > 0 ? state.roles.join(",") : undefined,
   346→          lang: state.lang || undefined,
   347→          slop: state.mode === "near" ? String(state.slop) : undefined,
   348→          ordered: state.mode === "near" && !state.ordered ? "0" : undefined,
   349→          page: undefined,
   350→          view: undefined,
   351→          ...extra,
   352→        },
   353→        { replace },
   354→      );
   355→    },
   356→    [navigate],
   357→  );
   358→
   359→  const submit = useCallback(() => {
   360→    setQuery(draft);
   361→  }, [draft, setQuery]);
   362→
   363→  const patchCommitted = useCallback(
   364→    (patch: Partial<SearchHashState>) => {
   365→      setQuery({ ...params, ...patch });
   366→    },
   367→    [params, setQuery],
   368→  );
   369→
   370→  /* لایهٔ ۱ — نشست جست‌وجو: صفحهٔ نخست + شمارش صادقانه */
   371→  const request = buildSearchRequest(params);
   372→
   373→  const sessionQuery = useQuery({
   374→    queryKey: ["search-session", searchKey],
   375→    queryFn: async ({ signal }) => {
   376→      const resp = await api.post<SearchResponse>("/search", request);
   377→      if (signal.aborted) throw new DOMException("aborted", "AbortError");
   378→      return resp;
   379→    },
   380→    enabled: hasQuery,
   381→    staleTime: 4 * 60_000,
   382→    gcTime: 10 * 60_000,
   383→    retry: false,
   384→    refetchOnWindowFocus: false,
   385→    refetchOnReconnect: false,
   386→  });
   387→
   388→  const session = hasQuery ? sessionQuery.data ?? null : null;
   389→  const sessionUid = session?.session_uid ?? null;
   390→  const sessionUidRef = useRef<string | null>(null);
   391→  useEffect(() => {
   392→    sessionUidRef.current = sessionUid;
   393→  }, [sessionUid]);
   394→
   395→  /* لایهٔ ۲ — صفحه‌های بیشتر با «ادامهٔ بررسی» (cursor امضاشده، §۴۷) */
   396→  const [extras, setExtras] = useState<ExtrasState>(EMPTY_EXTRAS);
   397→  const currentExtras = sessionUid && extras.uid === sessionUid ? extras : EMPTY_EXTRAS;
   398→  const extrasRef = useRef(currentExtras);
   399→  useEffect(() => {
   400→    extrasRef.current = currentExtras;
   401→  }, [currentExtras]);
   402→
   403→  const meta = currentExtras.meta ?? session;
   404→  const items = useMemo(
   405→    () => [...(session?.results ?? []), ...currentExtras.items],
   406→    [session, currentExtras.items],
   407→  );
   408→
   409→  const restoreStampRef = useRef<string>("");
   410→
   411→  const runContinue = useCallback(
   412→    async (targetSessionUid: string, count: number, updateHash: boolean) => {
   413→      let acc: ExtrasState =
   414→        extrasRef.current.uid === targetSessionUid
   415→          ? extrasRef.current
   416→          : { ...EMPTY_EXTRAS, uid: targetSessionUid };
   417→      if (acc.loading) return;
   418→      acc = { ...acc, loading: true, error: null };
   419→      extrasRef.current = acc;
   420→      setExtras(acc);
   421→
   422→      for (let i = 0; i < count; i += 1) {
   423→        if (sessionUidRef.current !== targetSessionUid) return; // جست‌وجوی تازه‌تری اجرا شده است
   424→        const cursorMeta = acc.meta ?? session ?? null;
   425→        if (!cursorMeta?.continuation_available || !cursorMeta.next_cursor) break;
   426→        try {
   427→          const resp = await api.post<SearchResponse>(
   428→            `/search/${targetSessionUid}/continue`,
   429→            { cursor: cursorMeta.next_cursor },
   430→          );
   431→          if (sessionUidRef.current !== targetSessionUid) return;
   432→          acc = {
   433→            ...acc,
   434→            count: acc.count + 1,
   435→            items: [...acc.items, ...(resp.results ?? [])],
   436→            meta: resp,
   437→            loading: i < count - 1,
   438→          };
   439→        } catch (err) {
   440→          if (err instanceof ApiError && err.code === "gone_cursor") {
   441→            acc = { ...acc, expired: true, loading: false };
   442→          } else {
   443→            acc = {
   444→              ...acc,
   445→              error: errorMessage(err, "ادامهٔ بررسی ناموفق بود؛ دوباره تلاش کنید."),
   446→              loading: false,
   447→            };
   448→          }
   449→          break;
   450→        }
   451→        extrasRef.current = acc;
   452→        setExtras(acc);
   453→      }
   454→      acc = { ...acc, loading: false };
   455→      extrasRef.current = acc;
   456→      setExtras(acc);
   457→
   458→      // هم‌گام‌سازی موضع در hash (WORK-07): شمارهٔ صفحه = شمار صفحه‌های اضافه + ۱
   459→      if (updateHash || acc.expired || acc.error) {
   460→        navigate(
   461→          "/search",
   462→          { ...route.query, page: acc.count > 0 ? String(acc.count + 1) : undefined },
   463→          { replace: true },
   464→        );
   465→      }
   466→    },
   467→    [navigate, route.query, session],  
   468→  );
   469→
   470→  /* بازسازی موضع پس از بازگشت به نتایج (WORK-07): زنجیرهٔ continue تا صفحهٔ hash */
   471→  useEffect(() => {
   472→    const targetExtras = hashPage - 1;
   473→    if (!sessionUid || targetExtras <= 0 || currentExtras.loading) return;
   474→    if (currentExtras.count >= targetExtras) return;
   475→    if (meta && !meta.continuation_available) return;
   476→    const stamp = `${searchKey}|${sessionUid}|${targetExtras}`;
   477→    if (restoreStampRef.current === stamp) return;
   478→    restoreStampRef.current = stamp;
   479→    const t = window.setTimeout(() => {
   480→      void runContinue(sessionUid, targetExtras - currentExtras.count, false);
   481→    }, 0);
   482→    return () => window.clearTimeout(t);
   483→  }, [sessionUid, hashPage, currentExtras.count, currentExtras.loading, meta, searchKey, runContinue]);
   484→
   485→  /* وضعیت نمایه — تفکیک «بدون نتیجه» از «نمایه آماده نیست» (SEARCH-21) */
   486→  const isEmpty = sessionQuery.isSuccess && items.length === 0 && !currentExtras.loading;
   487→  const statusQuery = useQuery({
   488→    queryKey: ["library-status"],
   489→    queryFn: ({ signal }) =>
   490→      api.get<{ index_ready?: boolean | null }>("/status", undefined, signal),
   491→    enabled: isEmpty,
   492→    staleTime: 60_000,
   493→  });
   494→
   495→  const loadingMore = currentExtras.loading;
   496→  const switchView = (next: string) => {
   497→    navigate(
   498→      "/search",
   499→      { ...route.query, view: next === "books" ? "books" : undefined },
   500→      { replace: true },
   501→    );
   502→  };
   503→
   504→  return (
   505→    <div className="mx-auto w-full max-w-6xl px-4 py-8">
   506→      <header className="mb-6">
   507→        <h1 className="text-xl font-bold">جست‌وجو در منابع</h1>
   508→        <p className="mt-1 text-sm text-muted-foreground">
   509→          جست‌وجوی عبارت در متن منابع با انتخاب حالت و دامنه؛ فقط مواضع تأییدشده نمایش داده می‌شود.
   510→        </p>
   511→      </header>
   512→
   513→      <div className="grid items-start gap-6 lg:grid-cols-[minmax(0,1fr)_290px]">
   514→        <div className="min-w-0 space-y-5">
   515→          {/* فرم جست‌وجو */}
   516→          <Card>
   517→            <CardContent className="p-4 sm:p-6">
   518→              <SearchForm
   519→                value={draft}
   520→                onChange={(patch) => setDraft((d) => ({ ...d, ...patch }))}
   521→                onSubmit={submit}
   522→                busy={hasQuery && sessionQuery.isPending}
   523→              />
   524→            </CardContent>
   525→          </Card>
   526→
   527→          {/* چیپ فیلترها (SEARCH-20) */}
   528→          {hasQuery ? <FilterChips state={params} onChange={patchCommitted} /> : null}
   529→
   530→          {/* نتایج */}
   531→          {!hasQuery ? (
   532→            <EmptyState
   533→              title="عبارتی برای جست‌وجو وارد نشده است"
   534→              hint="عبارت مورد نظر را در کادر بالا بنویسید و «جست‌وجو» را بزنید."

   537→            <div className="space-y-3" aria-busy="true">
   538→              <ResultCardSkeleton />
   539→              <ResultCardSkeleton />
   540→              <ResultCardSkeleton />
   541→              <span className="sr-only">در حال جست‌وجو در منابع…</span>

   544→            /* وضعیت خالی ۳ — خطای موتور جست‌وجو */

   546→              message={errorMessage(
   547→                sessionQuery.error,
   548→                "جست‌وجو با خطا روبه‌رو شد؛ دوباره تلاش کنید.",
   549→              )}
   550→              onRetry={() => void sessionQuery.refetch()}

   554→              {/* هشدار پایان مهلت نشست (۴۱۰ در ادامهٔ بررسی) */}
   555→              {currentExtras.expired ? (
   556→                <Alert>
   557→                  <TriangleAlert className="size-4" aria-hidden="true" />
   558→                  <AlertTitle>مهلت این جست‌وجو پایان یافته است</AlertTitle>
   559→                  <AlertDescription className="flex flex-wrap items-center gap-3">
   560→                    <span>
   561→                      مواضع بیشتری از این نشست قابل دریافت نیست؛ جست‌وجو را دوباره اجرا کنید.
   562→                    </span>
   563→                    <Button
   564→                      variant="outline"
   565→                      size="sm"
   566→                      className="min-h-9 gap-1.5"
   567→                      onClick={() => void sessionQuery.refetch()}
   568→                    >
   569→                      <RotateCcw className="size-4" aria-hidden="true" />
   570→                      اجرای دوباره
   571→                    </Button>
   572→                  </AlertDescription>
   573→                </Alert>
   574→              ) : null}
   575→              {currentExtras.error ? (
   576→                <Alert variant="destructive">
   577→                  <TriangleAlert className="size-4" aria-hidden="true" />
   578→                  <AlertTitle>ادامهٔ بررسی ناموفق بود</AlertTitle>
   579→                  <AlertDescription>{currentExtras.error}</AlertDescription>
   580→                </Alert>
   581→              ) : null}
   582→
   583→              {/* شمارش صادقانهٔ §۴۷ + دکمهٔ ذخیره (SEARCH-16) */}
   584→              <div className="flex flex-wrap items-center justify-between gap-3">
   585→                <div className="min-w-0 space-y-0.5">
   586→                  <VerifiedCountLine
   587→                    occurrences={meta?.verified_occurrences ?? null}
   588→                    countRelation={meta?.count_relation ?? ""}
   589→                    passages={meta?.verified_passages ?? null}
   590→                    hadiths={meta?.verified_hadiths ?? null}

   594→                      {meta.completion_state === "failed"
   595→                        ? "بررسی این جست‌وجو ناتمام ماند؛ مواضع نمایش‌داده‌شده تأییدشده‌اند."

   597→                    </p>
   598→                  ) : null}
   599→                </div>
   600→                <Button
   601→                  variant="outline"
   602→                  size="sm"
   603→                  className="min-h-9 gap-1.5"
   604→                  onClick={() => setSaveOpen(true)}
   605→                >
   606→                  <BookmarkPlus className="size-4" aria-hidden="true" />
   607→                  ذخیرهٔ این جست‌وجو
   608→                </Button>
   609→              </div>
   610→
   611→              {items.length === 0 ? (
   612→                statusQuery.data?.index_ready === false ? (
   613→                  /* وضعیت خالی ۲ — نمایه آماده نیست */
   614→                  <EmptyState
   615→                    title="نمایهٔ کتابخانه در حال آماده‌سازی است؛ به‌زودی در دسترس خواهد بود"

   619→                  /* وضعیت خالی ۱ — بدون نتیجه */
   620→                  <EmptyState
   621→                    title="موردی یافت نشد"
   622→                    hint="حالت «عبارت نرمال‌شده» را امتحان کنید یا فیلترها را بردارید؛ درست‌نویسی عبارت را هم بررسی کنید."
   623→                    action={
   624→                      params.mode !== "normalized_phrase" ? (
   625→                        <Button
   626→                          variant="outline"
   627→                          className="min-h-11"
   628→                          onClick={() => patchCommitted({ mode: "normalized_phrase" })}

   630→                          جست‌وجو با حالت نرمال‌شده

   633→                    }
   634→                  />
   635→                )
   636→              ) : (
   637→                <Tabs value={view} onValueChange={switchView}>
   638→                  <TabsList aria-label="سطح مشاهدهٔ نتایج" className="w-full sm:w-auto">
   639→                    <TabsTrigger value="results" className="min-h-9">
   640→                      مواضع
   641→                    </TabsTrigger>
   642→                    <TabsTrigger value="books" className="min-h-9">
   643→                      کتاب‌های دارای تطبیق تأییدشده
   644→                    </TabsTrigger>
   645→                  </TabsList>
   646→                  <TabsContent value="results" className="space-y-3">
   647→                    <GroupedPositions items={items} releaseUid={meta?.release_uid ?? null} />
   648→                    {/* صفحه‌بندی مواضع: ادامهٔ بررسی */}
   649→                    {meta?.continuation_available ? (
   650→                      <div className="flex flex-col items-center gap-1 pt-2">
   651→                        <Button
   652→                          variant="outline"
   653→                          className="min-h-11 gap-2 px-6"
   654→                          disabled={loadingMore}
   655→                          onClick={() => {
   656→                            if (sessionUid) void runContinue(sessionUid, 1, true);
   657→                          }}
   658→                        >
   659→                          {loadingMore ? (
   660→                            <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   661→                          ) : (
   662→                            <ChevronDown className="size-4" aria-hidden="true" />
   663→                          )}
   664→                          مواضع بیشتر
   665→                        </Button>
   666→                        <p className="text-xs text-muted-foreground">
   667→                          ادامهٔ بررسی برای نمایش مواضع تأییدشدهٔ بیشتر از همین جست‌وجو.
   668→                        </p>
   669→                      </div>
   670→                    ) : null}
   671→                  </TabsContent>
   672→                  <TabsContent value="books">
   673→                    <BooksGroupedView
   674→                      sessionUid={sessionUid}
   675→                      onSessionExpired={() => void sessionQuery.refetch()}

   678→                </Tabs>
   679→              )}
   680→            </div>
   681→          )}
   682→        </div>
   683→
   684→        {/* ستون کناری: جست‌وجوهای ذخیره‌شده (SEARCH-16) */}
   685→        <aside aria-label="جست‌وجوهای ذخیره‌شده" className="min-w-0">
   686→          <SavedSearchesPanel
   687→            currentPayload={request}
   688→            saveEnabled={hasQuery}
   689→            saveOpen={saveOpen}
   690→            onSaveOpenChange={setSaveOpen}
   691→          />
   692→        </aside>
   693→      </div>
   694→    </div>
   695→  );
