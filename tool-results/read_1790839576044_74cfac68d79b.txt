    41→    41→interface Segment {
    42→    42→  start: number;
    43→    43→  end: number;
    44→    44→  mark?: Mark;

    47→    47→/** پوشش بازه‌ها با اولویت؛ خروجی قطعه‌های بدون هم‌پوشانی است. */
    48→    48→function buildSegments(cpLen: number, marks: Mark[]): Segment[] {
    49→    49→  if (cpLen === 0) return [];
    50→    50→  const sorted = [...marks]
    51→    51→    .filter((m) => m.end > m.start)
    52→    52→    .sort((a, b) => a.start - b.start || b.priority - a.priority);
    53→    53→
    54→    54→  const segs: Segment[] = [];
    55→    55→  let cursor = 0;
    56→    56→  while (cursor < cpLen) {
    57→    57→    // مرز بعدی: نزدیک‌ترین شروع یا پایان markها بعد از cursor
    58→    58→    let boundary = cpLen;
    59→    59→    for (const m of sorted) {
    60→    60→      if (m.start > cursor) boundary = Math.min(boundary, m.start);
    61→    61→      if (m.end > cursor) boundary = Math.min(boundary, m.end);
    62→    62→    }
    63→    63→    if (boundary <= cursor) break;
    64→    64→    // پراولویت‌ترین mark که کل بازهٔ [cursor, boundary) را می‌پوشاند
    65→    65→    let best: Mark | undefined;
    66→    66→    for (const m of sorted) {
    67→    67→      if (m.start <= cursor && m.end >= boundary) {
    68→    68→        if (!best || m.priority > best.priority) best = m;
    69→    69→      }
    70→    70→    }
    71→    71→    segs.push({ start: cursor, end: boundary, mark: best });
    72→    72→    cursor = boundary;
    73→    73→  }
    74→    74→  return segs;
    75→    75→}

   100→   100→  const quranLh = prefs.quran_line_height ?? QURAN_LINE_HEIGHT_DEFAULT;

   106→   106→      lang: "ar",
   107→   107→      dir: "rtl",
   108→   108→    };
   109→   109→  }
   110→   110→  if (isLatinBlock(block.text)) {
   111→   111→    return {
   112→   112→      fontFamily: "var(--font-latin)",
   113→   113→      fontSize: `${prefs.font_size}px`,
   114→   114→      lineHeight: prefs.line_height,
   115→   115→      lang: "en",
   116→   116→      dir: "ltr",

   118→   118→  }
   119→   119→  const persian = blockIsPersian(block);
   120→   120→  return {
   121→   121→    fontFamily: persian ? "var(--font-ui)" : "var(--font-arabic)",
   122→   122→    fontSize: `${prefs.font_size}px`,
   123→   123→    lineHeight: prefs.line_height,
   124→   124→    lang: persian ? "fa" : "ar",

   129→   129→function citationAddress(block: PageBlock): string | null {
   130→   130→  for (const c of block.citations ?? []) {
   131→   131→    if (c.address && c.address.trim() !== "") return c.address.trim();
   132→   132→    if (c.sure) {

   167→   167→  block,

   187→   187→      out.push({
   188→   188→        start: Math.max(0, Math.min(s.start, cpLen)),
   189→   189→        end: Math.max(0, Math.min(s.end, cpLen)),
   190→   190→        priority: 1,
   191→   191→        span: s,
   192→   192→      });
   193→   193→    }
   194→   194→    for (const l of block.links ?? []) {
   195→   195→      out.push({
   196→   196→        start: Math.max(0, Math.min(l.start, cpLen)),
   197→   197→        end: Math.max(0, Math.min(l.end, cpLen)),
   198→   198→        priority: 2,
   199→   199→        link: l,
   200→   200→      });
   201→   201→    }
   202→   202→    for (const f of block.footnote_refs ?? []) {
   203→   203→      out.push({
   204→   204→        start: Math.max(0, Math.min(f.start, cpLen)),
   205→   205→        end: Math.max(0, Math.min(f.end, cpLen)),
   206→   206→        priority: 3,
   207→   207→        refUid: f.ref,
   208→   208→      });
   209→   209→    }
   210→   210→    if (highlight && highlight.end > highlight.start) {
   211→   211→      out.push({
   212→   212→        start: Math.max(0, Math.min(highlight.start, cpLen)),
   213→   213→        end: Math.max(0, Math.min(highlight.end, cpLen)),
   214→   214→        priority: 0,
   215→   215→        isHl: true,
   216→   216→      });

   218→   218→    return out;
   219→   219→  }, [block, cpLen, highlight]);
   220→   220→
   221→   221→  const segs = useMemo(() => buildSegments(cpLen, marks), [cpLen, marks]);
   222→   222→
   223→   223→  return (

   225→   225→      {segs.map((seg, i) => {

   232→   232→            </span>

   237→   237→            <mark
   238→   238→              key={i}
   239→   239→              id="reader-hl-anchor"
   240→   240→              data-cp-start={seg.start}
   241→   241→              data-cp-end={seg.end}
   242→   242→              className="rounded-sm bg-match px-0.5 text-foreground"

   244→   244→              {text}

   253→   253→              key={i}
   254→   254→              type="button"
   255→   255→              id={anchorId}
   256→   256→              data-cp-start={seg.start}
   257→   257→              data-cp-end={seg.end}
   258→   258→              className="mx-0.5 inline cursor-pointer align-super text-[0.62em] font-semibold text-link hover:underline"

   260→   260→              aria-label={`نمایش پاورقی ${label}`}
   261→   261→              title="نمایش پاورقی"

   287→   287→              {text}

   289→   289→          );
   290→   290→        }
   291→   291→        switch (span.role) {
   292→   292→          case "sanad":
   293→   293→            return (
   294→   294→              <span
   295→   295→                key={i}
   296→   296→                data-cp-start={seg.start}
   297→   297→                data-cp-end={seg.end}
   298→   298→                className="rounded bg-muted/55 px-0.5 text-muted-foreground"
   299→   299→              >
   300→   300→                {text}
   301→   301→              </span>
   302→   302→            );
   303→   303→          case "G":
   304→   304→            return (
   305→   305→              <Tooltip key={i}>
   306→   306→                <TooltipTrigger asChild>
   307→   307→                  <span
   308→   308→                    data-cp-start={seg.start}
   309→   309→                    data-cp-end={seg.end}
   310→   310→                    className="inline-flex items-baseline gap-0.5 rounded border border-border bg-surface px-0.5"
   311→   311→                  >

   499→   499→function cpOffsetFromNode(node: Node | null, offset: number): {
   500→   500→  blockId: string | null;
   501→   501→  cp: number;
   502→   502→} {
   503→   503→  if (!node) return { blockId: null, cp: 0 };
   504→   504→  let textNode: Text | null = null;
   505→   505→  let utf16Offset = offset;
   506→   506→  if (node.nodeType === Node.TEXT_NODE) {
   507→   507→    textNode = node as Text;
   508→   508→  } else {
   509→   509→    const child = node.childNodes[offset] ?? node.firstChild;
   510→   510→    if (child && child.nodeType === Node.TEXT_NODE) {
   511→   511→      textNode = child as Text;
   512→   512→      utf16Offset = 0;
   513→   513→    }
   514→   514→  }
   515→   515→  if (!textNode) return { blockId: null, cp: 0 };
   516→   516→  const spanEl = textNode.parentElement?.closest<HTMLElement>("span[data-cp-start], mark[data-cp-start]");
   517→   517→  const blockEl = textNode.parentElement?.closest<HTMLElement>("[data-block-id]");
   518→   518→  if (!blockEl) return { blockId: null, cp: 0 };
   519→   519→  const base = spanEl ? Number(spanEl.dataset.cpStart ?? 0) : 0;
   520→   520→  const rel = Array.from((textNode.textContent ?? "").slice(0, utf16Offset)).length;
   521→   521→  return { blockId: blockEl.dataset.blockId ?? null, cp: base + rel };
   522→   522→}

   724→   724→export function PageRenderer({
   725→   725→  page,
   726→   726→  prefs,
   727→   727→  sectionUidByBlockId,
   728→   728→  flashBlockId,
   729→   729→  highlight,
   730→   730→  collapsedSanads,
   731→   731→  onToggleSanad,
   732→   732→  onFootnoteClick,
   733→   733→  onLinkClick,
   734→   734→  onSelection,
   735→   735→  onVerseClick,
   736→   736→  onHadithClick,
   737→   737→}: PageRendererProps) {
   738→   738→  const wrapRef = useRef<HTMLDivElement>(null);
   739→   739→  const [selRect, setSelRect] = useState<{ top: number; left: number } | null>(null);
   740→   740→  const [sel, setSel] = useState<ReaderSelection | null>(null);
   741→   741→
   742→   742→  /* هایلایت نتیجهٔ جست‌وجو: پس از رندر، به آن اسکرول می‌شود (یک بار برای هر بازه) */
   743→   743→  const hlKey = highlight ? `${highlight.start}:${highlight.end}:${highlight.blockId ?? ""}` : "";
   744→   744→  const lastScrolled = useRef<string>("");
   745→   745→  useEffect(() => {
   746→   746→    if (!hlKey || lastScrolled.current === hlKey) return;
   747→   747→    lastScrolled.current = hlKey;
   748→   748→    const t = window.setTimeout(() => {
   749→   749→      document.getElementById("reader-hl-anchor")?.scrollIntoView({
   750→   750→        behavior: "smooth",
   751→   751→        block: "center",
   752→   752→      });
   753→   753→    }, 250);
   754→   754→    return () => window.clearTimeout(t);
   755→   755→  }, [hlKey]);
   756→   756→
   757→   757→  const clearSelection = useCallback(() => {
   758→   758→    setSel(null);
   759→   759→    setSelRect(null);
   760→   760→  }, []);
   761→   761→
   762→   762→  const computeSelection = useCallback(() => {
   763→   763→    const wrapper = wrapRef.current;
   764→   764→    if (!wrapper) return;
   765→   765→    const s = window.getSelection();
   766→   766→    if (!s || s.isCollapsed || s.rangeCount === 0) {
   767→   767→      clearSelection();
   768→   768→      return;
   769→   769→    }
   770→   770→    const range = s.getRangeAt(0);
   771→   771→    if (!wrapper.contains(range.commonAncestorContainer)) {
   772→   772→      clearSelection();
   773→   773→      return;
   774→   774→    }
   775→   775→    const a = cpOffsetFromNode(range.startContainer, range.startOffset);
   776→   776→    const b = cpOffsetFromNode(range.endContainer, range.endOffset);
   777→   777→    const blockId = a.blockId ?? b.blockId;
   778→   778→    if (!blockId) {
   779→   779→      clearSelection();
   780→   780→      return;
   781→   781→    }
   782→   782→    const block = page.blocks.find((bl) => bl.id === blockId);
   783→   783→    if (!block) {
   784→   784→      clearSelection();
   785→   785→      return;
   786→   786→    }
   787→   787→    const blockLen = Array.from(block.text).length;
   788→   788→    let start: number;
   789→   789→    let end: number;
   790→   790→    if (a.blockId === b.blockId) {
   791→   791→      start = Math.min(a.cp, b.cp);
   792→   792→      end = Math.max(a.cp, b.cp);
   793→   793→    } else {
   794→   794→      // انتخاب چندبلوکی: به بلوک لنگر محدود می‌شود (snapshot فیش یک بازهٔ پیوسته است)
   795→   795→      if (a.blockId === blockId) {
   796→   796→        start = a.cp;
   797→   797→        end = blockLen;
   798→   798→      } else {
   799→   799→        start = 0;
   800→   800→        end = Math.min(b.cp, blockLen);
   801→   801→      }
   802→   802→      if (end < start) [start, end] = [end, start];
   803→   803→    }
   804→   804→    if (end - start < 1) {
   805→   805→      clearSelection();
   806→   806→      return;
   807→   807→    }
   808→   808→    const text = Array.from(block.text).slice(start, end).join("");
   809→   809→    const rect = range.getBoundingClientRect();
   810→   810→    const wrapRect = wrapper.getBoundingClientRect();
   811→   811→    setSelRect({ top: rect.top - wrapRect.top, left: rect.left - wrapRect.left });
   812→   812→    setSel({ blockId, start, end, text, hadithUid: block.hadith_uid ?? null });
   813→   813→  }, [clearSelection, page.blocks]);
   814→   814→
   815→   815→  // با اسکرول، دکمهٔ شناور پاک می‌شود تا جای آن از متن جدا نماند
   816→   816→  useEffect(() => {
   817→   817→    if (!sel) return;
   818→   818→    const onScroll = () => clearSelection();
   819→   819→    window.addEventListener("scroll", onScroll, true);
   820→   820→    return () => window.removeEventListener("scroll", onScroll, true);
   821→   821→  }, [sel, clearSelection]);
   822→   822→
   823→   823→  const footnoteLabelFor = useCallback(
   824→   824→    (refUid: string) => {
   825→   825→      const fn = page.footnotes.find((f) => f.uid === refUid);
   826→   826→      return fn?.label ?? null;
   827→   827→    },
   828→   828→    [page.footnotes],
   829→   829→  );
   830→   830→
   831→   831→  return (
   832→   832→    <div
   833→   833→      ref={wrapRef}
   834→   834→      className="relative select-text rounded-2xl border border-border bg-reader px-4 py-5 shadow-sm md:px-8 md:py-6"
   835→   835→      onMouseUp={computeSelection}
   836→   836→      onKeyUp={(e) => {
   837→   837→        if (e.shiftKey || e.key.startsWith("Arrow")) computeSelection();
   838→   838→      }}
   839→   839→    >
   840→   840→      {page.blocks.length === 0 ? (
   841→   841→        <p className="py-10 text-center text-sm text-muted-foreground">
   842→   842→          این صفحه متنی برای نمایش ندارد.
   843→   843→        </p>
   844→   844→      ) : (
   845→   845→        page.blocks.map((block, idx) => {
   846→   846→          const sectionUid = sectionUidByBlockId[block.id];
   847→   847→          const flash = flashBlockId === block.id;
   848→   848→          const untaggedRun =
   849→   849→            block.untagged === true &&
   850→   850→            (idx === 0 || page.blocks[idx - 1]?.untagged !== true);
   851→   851→          const hlHere =
   852→   852→            highlight && highlight.end > highlight.start
   853→   853→              ? highlight.blockId
   854→   854→                ? highlight.blockId === block.id
   855→   855→                : // بدون blockId: اولین بلوکی که بازه در آن جا می‌شود
   856→   856→                  idx === 0 && highlight.end <= Array.from(block.text).length
   857→   857→              : null;
   858→   858→          return (
   859→   859→            <div key={block.id} className="group">
   860→   860→              {untaggedRun ? (
   861→   861→                <p className="mt-5 mb-2 inline-flex items-center gap-1.5 rounded-full border border-border bg-muted px-2.5 py-0.5 text-[11px] text-muted-foreground first:mt-0">
   862→   862→                  <span
   863→   863→                    className="inline-block size-1.5 rounded-full bg-muted-foreground/60"
   864→   864→                    aria-hidden="true"
   865→   865→                  />
   866→   866→                  این منبع نشانه‌گذاری تخصصی ندارد
   867→   867→                </p>
   868→   868→              ) : null}
