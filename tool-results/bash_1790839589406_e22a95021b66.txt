    67→function clampPage(raw: string | undefined): number {
    68→  const n = Number.parseInt(raw ?? "", 10);
    69→  return Number.isFinite(n) && n >= 1 && n <= 200 ? n : 1;

    72→/** بازسازی hash از payload جست‌وجوی ذخیره‌شده (اجرای دوباره با همان نشانی) */
    73→function hashFromPayload(payload: unknown): Record<string, string> | null {
    74→  if (!payload || typeof payload !== "object") return null;
    75→  const p = payload as Record<string, unknown>;
    76→  if (typeof p.query !== "string" || typeof p.mode !== "string") return null;
    77→  const out: Record<string, string> = { q: p.query, mode: p.mode };
    78→  const scope = (p.scope && typeof p.scope === "object" ? p.scope : {}) as Record<string, unknown>;
    79→  const target = typeof scope.target === "string" ? scope.target : "library";
    80→  if (target === "titles") {
    81→    out.scope = "titles";
    82→  } else if (target === "work" && typeof scope.work_uid === "string" && scope.work_uid) {
    83→    out.scope = "work";
    84→    out.work = scope.work_uid;
    85→  } else if (
    86→    target === "document" &&
    87→    Array.isArray(scope.document_uids) &&
    88→    typeof scope.document_uids[0] === "string"
    89→  ) {
    90→    out.scope = "document";
    91→    out.doc = scope.document_uids[0];
    92→    if (typeof scope.work_uid === "string") out.work = scope.work_uid;
    93→  } else {
    94→    out.scope = "library";
    95→  }
    96→  const roles = Array.isArray(scope.roles)
    97→    ? scope.roles.filter((r): r is string => typeof r === "string")
    98→    : [];
    99→  const roleValues = ROLE_FILTERS.filter((f) => f.codes.some((c) => roles.includes(c))).map(
   100→    (f) => f.value,
   101→  );
   102→  if (roleValues.length > 0) out.roles = roleValues.join(",");
   103→  if (typeof scope.language === "string" && scope.language) out.lang = scope.language;
   104→  const near = (p.near && typeof p.near === "object" ? p.near : null) as Record<string, unknown> | null;
   105→  if (near) {
   106→    out.slop = String(typeof near.slop === "number" ? near.slop : 3);
   107→    if (near.ordered === false) out.ordered = "0";
   108→  }
   109→  return out;
   110→}

   315→export function SearchView() {
   316→  const { route, navigate } = useRoute();
   317→
   318→  /* وضعیت اجراشدهٔ جست‌وجو (از hash) */
   319→  const params = parseSearchParams(route.query);
   320→  const searchKey = useMemo(() => searchStateKey(params), [params]);
   321→  const hashPage = clampPage(route.query.page);
   322→  const [saveOpen, setSaveOpen] = useState(false);
   323→  const view: "results" | "books" = route.query.view === "books" ? "books" : "results";
   324→  // عبارت یا استناد قرآنی (QUR-03) هر دو شروع‌کنندهٔ جست‌وجو هستند
   325→  const hasQuery = params.q.trim().length > 0 || params.sure.trim().length > 0;
   326→
   327→  /* پیش‌نویس فرم — هم‌گام‌سازی با hash در زمان رندر (الگوی مجاز React) */
   328→  const [draft, setDraft] = useState<SearchHashState>(() => parseSearchParams(route.query));
   329→  const [draftHash, setDraftHash] = useState(route.fullHash);
   330→  if (draftHash !== route.fullHash) {
   331→    setDraftHash(route.fullHash);
   332→    setDraft(parseSearchParams(route.query));
   333→  }
   334→
   335→  /** ساخت پارامترهای hash از وضعیت فرم — مشترک میان setQuery و submit */
   336→  const searchQueryParams = useCallback(
   337→    (state: SearchHashState): Record<string, string | undefined> => ({
   338→      q: state.q.trim() || undefined,
   339→      mode: state.mode,
   340→      scope: state.scope,
   341→      work: state.work || undefined,
   342→      doc: state.doc || undefined,
   343→      roles: state.roles.length > 0 ? state.roles.join(",") : undefined,
   344→      lang: state.lang || undefined,
   345→      slop: state.mode === "near" ? String(state.slop) : undefined,
   346→      ordered: state.mode === "near" && !state.ordered ? "0" : undefined,
   347→    }),
   348→    [],
   349→  );
   350→
   351→  /** نوشتن وضعیت در hash — هر بار کل پارامترها داده می‌شود */
   352→  const setQuery = useCallback(
   353→    (state: SearchHashState, extra: Record<string, string | undefined> = {}, replace = false) => {
   354→      navigate(
   355→        "/search",
   356→        {
   357→          ...searchQueryParams(state),
   358→          page: undefined,
   359→          view: undefined,
   360→          ...extra,
   361→        },
   362→        { replace },
   363→      );
   364→    },
   365→    [navigate, searchQueryParams],
   366→  );
   367→
   368→  const queryClient = useQueryClient();
   369→  const submit = useCallback(() => {
   370→    /* اگر پارامترهای جست‌وجو تغییر نکرده باشد، navigate بی‌اثر است و کشِ
   371→       نشست دوباره استفاده می‌شد — برای «اجرای دوبارهٔ همان جست‌وجو» (پس از
   372→       انقضای نشست سرور یا بازسازی نمایه) کش باطل و بازاجرایی می‌شود
   373→       (SEARCH-20/WORK-02). در غیر این صورت hash با پارامترهای تازه نوشته
   374→       می‌شود و کلید تازه، نشست تازه می‌سازد. */
   375→    if (searchStateKey(draft) === searchKey) {
   376→      queryClient.invalidateQueries({ queryKey: ["search-session", searchKey] });
   377→      return;
   378→    }
   379→    setQuery(draft);
   380→  }, [draft, queryClient, searchKey, setQuery]);
   381→
   382→  const patchCommitted = useCallback(
   383→    (patch: Partial<SearchHashState>) => {
   384→      setQuery({ ...params, ...patch });
   385→    },
   386→    [params, setQuery],
   387→  );
   388→
   389→  /* لایهٔ ۱ — نشست جست‌وجو: صفحهٔ نخست + شمارش صادقانه */
   390→  const request = buildSearchRequest(params);
   391→
   392→  const sessionQuery = useQuery({
   393→    queryKey: ["search-session", searchKey],
   394→    queryFn: async ({ signal }) => {
   395→      const resp = await api.post<SearchResponse>("/search", request);
   396→      if (signal.aborted) throw new DOMException("aborted", "AbortError");
   397→      return resp;
   398→    },
   399→    enabled: hasQuery,
   400→    staleTime: 4 * 60_000,
   401→    gcTime: 10 * 60_000,
   402→    retry: false,
   403→    refetchOnWindowFocus: false,
   404→    refetchOnReconnect: false,
   405→  });
   406→
   407→  const session = hasQuery ? sessionQuery.data ?? null : null;
   408→  const sessionUid = session?.session_uid ?? null;
   409→  const sessionUidRef = useRef<string | null>(null);
   410→  useEffect(() => {
   411→    sessionUidRef.current = sessionUid;
   412→  }, [sessionUid]);
   413→
   414→  /* لایهٔ ۲ — صفحه‌های بیشتر با «ادامهٔ بررسی» (cursor امضاشده، §۴۷) */
   415→  const [extras, setExtras] = useState<ExtrasState>(EMPTY_EXTRAS);
   416→  const currentExtras = sessionUid && extras.uid === sessionUid ? extras : EMPTY_EXTRAS;
   417→  const extrasRef = useRef(currentExtras);
   418→  useEffect(() => {
   419→    extrasRef.current = currentExtras;
   420→  }, [currentExtras]);
   421→
   422→  const meta = currentExtras.meta ?? session;
   423→  const items = useMemo(
   424→    () => [...(session?.results ?? []), ...currentExtras.items],
   425→    [session, currentExtras.items],
   426→  );
   427→
   428→  const restoreStampRef = useRef<string>("");
   429→
   430→  const runContinue = useCallback(
   431→    async (targetSessionUid: string, count: number, updateHash: boolean) => {
   432→      let acc: ExtrasState =
   433→        extrasRef.current.uid === targetSessionUid
   434→          ? extrasRef.current
   435→          : { ...EMPTY_EXTRAS, uid: targetSessionUid };
   436→      if (acc.loading) return;
   437→      acc = { ...acc, loading: true, error: null };
   438→      extrasRef.current = acc;
   439→      setExtras(acc);
   440→
   441→      for (let i = 0; i < count; i += 1) {
   442→        if (sessionUidRef.current !== targetSessionUid) return; // جست‌وجوی تازه‌تری اجرا شده است
   443→        const cursorMeta = acc.meta ?? session ?? null;
   444→        if (!cursorMeta?.continuation_available || !cursorMeta.next_cursor) break;
   445→        try {
   446→          const resp = await api.post<SearchResponse>(
   447→            `/search/${targetSessionUid}/continue`,
   448→            { cursor: cursorMeta.next_cursor },
   449→          );
   450→          if (sessionUidRef.current !== targetSessionUid) return;
   451→          acc = {
   452→            ...acc,
   453→            count: acc.count + 1,
   454→            items: [...acc.items, ...(resp.results ?? [])],
   455→            meta: resp,
   456→            loading: i < count - 1,
   457→          };
   458→        } catch (err) {
   459→          if (err instanceof ApiError && err.code === "gone_cursor") {
   460→            acc = { ...acc, expired: true, loading: false };
   461→          } else {
   462→            acc = {
   463→              ...acc,
   464→              error: errorMessage(err, "ادامهٔ بررسی ناموفق بود؛ دوباره تلاش کنید."),
   465→              loading: false,
   466→            };
   467→          }
   468→          break;
   469→        }
   470→        extrasRef.current = acc;
   471→        setExtras(acc);
   472→      }
   473→      acc = { ...acc, loading: false };
   474→      extrasRef.current = acc;
   475→      setExtras(acc);
   476→
   477→      // هم‌گام‌سازی موضع در hash (WORK-07): شمارهٔ صفحه = شمار صفحه‌های اضافه + ۱
   478→      if (updateHash || acc.expired || acc.error) {
   479→        navigate(
   480→          "/search",
   481→          { ...route.query, page: acc.count > 0 ? String(acc.count + 1) : undefined },
   482→          { replace: true },
   483→        );
   484→      }
   485→    },
   486→    [navigate, route.query, session],  
   487→  );
   488→
   489→  /* بازسازی موضع پس از بازگشت به نتایج (WORK-07): زنجیرهٔ continue تا صفحهٔ hash */
   490→  useEffect(() => {
   491→    const targetExtras = hashPage - 1;
   492→    if (!sessionUid || targetExtras <= 0 || currentExtras.loading) return;
   493→    if (currentExtras.count >= targetExtras) return;
   494→    if (meta && !meta.continuation_available) return;
   495→    const stamp = `${searchKey}|${sessionUid}|${targetExtras}`;
   496→    if (restoreStampRef.current === stamp) return;
   497→    restoreStampRef.current = stamp;
   498→    const t = window.setTimeout(() => {
   499→      void runContinue(sessionUid, targetExtras - currentExtras.count, false);
   500→    }, 0);
   501→    return () => window.clearTimeout(t);
   502→  }, [sessionUid, hashPage, currentExtras.count, currentExtras.loading, meta, searchKey, runContinue]);
   503→
   504→  /* وضعیت نمایه — تفکیک «بدون نتیجه» از «نمایه آماده نیست» (SEARCH-21) */
   505→  const isEmpty = sessionQuery.isSuccess && items.length === 0 && !currentExtras.loading;
   506→  const statusQuery = useQuery({
   507→    queryKey: ["library-status"],
   508→    queryFn: ({ signal }) =>
   509→      api.get<{ index_ready?: boolean | null }>("/status", undefined, signal),
   510→    enabled: isEmpty,
   511→    staleTime: 60_000,
   512→  });
   513→
   514→  const loadingMore = currentExtras.loading;
   515→  const switchView = (next: string) => {
   516→    navigate(
   517→      "/search",
   518→      { ...route.query, view: next === "books" ? "books" : undefined },
   519→      { replace: true },
   520→    );
   521→  };
   522→
   523→  /* ---------- پنل «کنار نتایج» (SEARCH-11) — مختصات در hash (WORK-13) ---------- */
   524→  const splitLocator: SplitLocator | null = useMemo(() => {
   525→    const doc = route.query.spd;
   526→    if (!doc) return null;
   527→    const hlRaw = route.query.sph ?? "";
   528→    const s, he] = hlRaw.split(":").map((v) => Number.parseInt(v, 10));
   529→    const hl: [number, number] | null =
   530→      Number.isFinite(hs) && Number.isFinite(he) && he > hs ? s, he] : null;
   531→    return {
   532→      documentUid: doc,
   533→      documentLabel: route.query.spdl ?? "",
   534→      workUid: route.query.spw ?? "",
   535→      workTitle: route.query.spwt ?? "",
   536→      pageUid: route.query.spp || null,
   537→      hl,
   538→      key: route.query.spk ?? `${doc}:${route.query.spp ?? ""}`,
   539→    };
   540→  }, [route.query]);
   541→
   542→  const openSplit = useCallback(
   543→    (result: SearchResultItem) => {
   544→      if (!result.document?.uid) return;
   545→      const offsets =
   546→        (result.match_offsets?.[0] !== undefined ? { start: result.match_offsets[0], end: result.match_offsets[1] } : null) ??
   547→        (result.source_offsets && typeof result.source_offsets === "object"
   548→          ? Array.isArray(result.source_offsets)
   549→            ? { start: result.source_offsets[0], end: result.source_offsets[1] }
   550→            : { start: result.source_offsets.start, end: result.source_offsets.end }
   551→          : null);
   552→      const pageUid = result.page?.uid ?? "";
   553→      navigate(
   554→        "/search",
   555→        {
   556→          ...route.query,
   557→          view: undefined,
   558→          spd: result.document.uid,
   559→          spdl: result.document.label ?? "",
   560→          spw: result.work?.uid ?? "",
   561→          spwt: result.work?.title ?? "",
   562→          spp: pageUid,
   563→          sph: offsets ? `${offsets.start}:${offsets.end}` : undefined,
   564→          spk: `${result.passage_uid}:${pageUid}`,
   565→        },
   566→        { replace: false },
   567→      );
   568→    },
   569→    [navigate, route.query],
   570→  );
   571→
   572→  const closeSplit = useCallback(() => {
   573→    navigate(
   574→      "/search",
   575→      {
   576→        ...route.query,
   577→        spd: undefined,
   578→        spdl: undefined,
   579→        spw: undefined,
   580→        spwt: undefined,
   581→        spp: undefined,
   582→        sph: undefined,
   583→        spk: undefined,
   584→      },
   585→      { replace: true },
   586→    );
   587→  }, [navigate, route.query]);
   588→
   589→  /** ناوبری درون پنل (صفحهٔ قبل/بعد) — نوشتن صفحهٔ تازه در hash */
   590→  const navigateSplit = useCallback(
   591→    (next: SplitLocator) => {
   592→      navigate(
   593→        "/search",
   594→        {
   595→          ...route.query,
   596→          spp: next.pageUid ?? undefined,
   597→          sph: next.hl ? `${next.hl[0]}:${next.hl[1]}` : undefined,
   598→          spk: next.key,
   599→        },
   600→        { replace: true },
   601→      );
   602→    },
   603→    [navigate, route.query],
   604→  );
   605→
   606→  return (
   607→    <div className="mx-auto w-full max-w-[110rem] px-4 py-8">
   608→      <header className="mb-6">
   609→        <h1 className="text-xl font-bold">جست‌وجو در منابع</h1>
   610→        <p className="mt-1 text-sm text-muted-foreground">
   611→          جست‌وجوی عبارت در متن منابع با انتخاب حالت و دامنه؛ فقط مواضع تأییدشده نمایش داده می‌شود.
   612→        </p>
   613→      </header>
   614→
   615→      <div
   616→        className={`grid items-start gap-6 ${
   617→          splitLocator
   618→            ? "xl:grid-cols-inmax(0,1fr)_minmax(380px,44%)]"
   619→            : "lg:grid-cols-inmax(0,1fr)_290px]"
   620→        }`}
   621→      >
   622→        <div className="min-w-0 space-y-5">
   623→          {/* فرم جست‌وجو */}
   624→          <Card>
   625→            <CardContent className="p-4 sm:p-6">
   626→              <SearchForm
   627→                value={draft}
   628→                onChange={(patch) => setDraft((d) => ({ ...d, ...patch }))}
   629→                onSubmit={submit}
   630→                busy={hasQuery && sessionQuery.isPending}
   631→              />
   632→            </CardContent>
   633→          </Card>
   634→
   635→          {/* چیپ فیلترها (SEARCH-20) */}
   636→          {hasQuery ? <FilterChips state={params} onChange={patchCommitted} /> : null}
   637→
   638→          {/* نتایج */}
   639→          {!hasQuery ? (
   640→            <EmptyState
   641→              title="عبارتی برای جست‌وجو وارد نشده است"
   642→              hint="عبارت مورد نظر را در کادر بالا بنویسید و «جست‌وجو» را بزنید."
   643→            />
   644→          ) : sessionQuery.isPending ? (
   645→            <div className="space-y-3" aria-busy="true">
   646→              <ResultCardSkeleton />
   647→              <ResultCardSkeleton />
   648→              <ResultCardSkeleton />
   649→              <span className="sr-only">در حال جست‌وجو در منابع…</span>
   650→            </div>
   651→          ) : sessionQuery.isError ? (
   652→            /* وضعیت خالی ۳ — خطای موتور جست‌وجو */
   653→            <ErrorState
   654→              message={errorMessage(
   655→                sessionQuery.error,
   656→                "جست‌وجو با خطا روبه‌رو شد؛ دوباره تلاش کنید.",
   657→              )}
   658→              onRetry={() => void sessionQuery.refetch()}
   659→            />
   660→          ) : (
   661→            <div className="space-y-4">
   662→              {/* هشدار پایان مهلت نشست (۴۱۰ در ادامهٔ بررسی) */}
   663→              {currentExtras.expired ? (
   664→                <Alert>
   665→                  <TriangleAlert className="size-4" aria-hidden="true" />
   666→                  <AlertTitle>مهلت این جست‌وجو پایان یافته است</AlertTitle>
   667→                  <AlertDescription className="flex flex-wrap items-center gap-3">
   668→                    <span>
   669→                      مواضع بیشتری از این نشست قابل دریافت نیست؛ جست‌وجو را دوباره اجرا کنید.
   670→                    </span>
   671→                    <Button
   672→                      variant="outline"
   673→                      size="sm"
   674→                      className="min-h-9 gap-1.5"
   675→                      onClick={() => void sessionQuery.refetch()}
   676→                    >
   677→                      <RotateCcw className="size-4" aria-hidden="true" />
   678→                      اجرای دوباره
   679→                    </Button>
   680→                  </AlertDescription>
   681→                </Alert>
   682→              ) : null}
   683→              {currentExtras.error ? (
   684→                <Alert variant="destructive">
   685→                  <TriangleAlert className="size-4" aria-hidden="true" />
   686→                  <AlertTitle>ادامهٔ بررسی ناموفق بود</AlertTitle>
   687→                  <AlertDescription>{currentExtras.error}</AlertDescription>
   688→                </Alert>
   689→              ) : null}
   690→
   691→              {/* شمارش صادقانهٔ §۴۷ + دکمهٔ ذخیره (SEARCH-16) */}
   692→              <div className="flex flex-wrap items-center justify-between gap-3">
   693→                <div className="min-w-0 space-y-0.5">
   694→                  <VerifiedCountLine
   695→                    occurrences={meta?.verified_occurrences ?? null}
   696→                    countRelation={meta?.count_relation ?? ""}
   697→                    passages={meta?.verified_passages ?? null}
   698→                    hadiths={meta?.verified_hadiths ?? null}
   699→                  />
   700→                  {meta && meta.completion_state !== "complete" ? (
   701→                    <p className="text-xs text-muted-foreground">
   702→                      {meta.completion_state === "failed"
   703→                        ? "بررسی این جست‌وجو ناتمام ماند؛ مواضع نمایش‌داده‌شده تأییدشده‌اند."
   704→                        : "بررسی همچنان ادامه دارد؛ شمارش تا این لحظه کامل نیست."}
   705→                    </p>
   706→                  ) : null}
   707→                </div>
   708→                <Button
   709→                  variant="outline"
   710→                  size="sm"
   711→                  className="min-h-9 gap-1.5"
   712→                  onClick={() => setSaveOpen(true)}
   713→                >
   714→                  <BookmarkPlus className="size-4" aria-hidden="true" />
   715→                  ذخیرهٔ این جست‌وجو
   716→                </Button>
   717→              </div>
   718→
   719→              {items.length === 0 ? (
   720→                statusQuery.data?.index_ready === false ? (
   721→                  /* وضعیت خالی ۲ — نمایه آماده نیست */
   722→                  <EmptyState
   723→                    title="نمایهٔ کتابخانه در حال آماده‌سازی است؛ به‌زودی در دسترس خواهد بود"
   724→                    hint="پس از آماده‌شدن نمایه، نتایج همین جست‌وجو نمایش داده می‌شود."
   725→                  />
   726→                ) : (
   727→                  /* وضعیت خالی ۱ — بدون نتیجه */
   728→                  <EmptyState
   729→                    title="موردی یافت نشد"
   730→                    hint="حالت «عبارت نرمال‌شده» را امتحان کنید یا فیلترها را بردارید؛ درست‌نویسی عبارت را هم بررسی کنید."
   731→                    action={
   732→                      params.mode !== "normalized_phrase" ? (
   733→                        <Button
   734→                          variant="outline"
   735→                          className="min-h-11"
   736→                          onClick={() => patchCommitted({ mode: "normalized_phrase" })}
   737→                        >
   738→                          جست‌وجو با حالت نرمال‌شده
   739→                        </Button>
   740→                      ) : undefined
   741→                    }
   742→                  />
   743→                )
   744→              ) : (
   745→                <Tabs value={view} onValueChange={switchView}>
   746→                  <TabsList aria-label="سطح مشاهدهٔ نتایج" className="w-full sm:w-auto">
   747→                    <TabsTrigger value="results" className="min-h-9">
   748→                      مواضع
   749→                    </TabsTrigger>
   750→                    <TabsTrigger value="books" className="min-h-9">
   751→                      کتاب‌های دارای تطبیق تأییدشده
   752→                    </TabsTrigger>
   753→                  </TabsList>
   754→                  <TabsContent value="results" className="space-y-3">
   755→                    <GroupedPositions
   756→                      items={items}
   757→                      releaseUid={meta?.release_uid ?? null}
   758→                      onOpenSplit={openSplit}
   759→                      splitActiveUid={splitLocator ? splitLocator.key.split(":")[0] : null}
   760→                    />
   761→                    {/* صفحه‌بندی مواضع: ادامهٔ بررسی */}
   762→                    {meta?.continuation_available ? (
   763→                      <div className="flex flex-col items-center gap-1 pt-2">
   764→                        <Button
   765→                          variant="outline"
   766→                          className="min-h-11 gap-2 px-6"
   767→                          disabled={loadingMore}
   768→                          onClick={() => {
   769→                            if (sessionUid) void runContinue(sessionUid, 1, true);
   770→                          }}
   771→                        >
   772→                          {loadingMore ? (
   773→                            <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   774→                          ) : (
   775→                            <ChevronDown className="size-4" aria-hidden="true" />
   776→                          )}
   777→                          مواضع بیشتر
   778→                        </Button>
   779→                        <p className="text-xs text-muted-foreground">
   780→                          ادامهٔ بررسی برای نمایش مواضع تأییدشدهٔ بیشتر از همین جست‌وجو.
   781→                        </p>
   782→                      </div>
   783→                    ) : null}
   784→                  </TabsContent>
   785→                  <TabsContent value="books">
   786→                    <BooksGroupedView
   787→                      sessionUid={sessionUid}
   788→                      onSessionExpired={() => void sessionQuery.refetch()}
   789→                      releaseUid={meta?.release_uid ?? null}
   790→                      onOpenSplit={openSplit}
   791→                      splitActiveUid={
   792→                        splitLocator ? splitLocator.key.split(":")[0] : null
   793→                      }
   794→                    />
   795→                  </TabsContent>
   796→                </Tabs>
   797→              )}
   798→            </div>
   799→          )}
   800→        </div>
   801→
   802→        {/* ستون کناری: پنل متن کنار نتایج (SEARCH-11) یا جست‌وجوهای ذخیره‌شده */}
   803→        {splitLocator ? (
   804→          <div className="min-w-0 xl:sticky xl:top-[76px]">
   805→            <SplitReaderPanel locator={splitLocator} onClose={closeSplit} onNavigate={navigateSplit} />
   806→            <p className="mt-2 text-[11px] leading-5 text-muted-foreground">
   807→              متن نتیجه در کنار نتایج باز است؛ مواضع، فیلترها و عبارت جست‌وجو حفظ شده‌اند (§۴۱).
   808→            </p>
   809→          </div>
   810→        ) : (
   811→          <aside aria-label="جست‌وجوهای ذخیره‌شده" className="min-w-0">
   812→            <SavedSearchesPanel
   813→              currentPayload={request}
   814→              saveEnabled={hasQuery}
   815→              saveOpen={saveOpen}
   816→              onSaveOpenChange={setSaveOpen}
   817→            />
   818→          </aside>
   819→        )}
   820→      </div>
   821→    </div>
   822→  );
   823→}
